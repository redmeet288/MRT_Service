    63→    63→  CANCELLED: "bg-terra hover:bg-terra-dark",
    64→    64→};
    65→    65→

    67→    67→const iso = (d: Date) => format(d, "yyyy-MM-dd");
    68→    68→
    69→    69→interface VisitForm {
    70→    70→  doctorId: string;
    71→    71→  lpuId: string; // "none" — ЛПУ не указан
    72→    72→  date: string; // yyyy-MM-dd
    73→    73→  purpose: string;
    74→    74→  comment: string;
    75→    75→  status: VisitStatus;
    76→    76→}
    77→    77→
    78→    78→export function CalendarView() {
    79→    79→  const qc = useQueryClient();
    80→    80→  const { data: session } = useSession();
    81→    81→  const isMP = (session?.user as { role?: Role } | undefined)?.role === "MP";
    82→    82→
    83→    83→  // —— Вид и навигация ——
    84→    84→  const [mode, setMode] = useState<ViewMode>("month");
    85→    85→  const [cursor, setCursor] = useState<Date>(new Date());
    86→    86→
    87→    87→  // —— Фильтры ("ALL" = без фильтра) ——
    88→    88→  const [mpFilter, setMpFilter] = useState("ALL");
    89→    89→  const [statusFilter, setStatusFilter] = useState("ALL");
    90→    90→  const [doctorFilter, setDoctorFilter] = useState("ALL");
    91→    91→
    92→    92→  // —— Диалог создания/редактирования ——
    93→    93→  const [dialogOpen, setDialogOpen] = useState(false);
    94→    94→  const [editing, setEditing] = useState<VisitItem | null>(null);
    95→    95→  const [form, setForm] = useState<VisitForm>({
    96→    96→    doctorId: "",
    97→    97→    lpuId: "none",
    98→    98→    date: format(new Date(), "yyyy-MM-dd"),
    99→    99→    purpose: "",
   100→   100→    comment: "",
   101→   101→    status: "PLANNED",
   102→   102→  });
   103→   103→
   104→   104→  // —— Справочники для фильтров и диалога ——
   105→   105→  const { data: users } = useQuery<UserBasic[]>({
   106→   106→    queryKey: ["users-basic"],
   107→   107→    queryFn: () => api.get("/api/app/users?basic=1"),
   108→   108→  });
   109→   109→  const mps = useMemo(() => (users ?? []).filter((u) => u.role === "MP"), [users]);
   110→   110→
   111→   111→  const { data: doctors } = useQuery<DoctorItem[]>({
   112→   112→    queryKey: ["doctors"],
   113→   113→    queryFn: () => api.get("/api/app/doctors"),
   114→   114→  });
   115→   115→  const doctorOptions = useMemo(
   116→   116→    () => (doctors ?? []).filter((d) => mpFilter === "ALL" || d.mpId === mpFilter),
   117→   117→    [doctors, mpFilter]
   118→   118→  );
   119→   119→
   120→   120→  // —— Период выборки (включая дни соседних месяцев в сетке) ——
   121→   121→  const range = useMemo(() => {
   122→   122→    if (mode === "month") {
   123→   123→      return {
   124→   124→        from: startOfWeek(startOfMonth(cursor), { weekStartsOn: 1 }),
   125→   125→        to: endOfWeek(endOfMonth(cursor), { weekStartsOn: 1 }),
   126→   126→      };
   127→   127→    }
   128→   128→    return {
   129→   129→      from: startOfWeek(cursor, { weekStartsOn: 1 }),
   130→   130→      to: endOfWeek(cursor, { weekStartsOn: 1 }),
   131→   131→    };
   132→   132→  }, [cursor, mode]);
   133→   133→
   134→   134→  // —— Визиты ——
   135→   135→  const visitsQuery = useQuery<VisitItem[]>({
   136→   136→    queryKey: [
   137→   137→      "visits",
   138→   138→      mode,
   139→   139→      iso(range.from),
   140→   140→      iso(range.to),
   141→   141→      mpFilter,
   142→   142→      statusFilter,
   143→   143→      doctorFilter,
   144→   144→    ],
   145→   145→    queryFn: () =>
   146→   146→      api.get<VisitItem[]>(
   147→   147→        `/api/app/visits${qs({
   148→   148→          from: iso(range.from),
   149→   149→          to: `${iso(range.to)}T23:59:59`,
   150→   150→          mpId: mpFilter === "ALL" ? undefined : mpFilter,
   151→   151→          status: statusFilter === "ALL" ? undefined : statusFilter,
   152→   152→          doctorId: doctorFilter === "ALL" ? undefined : doctorFilter,
   153→   153→        })}`
   154→   154→      ),
   155→   155→  });
   156→   156→
   157→   157→  // —— Группировка визитов по дням ——
   158→   158→  const grouped = useMemo(() => {
   159→   159→    const map = new Map<string, VisitItem[]>();
   160→   160→    (visitsQuery.data ?? []).forEach((v) => {
   161→   161→      const key = iso(new Date(v.date));
   162→   162→      const arr = map.get(key) ?? [];
   163→   163→      arr.push(v);
   164→   164→      map.set(key, arr);
   165→   165→    });
   166→   166→    return map;
   167→   167→  }, [visitsQuery.data]);
   168→   168→
   169→   169→  // —— Клетки месячной сетки (Пн–Вс) ——
   170→   170→  const days = useMemo(
   171→   171→    () => eachDayOfInterval({ start: range.from, end: range.to }),
   172→   172→    [range]
   173→   173→  );
   174→   174→
   175→   175→  const step = (dir: 1 | -1) =>
   176→   176→    setCursor(mode === "month" ? addMonths(cursor, dir) : addWeeks(cursor, dir));
   177→   177→
   178→   178→  // —— Диалог: открытие ——
   179→   179→  function openCreateForDay(day: Date) {
   180→   180→    setEditing(null);
   181→   181→    setForm({
   182→   182→      doctorId: "",
   183→   183→      lpuId: "none",
   184→   184→      date: iso(day),
   185→   185→      purpose: "",
   186→   186→      comment: "",
   187→   187→      status: "PLANNED",
   188→   188→    });
   189→   189→    setDialogOpen(true);
   190→   190→  }
   191→   191→  function openEdit(v: VisitItem) {
   192→   192→    setEditing(v);
   193→   193→    setForm({
   194→   194→      doctorId: v.doctorId,
   195→   195→      lpuId: v.lpuId ?? "none",
   196→   196→      date: iso(new Date(v.date)),
   197→   197→      purpose: v.purpose,
   198→   198→      comment: v.comment ?? "",
   199→   199→      status: v.status,
   200→   200→    });
   201→   201→    setDialogOpen(true);
   202→   202→  }
   203→   203→
   204→   204→  // —— Диалог: опции ЛПУ из привязок выбранного врача ——
   205→   205→  const selectedDoctor = (doctors ?? []).find((d) => d.id === form.doctorId);
   206→   206→  const lpuOptions = useMemo(() => {
   207→   207→    const opts: LpuWithCity[] = (selectedDoctor?.lpus ?? []).map((x) => x.lpu);
   208→   208→    const editLpu = editing?.lpu ?? null;
   209→   209→    if (editLpu && !opts.some((o) => o.id === editLpu.id)) {
   210→   210→      opts.unshift(editLpu);
   211→   211→    }
   212→   212→    return opts;
   213→   213→  }, [selectedDoctor, editing]);
   214→   214→
   215→   215→  // —— Мутации ——
   216→   216→  const save = useMutation({
   217→   217→    mutationFn: () => {
   218→   218→      const body = {
   219→   219→        doctorId: form.doctorId,
   220→   220→        lpuId: form.lpuId === "none" ? null : form.lpuId,
   221→   221→        date: form.date,
   222→   222→        purpose: form.purpose.trim(),
   223→   223→        comment: form.comment.trim() || null,
   224→   224→        status: form.status,
   225→   225→      };
   226→   226→      return editing
   227→   227→        ? api.put(`/api/app/visits/${editing.id}`, body)
   228→   228→        : api.post("/api/app/visits", body);
   229→   229→    },
   230→   230→    onSuccess: () => {
   231→   231→      toast.success(editing ? "Визит обновлен" : "Визит запланирован");
   232→   232→      setDialogOpen(false);
   233→   233→      qc.invalidateQueries({ queryKey: ["visits"] });
   234→   234→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   235→   235→    },
   236→   236→    onError: (e: Error) => toast.error(e.message),
   237→   237→  });
   238→   238→
   239→   239→  const remove = useMutation({
   240→   240→    mutationFn: (id: string) => api.del(`/api/app/visits/${id}`),
   241→   241→    onSuccess: () => {
   242→   242→      toast.success("Визит удален");
   243→   243→      setDialogOpen(false);
   244→   244→      qc.invalidateQueries({ queryKey: ["visits"] });
   245→   245→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   246→   246→    },
   247→   247→    onError: (e: Error) => toast.error(e.message),
   248→   248→  });
   249→   249→
   250→   250→  const periodTitle =
   251→   251→    mode === "month"
   252→   252→      ? cap(format(cursor, "LLLL yyyy", { locale: ru }))
   253→   253→      : `${cap(format(range.from, "d MMMM", { locale: ru }))} — ${format(
   254→   254→          range.to,
   255→   255→          "d MMMM yyyy",
   256→   256→          { locale: ru }
   257→   257→        )}`;
   258→   258→
   259→   259→  const noVisits =
   260→   260→    !visitsQuery.isLoading && !visitsQuery.error && (visitsQuery.data ?? []).length === 0;
   261→   261→
   262→   262→  return (
   263→   263→    <div className="space-y-6">
   264→   264→      {/* —— Фильтры —— */}
   265→   265→      <SectionFilters
   266→   266→        mps={mps}
   267→   267→        mpFilter={mpFilter}
   268→   268→        setMpFilter={setMpFilter}
   269→   269→        statusFilter={statusFilter}
   270→   270→        setStatusFilter={setStatusFilter}
   271→   271→        doctorOptions={doctorOptions}
   272→   272→        doctorFilter={doctorFilter}
   273→   273→        setDoctorFilter={setDoctorFilter}
   274→   274→        onCreate={() => openCreateForDay(new Date())}
   275→   275→      />
   276→   276→
   277→   277→      {/* —— Календарь —— */}
   278→   278→      <section className="rounded-xl border border-border bg-white p-6 shadow-[0_2px_12px_rgba(25,70,136,0.05)]">
   279→   279→        {/* Навигация периода */}
   280→   280→        <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
   281→   281→          <div className="flex items-center gap-2">
   282→   282→            <Button
   283→   283→              variant="ghost"
   284→   284→              size="icon"
   285→   285→              className="h-8 w-8 text-brand hover:bg-sky-light"
   286→   286→              onClick={() => step(-1)}
   287→   287→              aria-label="Предыдущий период"
   288→   288→            >
   289→   289→              <ChevronLeft className="h-4 w-4" strokeWidth={2} />
   290→   290→            </Button>
   291→   291→            <h2 className="font-display text-lg font-bold text-brand min-w-[190px] text-center">
   292→   292→              {periodTitle}
   293→   293→            </h2>
   294→   294→            <Button
   295→   295→              variant="ghost"
   296→   296→              size="icon"
   297→   297→              className="h-8 w-8 text-brand hover:bg-sky-light"
   298→   298→              onClick={() => step(1)}
   299→   299→              aria-label="Следующий период"
   300→   300→            >
   301→   301→              <ChevronRight className="h-4 w-4" strokeWidth={2} />
   302→   302→            </Button>
   303→   303→            <Button
   304→   304→              variant="outline"
   305→   305→              size="sm"
   306→   306→              className="ml-2 h-8 border-sky/40 text-brand hover:bg-sky-light"
   307→   307→              onClick={() => setCursor(new Date())}
   308→   308→            >
   309→   309→              Сегодня
   310→   310→            </Button>
   311→   311→          </div>
   312→   312→          <ToggleGroup
   313→   313→            type="single"
   314→   314→            size="sm"
   315→   315→            value={mode}
   316→   316→            onValueChange={(v) => {
   317→   317→              if (v === "month" || v === "week") setMode(v);
   318→   318→            }}
   319→   319→            className="rounded-lg border border-border bg-soft p-1"
   320→   320→          >
   321→   321→            <ToggleGroupItem
   322→   322→              value="month"
   323→   323→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   324→   324→            >
   325→   325→              <CalendarDays className="h-3.5 w-3.5" strokeWidth={2} /> Месяц
   326→   326→            </ToggleGroupItem>
   327→   327→            <ToggleGroupItem
   328→   328→              value="week"
   329→   329→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   330→   330→            >
   331→   331→              <CalendarRange className="h-3.5 w-3.5" strokeWidth={2} /> Неделя
   332→   332→            </ToggleGroupItem>
   333→   333→          </ToggleGroup>
   334→   334→        </div>
   335→   335→
   336→   336→        {visitsQuery.error ? (
   337→   337→          <div className="rounded-xl border border-terra/50 bg-terra-light p-6 text-sm text-brand-dark">
   338→   338→            Не удалось загрузить визиты: {(visitsQuery.error as Error).message}
   339→   339→          </div>
   340→   340→        ) : visitsQuery.isLoading ? (
   341→   341→          <RowsSkeleton rows={6} />
   342→   342→        ) : (
   343→   343→          <>
   344→   344→            {noVisits && (
   345→   345→              <EmptyState
   346→   346→                title="Визитов в выбранном периоде нет"
   347→   347→                hint="Кликните по дню в календаре или нажмите «Запланировать визит»"
   348→   348→                className="py-6"
   349→   349→              />
   350→   350→            )}
   351→   351→
   352→   352→            {mode === "month" ? (
   353→   353→              <>
   354→   354→                {/* Шапка дней недели */}
   355→   355→                <div className="mb-2 grid grid-cols-7 gap-2">
   356→   356→                  {WEEKDAYS.map((w) => (
   357→   357→                    <div
   358→   358→                      key={w}
   359→   359→                      className="text-center text-[11px] font-semibold uppercase tracking-wide text-sky"
   360→   360→                    >
   361→   361→                      {w}
   362→   362→                    </div>
   363→   363→                  ))}
   364→   364→                </div>
   365→   365→                {/* Сетка месяца */}
   366→   366→                <div className="grid grid-cols-7 gap-2">
   367→   367→                  {days.map((day) => {
   368→   368→                    const key = iso(day);
   369→   369→                    const list = grouped.get(key) ?? [];
   370→   370→                    const dim = !isSameMonth(day, cursor);
   371→   371→                    return (
   372→   372→                      <div
   373→   373→                        key={key}
   374→   374→                        onClick={() => openCreateForDay(day)}
   375→   375→                        className={cn(
   376→   376→                          "group/cell flex min-h-[112px] cursor-pointer flex-col gap-1 rounded-lg border p-1.5 transition-colors",
   377→   377→                          dim ? "border-border/60 bg-soft" : "border-border bg-white hover:border-sky/50",
   378→   378→                          isToday(day) && "border-azure/60 bg-sky-light/50 ring-1 ring-azure"
   379→   379→                        )}
   380→   380→                      >
   381→   381→                        <div className="flex items-center justify-between">
   382→   382→                          <span
   383→   383→                            className={cn(
   384→   384→                              "inline-flex h-5 w-5 items-center justify-center rounded-full font-display text-[11px] font-bold",
   385→   385→                              isToday(day)
   386→   386→                                ? "bg-azure text-white"
   387→   387→                                : dim
   388→   388→                                  ? "text-muted-foreground/50"
   389→   389→                                  : "text-brand"
   390→   390→                            )}
   391→   391→                          >
   392→   392→                            {day.getDate()}
   393→   393→                          </span>
   394→   394→                          <button
   395→   395→                            type="button"
   396→   396→                            onClick={(e) => {
   397→   397→                              e.stopPropagation();
   398→   398→                              openCreateForDay(day);
   399→   399→                            }}
   400→   400→                            className="text-sky opacity-0 transition-opacity hover:text-azure group-hover/cell:opacity-100"
   401→   401→                            aria-label={`Запланировать визит на ${key}`}
   402→   402→                          >
   403→   403→                            <Plus className="h-3.5 w-3.5" strokeWidth={2} />
   404→   404→                          </button>
   405→   405→                        </div>
   406→   406→                        <div className="flex flex-col gap-1">
   407→   407→                          {list.slice(0, 3).map((v) => (
   408→   408→                            <button
   409→   409→                              key={v.id}
   410→   410→                              type="button"
   411→   411→                              onClick={(e) => {
   412→   412→                                e.stopPropagation();
   413→   413→                                openEdit(v);
   414→   414→                              }}
   415→   415→                              title={`${v.doctor?.fullName ?? "—"} · ${v.purpose}`}
   416→   416→                              className={cn(
   417→   417→                                "w-full truncate rounded-md px-1.5 py-0.5 text-left text-[10px] font-medium text-white transition-opacity",
   418→   418→                                CHIP_BG[v.status],
   419→   419→                                dim && "opacity-60"
   420→   420→                              )}
   421→   421→                            >
   422→   422→                              {v.doctor?.fullName ?? "Врач"}
   423→   423→                            </button>
   424→   424→                          ))}
   425→   425→                          {list.length > 3 && (
   426→   426→                            <span className="px-1 font-display text-[10px] font-semibold text-sky">
   427→   427→                              +{list.length - 3} ещё
   428→   428→                            </span>
   429→   429→                          )}
   430→   430→                        </div>
   431→   431→                      </div>
   432→   432→                    );
   433→   433→                  })}
   434→   434→                </div>
   435→   435→              </>
   436→   436→            ) : (
   437→   437→              /* —— Сетка недели —— */
   438→   438→              <div className="grid grid-cols-7 gap-3">
   439→   439→                {days.map((day) => {
   440→   440→                  const key = iso(day);
   441→   441→                  const list = grouped.get(key) ?? [];
   442→   442→                  return (
   443→   443→                    <div
   444→   444→                      key={key}
   445→   445→                      className={cn(
   446→   446→                        "flex min-h-[400px] flex-col rounded-lg border",
   447→   447→                        isToday(day)
   448→   448→                          ? "border-azure/60 bg-sky-light/40"
   449→   449→                          : "border-border bg-white"
   450→   450→                      )}
   451→   451→                    >
   452→   452→                      <div className="border-b border-border/70 px-2 py-2 text-center">
   453→   453→                        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   454→   454→                          {cap(format(day, "EEEEEE", { locale: ru }))}
   455→   455→                        </div>
   456→   456→                        <div
