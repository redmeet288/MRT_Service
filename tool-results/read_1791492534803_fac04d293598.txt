   129→/* ============================================================
   130→   Транслитерация для PDF (jsPDF стандартно не поддерживает
   131→   кириллицу) — простая карта а-я → a-ya
   132→   ============================================================ */
   133→const TRANSLIT: Record<string, string> = {
   134→  а: "a", б: "b", в: "v", г: "g", д: "d", е: "e", ё: "e", ж: "zh", з: "z",
   135→  и: "i", й: "y", к: "k", л: "l", м: "m", н: "n", о: "o", п: "p", р: "r",
   136→  с: "s", т: "t", у: "u", ф: "f", х: "kh", ц: "ts", ч: "ch", ш: "sh",
   137→  щ: "sch", ъ: "", ы: "y", ь: "", э: "e", ю: "yu", я: "ya",
   138→};
   139→
   140→function translit(value: string): string {
   141→  return (value ?? "")
   142→    .split("")
   143→    .map((ch) => {
   144→      const lower = ch.toLowerCase();
   145→      const t = TRANSLIT[lower];
   146→      if (t === undefined) return ch; // латиница, цифры, пунктуация — как есть
   147→      if (t === "") return "";
   148→      if (ch !== lower) return t.charAt(0).toUpperCase() + t.slice(1);
   149→      return t;
   150→    })
   151→    .join("");
   152→}

   180→type GroupCtx = { mp?: string; lpu?: string; city?: string; doctor?: string; nom?: string };

   182→function collectRows(node: ReportGroupNode, ctx: GroupCtx, out: ExportRow[]): void {
   183→  const next: GroupCtx = { ...ctx };
   184→  if (node.level === "mp") next.mp = node.name;
   185→  else if (node.level === "lpu") {
   186→    next.lpu = node.name;
   187→    next.city = node.subtitle || ctx.city || "";
   188→  } else if (node.level === "doctor") next.doctor = node.name;
   189→  else next.nom = node.name;
   190→
   191→  if (node.children.length) {
   192→    node.children.forEach((child) => collectRows(child, next, out));
   193→  } else {
   194→    node.requests.forEach((r) =>
   195→      out.push({
   196→        mp: next.mp ?? r.mpName ?? "",
   197→        lpu: next.lpu ?? r.lpuName ?? "",
   198→        city: next.city ?? r.cityName ?? "",
   199→        doctor: next.doctor ?? r.doctorName ?? "",
   200→        nom: next.nom ?? r.assigned ?? "",
   201→        date: fmtDate(r.date),
   202→        patient: r.patientName,
   203→        destination: r.destination ?? "—",
   204→        assigned: r.assigned ?? "—",
   205→        done: r.done ?? "Не выполнено",
   206→      })
   207→    );
   208→  }
   209→}

   499→  // —— Управление деревом (контролируемое, ключ уникален по пути) ——
   500→  function toggleNode(key: string, depth: number) {
   501→    setTreeState((prev) => {
   502→      const open = new Set(prev.open);
   503→      const closed = new Set(prev.closed);
   504→      const isOpen = !closed.has(key) && (depth === 0 || open.has(key));
   505→      if (isOpen) {
   506→        open.delete(key);
   507→        closed.add(key);
   508→      } else {
   509→        open.add(key);
   510→        closed.delete(key);
   511→      }
   512→      return { open, closed };
   513→    });
   514→  }

  1090→function TreeRows({
  1091→  node,
  1092→  depth,
  1093→  parentKey,
  1094→  detailSpan,
  1095→  show,
  1096→  expansion,
  1097→  onToggle,
  1098→  detailSort,
  1099→  onDetailSort,
  1100→}: {
  1101→  node: ReportGroupNode;
  1102→  depth: number;
  1103→  parentKey: string;
  1104→  detailSpan: number;
  1105→  show: { nv: boolean; lpu: boolean; mp: boolean };
  1106→  expansion: TreeExpansion;
  1107→  onToggle: (key: string, depth: number) => void;
  1108→  detailSort?: SortState | null;
  1109→  onDetailSort?: (field: string) => void;
  1110→}) {
  1111→  // Уникальный ключ по пути в дереве: level:id на каждом уровне
  1112→  const key = `${parentKey}/${depth}:${node.level}:${node.id}`;
  1113→  const open = !expansion.closed.has(key) && (depth === 0 || expansion.open.has(key));
  1114→  const hasChildren = node.children.length > 0;
  1115→  const hasContent = hasChildren || node.requests.length > 0;
  1116→  const Icon = LEVEL_ICONS[node.level] ?? Building2;
  1117→  const p = pct(node.done, node.total);
  1118→
  1119→  // Детальные строки листа: клиентская сортировка по столбцам детализации
  1120→  const sortedRequests = useMemo(
  1121→    () => applySort(node.requests, detailSort ?? null, detailValue),
  1122→    [node.requests, detailSort]
  1123→  );
  1124→
  1125→  return (
  1126→    <>
  1127→      <tr className="onec-row-group">
  1128→        <td colSpan={detailSpan}>
  1129→          <div className="flex items-center gap-1.5" style={{ paddingLeft: depth * 20 }}>
  1130→            {hasContent ? (
  1131→              <button
  1132→                type="button"
  1133→                className="onec-tree-toggle"
  1134→                aria-expanded={open}
  1135→                aria-label={open ? `Свернуть группу «${node.name}»` : `Развернуть группу «${node.name}»`}
  1136→                onClick={() => onToggle(key, depth)}
  1137→              >
  1138→                {open ? "−" : "+"}
  1139→              </button>
  1140→            ) : (
  1141→              <span className="inline-block h-[15px] w-[15px] shrink-0" aria-hidden />
  1142→            )}
  1143→            <Icon className="h-3.5 w-3.5 shrink-0 text-brand" strokeWidth={2} />
  1144→            <span className="text-[13px] text-[#1c1c1c]">{node.name}</span>
  1145→            {node.subtitle && (
  1146→              <span className="font-normal text-[11px] text-[#8a8a8a]">· {node.subtitle}</span>
  1147→            )}
  1148→          </div>
  1149→        </td>
  1150→        <td style={{ textAlign: "right" }}>{node.total}</td>
  1151→        <td style={{ textAlign: "right" }}>{node.done}</td>
  1152→        <td style={{ textAlign: "right" }}>
  1153→          <span
  1154→            className={cn(
  1155→              "inline-block rounded-full px-1.5 py-[1px] text-[10px] font-semibold",
  1156→              pctBadgeClass(p)
  1157→            )}
  1158→          >
  1159→            {p}%
  1160→          </span>
  1161→        </td>
  1162→      </tr>
  1163→
  1164→      {/* Дочерние группы */}
  1165→      {open &&
  1166→        hasChildren &&
  1167→        node.children.map((child) => (
  1168→          <TreeRows
  1169→            key={child.id}
  1170→            node={child}
  1171→            depth={depth + 1}
  1172→            parentKey={key}
  1173→            detailSpan={detailSpan}
  1174→            show={show}
  1175→            expansion={expansion}
  1176→            onToggle={onToggle}
  1177→            detailSort={detailSort}
  1178→            onDetailSort={onDetailSort}
  1179→          />
  1180→        ))}
  1181→
  1182→      {/* Листья: детальные строки заявок */}
  1183→      {open &&
  1184→        !hasChildren &&
  1185→        (node.requests.length ? (
  1186→          sortedRequests.map((r) => <DetailRow key={r.id} r={r} show={show} />)
  1187→        ) : (
  1188→          <tr>
  1189→            <td colSpan={detailSpan + 3} className="italic text-[#9a9a9a]" style={{ fontSize: 12 }}>
  1190→              Нет записей в группе
  1191→            </td>
  1192→          </tr>
  1193→        ))}
  1194→    </>
  1195→  );
  1196→}
