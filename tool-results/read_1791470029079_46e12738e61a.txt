    80→/** Сегодняшняя дата в формате YYYY-MM-DD (локальная зона) */
    81→function todayISO(): string {
    82→  const d = new Date();
    83→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);

   329→interface DoctorFormState {
   330→  fullName: string;
   331→  phone: string;
   332→  birthDate: string;
   333→  statusId: string;
   334→  mpId: string;
   335→  cityId: string;
   336→  lpuIds: string[];
   337→  comment: string;
   338→}
   339→
   340→const emptyDoctorForm: DoctorFormState = {
   341→  fullName: "",
   342→  phone: "",
   343→  birthDate: "",
   344→  statusId: NONE,
   345→  mpId: "",
   346→  cityId: NONE,
   347→  lpuIds: [],
   348→  comment: "",
   349→};
   350→
   351→/** Начальное состояние формы: из карточки при редактировании / пустая при создании */
   352→function initDoctorForm(doctor: EditableDoctor | null): DoctorFormState {
   353→  if (!doctor) return emptyDoctorForm;
   354→  return {
   355→    fullName: doctor.fullName,
   356→    phone: doctor.phone ?? "",
   357→    birthDate: doctor.birthDate ? doctor.birthDate.slice(0, 10) : "",
   358→    statusId: doctor.statusId ?? NONE,
   359→    mpId: doctor.mpId,
   360→    cityId: doctor.cityId ?? NONE,
   361→    lpuIds: doctor.lpus?.map((x) => x.lpu.id) ?? [],
   362→    comment: doctor.comment ?? "",
   363→  };
   364→}
   365→
   366→function DoctorFormDialog({
   367→  onOpenChange,
   368→  doctor,
   369→}: {
   370→  onOpenChange: (o: boolean) => void;
   371→  doctor: EditableDoctor | null;
   372→}) {
   373→  const qc = useQueryClient();
   374→  const [form, setForm] = useState<DoctorFormState>(() => initDoctorForm(doctor));
   375→
   376→  const { data: users } = useQuery<UserBasic[]>({
   377→    queryKey: ["users-basic"],
   378→    queryFn: () => api.get("/api/app/users?basic=1"),
   379→  });
   380→  const { data: statuses } = useQuery<CooperationStatus[]>({
   381→    queryKey: ["statuses"],
   382→    queryFn: () => api.get("/api/app/statuses"),
   383→  });
   384→  const { data: cities } = useQuery<City[]>({
   385→    queryKey: ["cities"],
   386→    queryFn: () => api.get("/api/app/cities"),
   387→  });
   388→  const { data: lpus } = useQuery<CityData[]>({
   389→    queryKey: ["lpus"],
   390→    queryFn: () => api.get("/api/app/lpus"),
   391→  });
   392→
   393→  const mps = useMemo(() => (users ?? []).filter((u) => u.role === "MP"), [users]);
   394→
   395→  /** Переключение ЛПУ; автоподстановка города первого ЛПУ, если город не указан */
   396→  function toggleLpu(id: string, checked: boolean) {
   397→    setForm((f) => {
   398→      const lpuIds = checked ? [...f.lpuIds, id] : f.lpuIds.filter((x) => x !== id);
   399→      let cityId = f.cityId;
   400→      if (checked && cityId === NONE && lpuIds.length > 0) {
   401→        const first = lpus?.find((l) => l.id === lpuIds[0]);
   402→        if (first) cityId = first.city.id;
   403→      }
   404→      return { ...f, lpuIds, cityId };
   405→    });
   406→  }
   407→
   408→  const save = useMutation({
   409→    mutationFn: () => {
   410→      const body = {
   411→        fullName: form.fullName.trim(),
   412→        phone: form.phone.trim() || null,
   413→        birthDate: form.birthDate || null,
   414→        statusId: form.statusId === NONE ? null : form.statusId,
   415→        mpId: form.mpId,
   416→        cityId: form.cityId === NONE ? null : form.cityId,
   417→        lpuIds: form.lpuIds,
   418→        comment: form.comment.trim() || null,
   419→      };
   420→      return doctor ? api.put(`/api/app/doctors/${doctor.id}`, body) : api.post("/api/app/doctors", body);
   421→    },
   422→    onSuccess: () => {
   423→      toast.success(doctor ? "Карточка врача обновлена" : "Врач добавлен");
   424→      qc.invalidateQueries({ queryKey: ["doctors"] });
   425→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   426→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   427→      onOpenChange(false);
   428→    },
   429→    onError: (e: Error) => toast.error(e.message),
   430→  });
   431→
   432→  return (
   433→    <Dialog open onOpenChange={onOpenChange}>
   434→      <DialogContent className="sm:max-w-2xl">
   435→        <DialogHeader>
   436→          <DialogTitle className="font-display text-brand">
   437→            {doctor ? "Редактирование врача" : "Новый направляющий врач"}
   438→          </DialogTitle>
   439→          <DialogDescription>
   440→            Карточка НВ — ключевая сущность CRM: статус, курирующий МП и привязки к ЛПУ
   441→          </DialogDescription>
   442→        </DialogHeader>
   443→
   444→        <div className="grid gap-4 py-1 md:grid-cols-2">
   445→          <div className="space-y-1.5">
   446→            <Label>
   447→              ФИО <span className="text-terra-dark">*</span>
   448→            </Label>
   449→            <Input
   450→              value={form.fullName}
   451→              onChange={(e) => setForm({ ...form, fullName: e.target.value })}
   452→              placeholder="Иванова Анна Петровна"
   453→            />
   454→          </div>
   455→          <div className="space-y-1.5">
   456→            <Label>Телефон</Label>
   457→            <Input
   458→              type="tel"
   459→              value={form.phone}
   460→              onChange={(e) => setForm({ ...form, phone: e.target.value })}
   461→              placeholder="+7 (___) ___-__-__"
   462→            />
   463→          </div>
   464→          <div className="space-y-1.5">
   465→            <Label>Дата рождения</Label>
   466→            <Input
   467→              type="date"
   468→              value={form.birthDate}
   469→              onChange={(e) => setForm({ ...form, birthDate: e.target.value })}
   470→            />
   471→          </div>
   472→          <div className="space-y-1.5">
   473→            <Label>Статус сотрудничества</Label>
   474→            <Select value={form.statusId} onValueChange={(v) => setForm({ ...form, statusId: v })}>
   475→              <SelectTrigger className="bg-white">
   476→                <SelectValue placeholder="Выберите статус" />
   477→              </SelectTrigger>
   478→              <SelectContent>
   479→                <SelectItem value={NONE}>Не указан</SelectItem>
   480→                {(statuses ?? []).map((s) => (
   481→                  <SelectItem key={s.id} value={s.id}>
   482→                    {s.name}
   483→                  </SelectItem>
   484→                ))}
   485→              </SelectContent>
   486→            </Select>
   487→          </div>
   488→          <div className="space-y-1.5">
   489→            <Label>
   490→              Курирующий МП <span className="text-terra-dark">*</span>
   491→            </Label>
   492→            <Select value={form.mpId} onValueChange={(v) => setForm({ ...form, mpId: v })}>
   493→              <SelectTrigger className="bg-white">
   494→                <SelectValue placeholder="Выберите МП" />
   495→              </SelectTrigger>
   496→              <SelectContent>
   497→                {mps.map((m) => (
   498→                  <SelectItem key={m.id} value={m.id}>
   499→                    {m.name}
   500→                  </SelectItem>
   501→                ))}
   502→              </SelectContent>
   503→            </Select>
   504→          </div>
   505→          <div className="space-y-1.5">
   506→            <Label>Город</Label>
   507→            <Select value={form.cityId} onValueChange={(v) => setForm({ ...form, cityId: v })}>
   508→              <SelectTrigger className="bg-white">
   509→                <SelectValue placeholder="Не указан" />
   510→              </SelectTrigger>
   511→              <SelectContent>
   512→                <SelectItem value={NONE}>Не указан</SelectItem>
   513→                {(cities ?? []).map((c) => (
   514→                  <SelectItem key={c.id} value={c.id}>
   515→                    {c.name}
   516→                  </SelectItem>
   517→                ))}
   518→              </SelectContent>
   519→            </Select>
   520→          </div>
   521→
   522→          {/* Множественный выбор ЛПУ */}
   523→          <div className="space-y-1.5 md:col-span-2">
   524→            <div className="flex items-center justify-between">
   525→              <Label>Привязка к ЛПУ</Label>
   526→              <span className="text-[11px] text-muted-foreground">Выбрано: {form.lpuIds.length}</span>
   527→            </div>
   528→            <div className="max-h-48 space-y-1 overflow-y-auto rounded-lg border border-border bg-white p-3">
   529→              {!lpus?.length ? (
   530→                <p className="py-2 text-center text-xs text-muted-foreground">
   531→                  Сначала добавьте ЛПУ в разделе «Справочники»
   532→                </p>
   533→              ) : (
   534→                lpus.map((l) => (
   535→                  <label
   536→                    key={l.id}
