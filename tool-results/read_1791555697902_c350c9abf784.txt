    58→/** Сегодняшняя дата в формате YYYY-MM-DD (локальная зона) */
    59→function todayISO(): string {
    60→  const d = new Date();
    61→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);
    62→}
    63→
    64→/** ISO-дата (из drill-down дашборда) → yyyy-MM-dd для input[type=date] */
    65→function isoToYMD(iso: string | undefined): string {
    66→  if (!iso) return "";
    67→  const d = new Date(iso);
    68→  return Number.isNaN(d.getTime()) ? "" : format(d, "yyyy-MM-dd");
    69→}

   466→/* ============================================================
   467→   Диалог регистрации / редактирования заявки (строго по ТЗ)
   468→   ============================================================ */
   469→interface RequestFormState {
   470→  date: string;
   471→  patientName: string;
   472→  doctorId: string;
   473→  lpuId: string;
   474→  destination: string;
   475→  /** Мультивыбор из номенклатуры: «Что назначено» — обязателен, «Что сделано» — нет */
   476→  assignedIds: string[];
   477→  doneIds: string[];
   478→  comment: string;
   479→}
   480→
   481→/** Переключение позиции в списке множественного выбора */
   482→function toggleInList(list: string[], id: string, checked: boolean): string[] {
   483→  return checked
   484→    ? list.includes(id)
   485→      ? list
   486→      : [...list, id]
   487→    : list.filter((x) => x !== id);
   488→}
   489→
   490→/** Начальное состояние формы: из заявки при редактировании / пустая (дата — сегодня) при создании */
   491→function initRequestForm(request: RequestItem | null): RequestFormState {
   492→  if (!request) {
   493→    return {
   494→      date: todayISO(),
   495→      patientName: "",
   496→      doctorId: "",
   497→      lpuId: "",
   498→      destination: "",
   499→      assignedIds: [],
   500→      doneIds: [],
   501→      comment: "",
   502→    };
   503→  }
   504→  return {
   505→    date: request.date.slice(0, 10),
   506→    patientName: request.patientName,
   507→    doctorId: request.doctorId ?? "",
   508→    lpuId: request.lpuId ?? "",
   509→    destination: request.destination ?? "",
   510→    assignedIds: request.assignedItems.map((i) => i.id),
   511→    doneIds: request.doneItems.map((i) => i.id),
   512→    comment: request.comment ?? "",
   513→  };
   514→}
   515→
   516→function RequestFormDialog({
   517→  request,
   518→  onOpenChange,
   519→}: {
   520→  request: RequestItem | null;
   521→  onOpenChange: (o: boolean) => void;
   522→}) {
   523→  const qc = useQueryClient();
   524→  const [form, setForm] = useState<RequestFormState>(() => initRequestForm(request));
   525→
   526→  const { data: doctors } = useQuery<DoctorItem[]>({
   527→    queryKey: ["doctors", "all"],
   528→    queryFn: () => api.get("/api/app/doctors"),
   529→  });
   530→  const { data: nomenclature } = useQuery<NomenclatureItem[]>({
   531→    queryKey: ["nomenclature", "active"],
   532→    queryFn: () => api.get("/api/app/nomenclature?active=1"),
   533→  });
   534→  const { data: lpus } = useQuery<CityData[]>({
   535→    queryKey: ["lpus"],
   536→    queryFn: () => api.get("/api/app/lpus"),
   537→  });
   538→
   539→  // Варианты НВ: активные + текущий врач редактируемой заявки (если помечен на удаление)
   540→  const doctorOptions = useMemo<Array<Pick<DoctorItem, "id" | "fullName">>>(() => {
   541→    const list = (doctors ?? [])
   542→      .filter((d) => d.isActive)
   543→      .map((d) => ({ id: d.id, fullName: d.fullName }));
   544→    const cur = request?.doctor;
   545→    if (cur && !list.some((d) => d.id === cur.id)) {
   546→      list.unshift({ id: cur.id, fullName: cur.fullName });
   547→    }
   548→    return list;
   549→  }, [doctors, request]);
   550→
   551→  const selectedDoctor = useMemo(
   552→    () => doctors?.find((d) => d.id === form.doctorId) ?? null,
   553→    [doctors, form.doctorId]
   554→  );
   555→
   556→  // ЛПУ, привязанные к выбранному НВ (данные lpus уже приходят в ответе doctors)
   557→  const doctorLpus = useMemo(
   558→    () => (selectedDoctor?.lpus ?? []).map((x) => x.lpu).filter((l) => l.isActive !== false),
   559→    [selectedDoctor]
   560→  );
   561→
   562→  // Свободные варианты ЛПУ (без НВ / у НВ нет привязок): активные + текущее ЛПУ заявки
   563→  const freeLpuOptions = useMemo<CityData[]>(() => {
   564→    const list = (lpus ?? []).filter((l) => l.isActive);
   565→    if (request?.lpu) {
   566→      const cur: CityData = {
   567→        id: request.lpu.id,
   568→        name: request.lpu.name,
   569→        city: request.lpu.city,
   570→        address: request.lpu.address,
   571→        isActive: request.lpu.isActive ?? true,
   572→      };
   573→      if (!list.some((l) => l.id === cur.id)) list.unshift(cur);
   574→    }
   575→    return list;
   576→  }, [lpus, request]);
   577→
   578→  // Режим выбора ЛПУ: 1 привязка — авто; несколько — Select из привязок; 0/без НВ — свободный Select
   579→  const lpuMode: "single" | "multi" | "free" = !selectedDoctor
   580→    ? "free"
   581→    : doctorLpus.length === 1
   582→      ? "single"
   583→      : doctorLpus.length > 1
   584→        ? "multi"
   585→        : "free";
   586→
   587→  // Варианты для режима «несколько ЛПУ» (+ защита для редактируемой заявки)
   588→  const multiOptions = useMemo(() => {
   589→    const opts = [...doctorLpus];
   590→    if (request?.lpu && request.lpuId && !opts.some((l) => l.id === request.lpu?.id)) {
   591→      opts.push(request.lpu);
   592→    }
   593→    return opts;
   594→  }, [doctorLpus, request]);
   595→
   596→  // Итоговый lpuId: при единственном ЛПУ НВ — подставляется автоматически
   597→  const resolvedLpuId = lpuMode === "single" ? (doctorLpus[0]?.id ?? "") : form.lpuId;
   598→
   599→  const lpuCity = useMemo(() => {
   600→    if (!resolvedLpuId) return null;
   601→    const fromDoctor = doctorLpus.find((l) => l.id === resolvedLpuId);
   602→    if (fromDoctor) return fromDoctor.city.name;
   603→    if (request?.lpu && request.lpu.id === resolvedLpuId) return request.lpu.city.name;
   604→    return lpus?.find((l) => l.id === resolvedLpuId)?.city.name ?? null;
   605→  }, [resolvedLpuId, doctorLpus, lpus, request]);
   606→
   607→  const save = useMutation({
   608→    mutationFn: () => {
   609→      const body = {
   610→        date: form.date,
   611→        patientName: form.patientName.trim(),
   612→        // НВ и ЛПУ необязательны (доработка): обязателен только пациент
   613→        doctorId: form.doctorId || null,
   614→        lpuId: resolvedLpuId || null,
   615→        destination: form.destination.trim() || "МРТ ТОЧНО",
   616→        assignedIds: form.assignedIds,
   617→        doneIds: form.doneIds,
   618→        comment: form.comment.trim() || null,
   619→      };
   620→      return request
   621→        ? api.put(`/api/app/requests/${request.id}`, body)
   622→        : api.post("/api/app/requests", body);
   623→    },
   624→    onSuccess: () => {
   625→      toast.success(request ? "Заявка обновлена" : "Заявка зарегистрирована");
   626→      qc.invalidateQueries({ queryKey: ["requests"] });
   627→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   628→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   629→      onOpenChange(false);
   630→    },
   631→    onError: (e: Error) => toast.error(e.message),
   632→  });
   633→
   634→  return (
   635→    <Dialog open onOpenChange={onOpenChange}>
   636→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-lg">
   637→        <DialogHeader>
   638→          <DialogTitle className="font-display text-brand">
   639→            {request ? "Карточка заявки" : "Регистрация заявки"}
   640→          </DialogTitle>
   641→          <DialogDescription>
   642→            {request
   643→              ? "Просмотр и редактирование направления пациента на обследование"
   644→              : "Учет направления пациента на обследование"}
   645→          </DialogDescription>
   646→        </DialogHeader>
   647→
   648→        <div className="space-y-4 py-1">
   649→          {/* 1. Дата */}
   650→          <div className="space-y-1.5">
   651→            <Label>
   652→              Дата <span className="text-azure-dark">*</span>
   653→            </Label>
   654→            <Input
   655→              type="date"
   656→              value={form.date}
   657→              onChange={(e) => setForm({ ...form, date: e.target.value })}
   658→            />
   659→          </div>
   660→
   661→          {/* 2. ФИО пациента */}
   662→          <div className="space-y-1.5">
   663→            <Label>
   664→              ФИО пациента <span className="text-azure-dark">*</span>
   665→            </Label>
   666→            <Input
   667→              value={form.patientName}
   668→              onChange={(e) => setForm({ ...form, patientName: e.target.value })}
   669→              placeholder="Иванов Иван Иванович"
   670→            />
   671→          </div>
   672→
   673→          {/* 3. ФИО НВ — необязательно: пациент может не помнить врача */}
   674→          <div className="space-y-1.5">
   675→            <Label>ФИО НВ</Label>
   676→            <Select
   677→              value={form.doctorId || NONE}
   678→              onValueChange={(v) => {
   679→                const nd = doctors?.find((d) => d.id === v);
   680→                const dl = (nd?.lpus ?? []).map((x) => x.lpu).filter((l) => l.isActive !== false);
   681→                setForm((f) => ({
   682→                  ...f,
   683→                  doctorId: v === NONE ? "" : v,
   684→                  lpuId: v === NONE ? "" : dl.length === 1 ? dl[0].id : "",
   685→                }));
   686→              }}
   687→            >
   688→              <SelectTrigger className="bg-white">
   689→                <SelectValue placeholder="Не указан" />
   690→              </SelectTrigger>
   691→              <SelectContent className="max-h-72">
   692→                <SelectItem value={NONE}>Не указан</SelectItem>
   693→                {doctorOptions.map((d) => (
   694→                  <SelectItem key={d.id} value={d.id}>
   695→                    {d.fullName}
   696→                  </SelectItem>
   697→                ))}
   698→              </SelectContent>
   699→            </Select>
   700→            <p className="text-[11px] text-muted-foreground">
   701→              Если пациент не помнит врача — оставьте «Не указан», информацию можно указать в комментарии
   702→            </p>
   703→          </div>
   704→
   705→          {/* 4. ЛПУ и город — автозаполнение из карточки НВ (необязательно) */}
   706→          <div className="space-y-1.5">
   707→            <Label>ЛПУ</Label>
   708→            {lpuMode === "single" && (
   709→              <>
   710→                <Input
   711→                  disabled
   712→                  readOnly
   713→                  value={
   714→                    doctorLpus[0] ? `${doctorLpus[0].name} · ${doctorLpus[0].city.name}` : "—"
   715→                  }
   716→                  className="bg-white text-muted-foreground"
   717→                />
   718→                <p className="text-[11px] text-muted-foreground">
   719→                  Единственное активное ЛПУ врача — подставлено автоматически
   720→                </p>
   721→              </>
   722→            )}
   723→            {lpuMode === "multi" && (
   724→              <Select
   725→                value={form.lpuId || NONE}
   726→                onValueChange={(v) => setForm({ ...form, lpuId: v === NONE ? "" : v })}
   727→              >
   728→                <SelectTrigger className="bg-white">
   729→                  <SelectValue placeholder="Выберите ЛПУ из привязок НВ" />
   730→                </SelectTrigger>
   731→                <SelectContent className="max-h-72">
   732→                  <SelectItem value={NONE}>Не указано</SelectItem>
   733→                  {multiOptions.map((l) => (
   734→                    <SelectItem key={l.id} value={l.id}>
   735→                      {l.name} · {l.city.name}
   736→                    </SelectItem>
   737→                  ))}
   738→                </SelectContent>
   739→              </Select>
   740→            )}
   741→            {lpuMode === "free" && (
   742→              <Select
   743→                value={form.lpuId || NONE}
   744→                onValueChange={(v) => setForm({ ...form, lpuId: v === NONE ? "" : v })}
   745→              >
   746→                <SelectTrigger className="bg-white">
   747→                  <SelectValue placeholder="Не указано (опционально)" />
   748→                </SelectTrigger>
   749→                <SelectContent className="max-h-72">
   750→                  <SelectItem value={NONE}>Не указано</SelectItem>
   751→                  {freeLpuOptions.map((l) => (
   752→                    <SelectItem key={l.id} value={l.id}>
   753→                      {l.name} · {l.city.name}
   754→                    </SelectItem>
   755→                  ))}
   756→                </SelectContent>
   757→              </Select>
   758→            )}
   759→            {lpuCity && (
   760→              <p className="text-[11px] text-muted-foreground">Город: {lpuCity}</p>
   761→            )}
   762→          </div>
   763→
   764→          {/* 5. Куда направление — по умолчанию сохраняется «МРТ ТОЧНО» */}
   765→          <div className="space-y-1.5">
   766→            <Label>Куда направление</Label>
   767→            <Input
   768→              value={form.destination}
   769→              onChange={(e) => setForm({ ...form, destination: e.target.value })}
   770→              placeholder="МРТ ТОЧНО"
   771→            />
   772→            <p className="text-[11px] text-muted-foreground">
   773→              Если не изменить — сохранится «МРТ ТОЧНО»
   774→            </p>
   775→          </div>
   776→
   777→          {/* 6. Что назначено — мультивыбор из номенклатуры (необязательно) */}
   778→          <div className="space-y-1.5">
   779→            <div className="flex items-center justify-between">
   780→              <Label>Что назначено</Label>
   781→              <span className="text-[11px] text-muted-foreground">
   782→                Выбрано: {form.assignedIds.length}
   783→              </span>
   784→            </div>
   785→            <div className="max-h-40 space-y-1 overflow-y-auto rounded-lg border border-border bg-white p-3">
   786→              {!nomenclature?.length ? (
   787→                <p className="py-2 text-center text-xs text-muted-foreground">
   788→                  Справочник номенклатуры пуст — добавьте позиции в разделе «Справочники»
   789→                </p>
   790→              ) : (
   791→                nomenclature.map((n) => (
   792→                  <label
   793→                    key={n.id}
   794→                    className="flex cursor-pointer items-center gap-2.5 rounded-md px-2 py-1.5 hover:bg-sky-light/60"
   795→                  >
   796→                    <Checkbox
   797→                      checked={form.assignedIds.includes(n.id)}
   798→                      onCheckedChange={(c) =>
   799→                        setForm((f) => ({
   800→                          ...f,
   801→                          assignedIds: toggleInList(f.assignedIds, n.id, c === true),
   802→                        }))
   803→                      }
   804→                      aria-label={`Назначить ${n.name}`}
   805→                    />
   806→                    <span className="text-sm text-brand">{n.name}</span>
   807→                  </label>
   808→                ))
   809→              )}
   810→            </div>
   811→          </div>
   812→
   813→          {/* 7. Что сделано — мультивыбор из номенклатуры (опционально) */}
   814→          <div className="space-y-1.5">
   815→            <div className="flex items-center justify-between">
   816→              <Label>Что сделано</Label>
   817→              <span className="text-[11px] text-muted-foreground">
   818→                Выбрано: {form.doneIds.length}
   819→              </span>
   820→            </div>
   821→            <div className="max-h-40 space-y-1 overflow-y-auto rounded-lg border border-border bg-white p-3">
   822→              {!nomenclature?.length ? (
   823→                <p className="py-2 text-center text-xs text-muted-foreground">
   824→                  Справочник номенклатуры пуст — добавьте позиции в разделе «Справочники»
   825→                </p>
   826→              ) : (
   827→                nomenclature.map((n) => (
   828→                  <label
   829→                    key={n.id}
   830→                    className="flex cursor-pointer items-center gap-2.5 rounded-md px-2 py-1.5 hover:bg-sky-light/60"
   831→                  >
   832→                    <Checkbox
   833→                      checked={form.doneIds.includes(n.id)}
   834→                      onCheckedChange={(c) =>
   835→                        setForm((f) => ({
   836→                          ...f,
   837→                          doneIds: toggleInList(f.doneIds, n.id, c === true),
   838→                        }))
   839→                      }
   840→                      aria-label={`Отметить выполненным ${n.name}`}
   841→                    />
   842→                    <span className="text-sm text-brand">{n.name}</span>
   843→                  </label>
   844→                ))
   845→              )}
   846→            </div>
   847→            <p className="text-[11px] text-muted-foreground">
   848→              Опционально — можно заполнить позже при редактировании заявки
   849→            </p>
   850→          </div>
   851→
   852→          {/* 8. Комментарий (опционально) */}
   853→          <div className="space-y-1.5">
   854→            <Label>Комментарий</Label>
   855→            <Textarea
   856→              rows={2}
   857→              value={form.comment}
   858→              onChange={(e) => setForm({ ...form, comment: e.target.value })}
   859→              placeholder="Опционально — заметки по заявке, договоренности…"
   860→            />
   861→          </div>
   862→        </div>
   863→
   864→        <DialogFooter>
   865→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   866→            Отмена
   867→          </Button>
   868→          <Button
   869→            onClick={() => save.mutate()}
   870→            disabled={!form.date || !form.patientName.trim() || save.isPending}
   871→            className="bg-azure hover:bg-azure-dark"
   872→          >
   873→            {save.isPending && <ClipboardList className="h-4 w-4 animate-spin" strokeWidth={2} />}
   874→            {request ? "Сохранить" : "Зарегистрировать"}
   875→          </Button>
   876→        </DialogFooter>
   877→      </DialogContent>
   878→    </Dialog>
   879→  );
   880→}
