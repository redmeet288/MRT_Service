   175→   175→type CityDialogState = { mode: "create" } | { mode: "edit"; city: City } | null;
   176→   176→
   177→   177→function CitiesTab() {
   178→   178→  const qc = useQueryClient();
   179→   179→  const { data: session } = useSession();
   180→   180→  const isMP = (session?.user as SessionUserWithRole)?.role === "MP";
   181→   181→  const [selectedId, setSelectedId] = useState<string | null>(null);
   182→   182→  const [search, setSearch] = useState("");
   183→   183→  const [sort, setSort] = useState<SortState | null>(null);
   184→   184→  const [dialog, setDialog] = useState<CityDialogState>(null);
   185→   185→
   186→   186→  const { data: cities, isLoading } = useQuery<City[]>({
   187→   187→    queryKey: ["cities"],
   188→   188→    queryFn: () => api.get("/api/app/cities"),
   189→   189→  });
   190→   190→
   191→   191→  const filtered = useMemo(() => {
   192→   192→    const q = search.trim().toLowerCase();
   193→   193→    const list = cities ?? [];
   194→   194→    return q ? list.filter((c) => c.name.toLowerCase().includes(q)) : list;
   195→   195→  }, [cities, search]);
   196→   196→
   197→   197→  const sorted = useMemo(
   198→   198→    () =>
   199→   199→      applySort(filtered, sort, (c, f) => {
   200→   200→        switch (f) {
   201→   201→          case "name":
   202→   202→            return c.name;
   203→   203→          case "lpus":
   204→   204→            return c._count?.lpus ?? 0;
   205→   205→          case "doctors":
   206→   206→            return c._count?.doctors ?? 0;
   207→   207→          default:
   208→   208→            return null;
   209→   209→        }
   210→   210→      }),
   211→   211→    [filtered, sort]
   212→   212→  );
   213→   213→
   214→   214→  const selected = (cities ?? []).find((c) => c.id === selectedId) ?? null;
   215→   215→
   216→   216→  const remove = useMutation({
   217→   217→    mutationFn: (id: string) => api.del(`/api/app/cities?id=${id}`),
   218→   218→    onSuccess: () => {
   219→   219→      toast.success("Город удален");
   220→   220→      setSelectedId(null);
   221→   221→      qc.invalidateQueries({ queryKey: ["cities"] });
   222→   222→    },
   223→   223→    onError: (e: Error) => toast.error(e.message),
   224→   224→  });
   225→   225→
   226→   226→  return (
   227→   227→    <>
   228→   228→      <DictWindow
   229→   229→        icon={<MapPin className="h-4 w-4 text-sky" strokeWidth={2} />}
   230→   230→        title="Города"
   231→   231→        count={filtered.length}
   232→   232→        toolbar={
   233→   233→          <>
   234→   234→            <DictCrudButtons
   235→   235→              onCreate={() => setDialog({ mode: "create" })}
   236→   236→              onEdit={() => selected && setDialog({ mode: "edit", city: selected })}
   237→   237→              canEdit={!!selected}
   238→   238→              deleteSlot={
   239→   239→                isMP ? (
   240→   240→                  <DeleteButton
   241→   241→                    disabled={!selected}
   242→   242→                    title={`Удалить город «${selected?.name ?? ""}»?`}
   243→   243→                    onConfirm={() => {
   244→   244→                      if (selected) remove.mutateAsync(selected.id).catch(() => undefined);
   245→   245→                    }}
   246→   246→                  />
   247→   247→                ) : undefined
   248→   248→              }
   249→   249→            />
   250→   250→            <DictImportExport kind="cities" />
   251→   251→            <OnecSearch value={search} onChange={setSearch} placeholder="Поиск по названию города…" />
   252→   252→          </>
   253→   253→        }
   254→   254→      >
   255→   255→        {isLoading ? (
   256→   256→          <div className="p-4"><RowsSkeleton /></div>
   257→   257→        ) : filtered.length === 0 ? (
   258→   258→          <EmptyState
   259→   259→            className="py-8"
   260→   260→            title={search ? "Ничего не найдено" : "Городов пока нет"}
   261→   261→            hint={search ? "Измените условие поиска" : "Нажмите «Создать» или импортируйте список из файла"}
   262→   262→          />
   263→   263→        ) : (
   264→   264→          <div className="max-h-96 overflow-y-auto">
   265→   265→            <table className="onec-table">
   266→   266→              <thead>
   267→   267→                <tr>
   268→   268→                  <SortableTh label="Название" field="name" sort={sort} onSort={(f) => setSort(cycleSort(sort, f))} />
   269→   269→                  <SortableTh label="ЛПУ" field="lpus" sort={sort} onSort={(f) => setSort(cycleSort(sort, f))} />
   270→   270→                  <SortableTh label="НВ" field="doctors" sort={sort} onSort={(f) => setSort(cycleSort(sort, f))} />
   271→   271→                </tr>
   272→   272→              </thead>
   273→   273→              <tbody>
   274→   274→                {sorted.map((c) => (
   275→   275→                  <tr
   276→   276→                    key={c.id}
   277→   277→                    onClick={() => setSelectedId(c.id)}
   278→   278→                    className={cn("cursor-pointer", selectedId === c.id && "onec-row-selected")}
   279→   279→                  >
   280→   280→                    <td>
   281→   281→                      <button
   282→   282→                        type="button"
   283→   283→                        className="font-medium text-brand hover:underline text-left"
   284→   284→                        title="Изменить"
   285→   285→                        onClick={(e) => {
   286→   286→                          e.stopPropagation();
   287→   287→                          setDialog({ mode: "edit", city: c });
   288→   288→                        }}
   289→   289→                      >
   290→   290→                        {c.name}
   291→   291→                      </button>
   292→   292→                    </td>
   293→   293→                    <td>{c._count?.lpus ?? 0}</td>
   294→   294→                    <td>{c._count?.doctors ?? 0}</td>
   295→   295→                  </tr>
   296→   296→                ))}
   297→   297→              </tbody>
   298→   298→            </table>
   299→   299→          </div>
   300→   300→        )}
   301→   301→      </DictWindow>
   302→   302→
   303→   303→      {dialog && (
   304→   304→        <CityDialog
   305→   305→          editing={dialog.mode === "edit" ? dialog.city : null}
   306→   306→          onClose={() => setDialog(null)}
   307→   307→        />
   308→   308→      )}
   309→   309→    </>
   310→   310→  );
   311→   311→}

   313→   313→function CityDialog({ editing, onClose }: { editing: City | null; onClose: () => void }) {
   314→   314→  const qc = useQueryClient();
   315→   315→  const [name, setName] = useState(() => editing?.name ?? "");
   316→   316→
   317→   317→  const save = useMutation({
   318→   318→    mutationFn: () =>
   319→   319→      editing
   320→   320→        ? api.put("/api/app/cities", { id: editing.id, name })
   321→   321→        : api.post("/api/app/cities", { name }),
   322→   322→    onSuccess: () => {
   323→   323→      toast.success(editing ? "Город обновлен" : "Город добавлен");
   324→   324→      qc.invalidateQueries({ queryKey: ["cities"] });
   325→   325→      onClose();
   326→   326→    },
   327→   327→    onError: (e: Error) => toast.error(e.message),
   328→   328→  });
   329→   329→
   330→   330→  return (
   331→   331→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   332→   332→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-md">
   333→   333→        <DialogHeader>
   334→   334→          <DialogTitle className="font-display text-brand">
   335→   335→            {editing ? "Редактирование города" : "Новый город"}
   336→   336→          </DialogTitle>
   337→   337→          <DialogDescription>
   338→   338→            {editing ? "Измените название и нажмите «Сохранить»." : "Введите название города — он станет доступен для ЛПУ и врачей."}
   339→   339→          </DialogDescription>
   340→   340→        </DialogHeader>
   341→   341→        <div className="space-y-4 py-2">
   342→   342→          <div className="space-y-1.5">
   343→   343→            <Label>Название</Label>
   344→   344→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Например: Казань" autoFocus />
   345→   345→          </div>
   346→   346→        </div>
   347→   347→        <DialogFooter>
   348→   348→          <Button variant="outline" onClick={onClose}>Отмена</Button>
   349→   349→          <Button onClick={() => save.mutate()} disabled={!name.trim() || save.isPending}>
   350→   350→            {editing ? "Сохранить" : "Добавить"}
   351→   351→          </Button>
   352→   352→        </DialogFooter>
   353→   353→      </DialogContent>
   354→   354→    </Dialog>
   355→   355→  );
   356→   356→}
