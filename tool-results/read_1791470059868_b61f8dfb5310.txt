    55→/** Сегодняшняя дата в формате YYYY-MM-DD (локальная зона) */

   370→/* ============================================================
   371→   Диалог регистрации / редактирования заявки (строго по ТЗ)
   372→   ============================================================ */
   373→interface RequestFormState {
   374→  date: string;
   375→  patientName: string;
   376→  doctorId: string;
   377→  lpuId: string;
   378→  destination: string;
   379→  assignedId: string;
   380→  doneId: string;
   381→}
   382→
   383→/** Начальное состояние формы: из заявки при редактировании / пустая (дата — сегодня) при создании */
   384→function initRequestForm(request: RequestItem | null): RequestFormState {
   385→  if (!request) {
   386→    return {
   387→      date: todayISO(),
   388→      patientName: "",
   389→      doctorId: "",
   390→      lpuId: "",
   391→      destination: "",
   392→      assignedId: "",
   393→      doneId: "",
   394→    };
   395→  }
   396→  return {
   397→    date: request.date.slice(0, 10),
   398→    patientName: request.patientName,
   399→    doctorId: request.doctorId,
   400→    lpuId: request.lpuId,
   401→    destination: request.destination ?? "",
   402→    assignedId: request.assignedId ?? "",
   403→    doneId: request.doneId ?? "",
   404→  };
   405→}
   406→
   407→function RequestFormDialog({
   408→  request,
   409→  onOpenChange,
   410→}: {
   411→  request: RequestItem | null;
   412→  onOpenChange: (o: boolean) => void;
   413→}) {
   414→  const qc = useQueryClient();
   415→  const [form, setForm] = useState<RequestFormState>(() => initRequestForm(request));
   416→
   417→  const { data: doctors } = useQuery<DoctorItem[]>({
   418→    queryKey: ["doctors", "all"],
   419→    queryFn: () => api.get("/api/app/doctors"),
   420→  });
   421→  const { data: nomenclature } = useQuery<NomenclatureItem[]>({
   422→    queryKey: ["nomenclature", "active"],
   423→    queryFn: () => api.get("/api/app/nomenclature?active=1"),
   424→  });
   425→  const { data: lpus } = useQuery<CityData[]>({
   426→    queryKey: ["lpus"],
   427→    queryFn: () => api.get("/api/app/lpus"),
   428→  });
   429→
   430→  const selectedDoctor = useMemo(
   431→    () => doctors?.find((d) => d.id === form.doctorId) ?? null,
   432→    [doctors, form.doctorId]
   433→  );
   434→
   435→  // ЛПУ, привязанные к выбранному НВ (данные lpus уже приходят в ответе doctors)
   436→  const doctorLpus = useMemo(() => selectedDoctor?.lpus.map((x) => x.lpu) ?? [], [selectedDoctor]);
   437→
   438→  // Режим выбора ЛПУ: 1 привязка — авто; несколько — Select из привязок; 0 — свободный Select
   439→  const lpuMode: "none" | "single" | "multi" | "free" = !selectedDoctor
   440→    ? "none"
   441→    : doctorLpus.length === 1
   442→      ? "single"
   443→      : doctorLpus.length > 1
   444→        ? "multi"
   445→        : "free";
   446→
   447→  // Варианты для режима «несколько ЛПУ» (+ защита для редактируемой заявки)
   448→  const multiOptions = useMemo(() => {
   449→    const opts = [...doctorLpus];
   450→    if (request && request.lpuId && !opts.some((l) => l.id === request.lpuId)) {
   451→      opts.push(request.lpu);
   452→    }
   453→    return opts;
   454→  }, [doctorLpus, request]);
   455→
   456→  // Итоговый lpuId: при единственном ЛПУ НВ — подставляется автоматически
   457→  const resolvedLpuId = lpuMode === "single" ? (doctorLpus[0]?.id ?? "") : form.lpuId;
   458→
   459→  const lpuCity = useMemo(() => {
   460→    if (!resolvedLpuId) return null;
   461→    const fromDoctor = doctorLpus.find((l) => l.id === resolvedLpuId);
   462→    if (fromDoctor) return fromDoctor.city.name;
   463→    if (request?.lpu.id === resolvedLpuId) return request.lpu.city.name;
   464→    return lpus?.find((l) => l.id === resolvedLpuId)?.city.name ?? null;
   465→  }, [resolvedLpuId, doctorLpus, lpus, request]);
   466→
   467→  const save = useMutation({
   468→    mutationFn: () => {
   469→      const body = {
   470→        date: form.date,
   471→        patientName: form.patientName.trim(),
   472→        doctorId: form.doctorId,
   473→        lpuId: resolvedLpuId,
   474→        destination: form.destination.trim() || null,
   475→        assignedId: form.assignedId || null,
   476→        doneId: form.doneId || null,
   477→      };
   478→      return request
   479→        ? api.put(`/api/app/requests/${request.id}`, body)
   480→        : api.post("/api/app/requests", body);
   481→    },
   482→    onSuccess: () => {
   483→      toast.success(request ? "Заявка обновлена" : "Заявка зарегистрирована");
   484→      qc.invalidateQueries({ queryKey: ["requests"] });
   485→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   486→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   487→      onOpenChange(false);
   488→    },
   489→    onError: (e: Error) => toast.error(e.message),
   490→  });
   491→
   492→  return (
   493→    <Dialog open onOpenChange={onOpenChange}>
   494→      <DialogContent className="sm:max-w-lg">
   495→        <DialogHeader>
   496→          <DialogTitle className="font-display text-brand">
   497→            {request ? "Редактирование заявки" : "Регистрация заявки"}
   498→          </DialogTitle>
   499→          <DialogDescription>Учет направления пациента на обследование</DialogDescription>
   500→        </DialogHeader>
   501→
   502→        <div className="space-y-4 py-1">
   503→          {/* 1. Дата */}
   504→          <div className="space-y-1.5">
   505→            <Label>
   506→              Дата <span className="text-terra-dark">*</span>
   507→            </Label>
   508→            <Input
   509→              type="date"
   510→              value={form.date}
   511→              onChange={(e) => setForm({ ...form, date: e.target.value })}
   512→            />
   513→          </div>
   514→
   515→          {/* 2. ФИО пациента */}
   516→          <div className="space-y-1.5">
   517→            <Label>
   518→              ФИО пациента <span className="text-terra-dark">*</span>
   519→            </Label>
   520→            <Input
   521→              value={form.patientName}
   522→              onChange={(e) => setForm({ ...form, patientName: e.target.value })}
   523→              placeholder="Иванов Иван Иванович"
   524→            />
   525→          </div>
   526→
   527→          {/* 3. ФИО НВ */}
   528→          <div className="space-y-1.5">
   529→            <Label>
   530→              ФИО НВ <span className="text-terra-dark">*</span>
   531→            </Label>
   532→            <Select
   533→              value={form.doctorId}
   534→              onValueChange={(v) => {
   535→                const nd = doctors?.find((d) => d.id === v);
   536→                const dl = nd?.lpus.map((x) => x.lpu) ?? [];
   537→                setForm((f) => ({
   538→                  ...f,
   539→                  doctorId: v,
   540→                  lpuId: dl.length === 1 ? dl[0].id : "",
   541→                }));
   542→              }}
   543→            >
   544→              <SelectTrigger className="bg-white">
   545→                <SelectValue placeholder="Выберите направляющего врача" />
   546→              </SelectTrigger>
   547→              <SelectContent className="max-h-72">
   548→                {(doctors ?? []).map((d) => (
   549→                  <SelectItem key={d.id} value={d.id}>
   550→                    {d.fullName}
   551→                  </SelectItem>
   552→                ))}
   553→              </SelectContent>
   554→            </Select>
   555→          </div>
   556→
   557→          {/* 4. ЛПУ и город — автозаполнение из карточки НВ */}
   558→          <div className="space-y-1.5">
   559→            <Label>
   560→              ЛПУ <span className="text-terra-dark">*</span>
   561→            </Label>
   562→            {lpuMode === "none" && (
   563→              <p className="text-xs text-muted-foreground">
   564→                Сначала выберите НВ — ЛПУ подставится из его карточки
   565→              </p>
   566→            )}
   567→            {lpuMode === "single" && (
   568→              <>
   569→                <Input
   570→                  disabled
   571→                  readOnly
   572→                  value={
   573→                    doctorLpus[0] ? `${doctorLpus[0].name} · ${doctorLpus[0].city.name}` : "—"
   574→                  }
   575→                  className="bg-white text-muted-foreground"
   576→                />
   577→                <p className="text-[11px] text-muted-foreground">
   578→                  Единственное ЛПУ врача — подставлено автоматически
   579→                </p>
   580→              </>
   581→            )}
   582→            {lpuMode === "multi" && (
   583→              <Select
   584→                value={form.lpuId}
   585→                onValueChange={(v) => setForm({ ...form, lpuId: v })}
   586→              >
   587→                <SelectTrigger className="bg-white">
   588→                  <SelectValue placeholder="Выберите ЛПУ из привязок НВ" />
   589→                </SelectTrigger>
   590→                <SelectContent className="max-h-72">
   591→                  {multiOptions.map((l) => (
   592→                    <SelectItem key={l.id} value={l.id}>
   593→                      {l.name} · {l.city.name}
   594→                    </SelectItem>
   595→                  ))}
   596→                </SelectContent>
   597→              </Select>
   598→            )}
   599→            {lpuMode === "free" && (
   600→              <Select
   601→                value={form.lpuId}
   602→                onValueChange={(v) => setForm({ ...form, lpuId: v })}
   603→              >
   604→                <SelectTrigger className="bg-white">
   605→                  <SelectValue placeholder="У НВ нет привязок — выберите ЛПУ" />
   606→                </SelectTrigger>
   607→                <SelectContent className="max-h-72">
   608→                  {(lpus ?? []).map((l) => (
   609→                    <SelectItem key={l.id} value={l.id}>
   610→                      {l.name} · {l.city.name}
   611→                    </SelectItem>
   612→                  ))}
   613→                </SelectContent>
   614→              </Select>
   615→            )}
   616→            {lpuCity && (
   617→              <p className="text-[11px] text-muted-foreground">Город: {lpuCity}</p>
   618→            )}
   619→          </div>
   620→
   621→          {/* 5. Куда направление */}
   622→          <div className="space-y-1.5">
   623→            <Label>Куда направление</Label>
   624→            <Input
   625→              value={form.destination}
   626→              onChange={(e) => setForm({ ...form, destination: e.target.value })}
   627→              placeholder="Опционально — отделение, кабинет"
   628→            />
   629→          </div>
   630→
   631→          {/* 6. Что назначено */}
   632→          <div className="space-y-1.5">
   633→            <Label>
   634→              Что назначено <span className="text-terra-dark">*</span>
   635→            </Label>
   636→            <Select
   637→              value={form.assignedId}
   638→              onValueChange={(v) => setForm({ ...form, assignedId: v })}
   639→            >
   640→              <SelectTrigger className="bg-white">
   641→                <SelectValue
   642→                  placeholder={
   643→                    nomenclature?.length ? "Выберите позицию номенклатуры" : "Справочник номенклатуры пуст"
   644→                  }
   645→                />
   646→              </SelectTrigger>
   647→              <SelectContent className="max-h-72">
   648→                {(nomenclature ?? []).map((n) => (
   649→                  <SelectItem key={n.id} value={n.id}>
   650→                    {n.name}
   651→                  </SelectItem>
   652→                ))}
   653→              </SelectContent>
   654→            </Select>
   655→          </div>
   656→
   657→          {/* 7. Что сделано (опционально — можно заполнить позже) */}
   658→          <div className="space-y-1.5">
   659→            <Label>Что сделано</Label>
   660→            <Select
   661→              value={form.doneId || NONE}
   662→              onValueChange={(v) => setForm({ ...form, doneId: v === NONE ? "" : v })}
   663→            >
   664→              <SelectTrigger className="bg-white">
   665→                <SelectValue placeholder="Пока не выполнено" />
   666→              </SelectTrigger>
   667→              <SelectContent className="max-h-72">
   668→                <SelectItem value={NONE}>Пока не выполнено</SelectItem>
   669→                {(nomenclature ?? []).map((n) => (
   670→                  <SelectItem key={n.id} value={n.id}>
   671→                    {n.name}
   672→                  </SelectItem>
   673→                ))}
   674→              </SelectContent>
   675→            </Select>
   676→            <p className="text-[11px] text-muted-foreground">
   677→              Опционально — можно заполнить позже при редактировании заявки
   678→            </p>
   679→          </div>
   680→        </div>
   681→
   682→        <DialogFooter>
   683→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   684→            Отмена
   685→          </Button>
   686→          <Button
   687→            onClick={() => save.mutate()}
   688→            disabled={
   689→              !form.date ||
   690→              !form.patientName.trim() ||
   691→              !form.doctorId ||
   692→              !resolvedLpuId ||
   693→              !form.assignedId ||
   694→              save.isPending
   695→            }
   696→            className="bg-azure hover:bg-azure-dark"
   697→          >
   698→            {save.isPending && <ClipboardList className="h-4 w-4 animate-spin" strokeWidth={2} />}
   699→            {request ? "Сохранить" : "Зарегистрировать"}
   700→          </Button>
   701→        </DialogFooter>
   702→      </DialogContent>
   703→    </Dialog>
   704→  );
