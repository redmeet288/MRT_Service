   125→/* ============================================================
   126→   Транслитерация для PDF (jsPDF стандартно не поддерживает
   127→   кириллицу) — простая карта а-я → a-ya
   128→   ============================================================ */
   129→const TRANSLIT: Record<string, string> = {
   130→  а: "a", б: "b", в: "v", г: "g", д: "d", е: "e", ё: "e", ж: "zh", з: "z",
   131→  и: "i", й: "y", к: "k", л: "l", м: "m", н: "n", о: "o", п: "p", р: "r",
   132→  с: "s", т: "t", у: "u", ф: "f", х: "kh", ц: "ts", ч: "ch", ш: "sh",
   133→  щ: "sch", ъ: "", ы: "y", ь: "", э: "e", ю: "yu", я: "ya",
   134→};
   135→
   136→function translit(value: string): string {
   137→  return (value ?? "")
   138→    .split("")
   139→    .map((ch) => {
   140→      const lower = ch.toLowerCase();
   141→      const t = TRANSLIT[lower];
   142→      if (t === undefined) return ch; // латиница, цифры, пунктуация — как есть
   143→      if (t === "") return "";
   144→      if (ch !== lower) return t.charAt(0).toUpperCase() + t.slice(1);
   145→      return t;
   146→    })
   147→    .join("");
   148→}

   176→type GroupCtx = { mp?: string; lpu?: string; city?: string; doctor?: string; nom?: string };

   178→function collectRows(node: ReportGroupNode, ctx: GroupCtx, out: ExportRow[]): void {
   179→  const next: GroupCtx = { ...ctx };
   180→  if (node.level === "mp") next.mp = node.name;
   181→  else if (node.level === "lpu") {
   182→    next.lpu = node.name;
   183→    next.city = node.subtitle || ctx.city || "";
   184→  } else if (node.level === "doctor") next.doctor = node.name;
   185→  else next.nom = node.name;
   186→
   187→  if (node.children.length) {
   188→    node.children.forEach((child) => collectRows(child, next, out));
   189→  } else {
   190→    node.requests.forEach((r) =>
   191→      out.push({
   192→        mp: next.mp ?? r.mpName ?? "",
   193→        lpu: next.lpu ?? r.lpuName ?? "",
   194→        city: next.city ?? r.cityName ?? "",
   195→        doctor: next.doctor ?? r.doctorName ?? "",
   196→        nom: next.nom ?? r.assigned ?? "",
   197→        date: fmtDate(r.date),
   198→        patient: r.patientName,
   199→        destination: r.destination ?? "—",
   200→        assigned: r.assigned ?? "—",
   201→        done: r.done ?? "Не выполнено",
   202→      })
   203→    );
   204→  }
   205→}

   448→  function toggleNode(key: string, depth: number) {
   449→    setTreeState((prev) => {
   450→      const open = new Set(prev.open);
   451→      const closed = new Set(prev.closed);

   455→        closed.add(key);
   456→      } else {
   457→        open.add(key);
   458→        closed.delete(key);

  1002→/* ============================================================
  1003→   Рекурсивная строка-группа дерева отчета (.onec-row-group)
  1004→   ============================================================ */
  1005→
  1006→function TreeRows({
  1007→  node,
  1008→  depth,
  1009→  parentKey,
  1010→  detailSpan,
  1011→  show,
  1012→  expansion,
  1013→  onToggle,
  1014→}: {
  1015→  node: ReportGroupNode;
  1016→  depth: number;
  1017→  parentKey: string;
  1018→  detailSpan: number;
  1019→  show: { nv: boolean; lpu: boolean; mp: boolean };
  1020→  expansion: TreeExpansion;
  1021→  onToggle: (key: string, depth: number) => void;
  1022→}) {
  1023→  // Уникальный ключ по пути в дереве: level:id на каждом уровне
  1024→  const key = `${parentKey}/${depth}:${node.level}:${node.id}`;
  1025→  const open = !expansion.closed.has(key) && (depth === 0 || expansion.open.has(key));
  1026→  const hasChildren = node.children.length > 0;
  1027→  const hasContent = hasChildren || node.requests.length > 0;
  1028→  const Icon = LEVEL_ICONS[node.level] ?? Building2;
  1029→  const p = pct(node.done, node.total);
  1030→
  1031→  return (
  1032→    <>
  1033→      <tr className="onec-row-group">
  1034→        <td colSpan={detailSpan}>
  1035→          <div className="flex items-center gap-1.5" style={{ paddingLeft: depth * 20 }}>
  1036→            {hasContent ? (
  1037→              <button
  1038→                type="button"
  1039→                className="onec-tree-toggle"
  1040→                aria-expanded={open}
  1041→                aria-label={open ? `Свернуть группу «${node.name}»` : `Развернуть группу «${node.name}»`}
  1042→                onClick={() => onToggle(key, depth)}
  1043→              >
  1044→                {open ? "−" : "+"}
  1045→              </button>
  1046→            ) : (
  1047→              <span className="inline-block h-[15px] w-[15px] shrink-0" aria-hidden />
  1048→            )}
  1049→            <Icon className="h-3.5 w-3.5 shrink-0 text-brand" strokeWidth={2} />
  1050→            <span className="text-[13px] text-[#1c1c1c]">{node.name}</span>
  1051→            {node.subtitle && (
  1052→              <span className="font-normal text-[11px] text-[#8a8a8a]">· {node.subtitle}</span>
  1053→            )}
  1054→          </div>
  1055→        </td>
  1056→        <td style={{ textAlign: "right" }}>{node.total}</td>
  1057→        <td style={{ textAlign: "right" }}>{node.done}</td>
  1058→        <td style={{ textAlign: "right" }}>
  1059→          <span
  1060→            className={cn(
  1061→              "inline-block rounded-full px-1.5 py-[1px] text-[10px] font-semibold",
  1062→              pctBadgeClass(p)
  1063→            )}
  1064→          >
  1065→            {p}%
  1066→          </span>
  1067→        </td>
  1068→      </tr>
  1069→
  1070→      {/* Дочерние группы */}
  1071→      {open &&
  1072→        hasChildren &&
  1073→        node.children.map((child) => (
  1074→          <TreeRows
  1075→            key={child.id}
  1076→            node={child}
  1077→            depth={depth + 1}
  1078→            parentKey={key}
  1079→            detailSpan={detailSpan}
  1080→            show={show}
  1081→            expansion={expansion}
  1082→            onToggle={onToggle}
  1083→          />
  1084→        ))}
  1085→
  1086→      {/* Листья: детальные строки заявок */}
  1087→      {open &&
  1088→        !hasChildren &&
  1089→        (node.requests.length ? (
  1090→          node.requests.map((r) => <DetailRow key={r.id} r={r} show={show} />)
  1091→        ) : (
  1092→          <tr>
  1093→            <td colSpan={detailSpan + 3} className="italic text-[#9a9a9a]" style={{ fontSize: 12 }}>
  1094→              Нет записей в группе
  1095→            </td>
  1096→          </tr>
  1097→        ))}
  1098→    </>
  1099→  );
  1100→}
