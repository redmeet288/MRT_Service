    54→    54→/** Сегодняшняя дата в формате YYYY-MM-DD (локальная зона) */
    55→    55→function todayISO(): string {
    56→    56→  const d = new Date();
    57→    57→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);
    58→    58→}
    59→    59→
    60→    60→/** ISO-дата (из drill-down дашборда) → yyyy-MM-dd для input[type=date] */
    61→    61→function isoToYMD(iso: string | undefined): string {
    62→    62→  if (!iso) return "";
    63→    63→  const d = new Date(iso);
    64→    64→  return Number.isNaN(d.getTime()) ? "" : format(d, "yyyy-MM-dd");
    65→    65→}

   451→   451→/** Начальное состояние формы: из заявки при редактировании / пустая (дата — сегодня) при создании */

   454→   454→    return {
   455→   455→      date: todayISO(),
   456→   456→      patientName: "",
   457→   457→      doctorId: "",
   458→   458→      lpuId: "",
   459→   459→      destination: "",
   460→   460→      assignedId: "",
   461→   461→      doneId: "",
   462→   462→      comment: "",
   463→   463→    };
   464→   464→  }
   465→   465→  return {
   466→   466→    date: request.date.slice(0, 10),
   467→   467→    patientName: request.patientName,
   468→   468→    doctorId: request.doctorId,
   469→   469→    lpuId: request.lpuId,
   470→   470→    destination: request.destination ?? "",
   471→   471→    assignedId: request.assignedId ?? "",
   472→   472→    doneId: request.doneId ?? "",
   473→   473→    comment: request.comment ?? "",
   474→   474→  };
   475→   475→}
   476→   476→
   477→   477→function RequestFormDialog({
   478→   478→  request,
   479→   479→  onOpenChange,
   480→   480→}: {
   481→   481→  request: RequestItem | null;
   482→   482→  onOpenChange: (o: boolean) => void;
   483→   483→}) {
   484→   484→  const qc = useQueryClient();
   485→   485→  const [form, setForm] = useState<RequestFormState>(() => initRequestForm(request));
   486→   486→
   487→   487→  const { data: doctors } = useQuery<DoctorItem[]>({
   488→   488→    queryKey: ["doctors", "all"],
   489→   489→    queryFn: () => api.get("/api/app/doctors"),
   490→   490→  });
   491→   491→  const { data: nomenclature } = useQuery<NomenclatureItem[]>({
   492→   492→    queryKey: ["nomenclature", "active"],
   493→   493→    queryFn: () => api.get("/api/app/nomenclature?active=1"),
   494→   494→  });
   495→   495→  const { data: lpus } = useQuery<CityData[]>({
   496→   496→    queryKey: ["lpus"],
   497→   497→    queryFn: () => api.get("/api/app/lpus"),
   498→   498→  });
   499→   499→
   500→   500→  const selectedDoctor = useMemo(
   501→   501→    () => doctors?.find((d) => d.id === form.doctorId) ?? null,
   502→   502→    [doctors, form.doctorId]
   503→   503→  );
   504→   504→
   505→   505→  // ЛПУ, привязанные к выбранному НВ (данные lpus уже приходят в ответе doctors)
   506→   506→  const doctorLpus = useMemo(() => selectedDoctor?.lpus.map((x) => x.lpu) ?? [], [selectedDoctor]);
   507→   507→
   508→   508→  // Режим выбора ЛПУ: 1 привязка — авто; несколько — Select из привязок; 0 — свободный Select
   509→   509→  const lpuMode: "none" | "single" | "multi" | "free" = !selectedDoctor
   510→   510→    ? "none"
   511→   511→    : doctorLpus.length === 1
   512→   512→      ? "single"
   513→   513→      : doctorLpus.length > 1
   514→   514→        ? "multi"
   515→   515→        : "free";
   516→   516→
   517→   517→  // Варианты для режима «несколько ЛПУ» (+ защита для редактируемой заявки)
   518→   518→  const multiOptions = useMemo(() => {
   519→   519→    const opts = [...doctorLpus];
   520→   520→    if (request && request.lpuId && !opts.some((l) => l.id === request.lpuId)) {
   521→   521→      opts.push(request.lpu);
   522→   522→    }
   523→   523→    return opts;
   524→   524→  }, [doctorLpus, request]);
   525→   525→
   526→   526→  // Итоговый lpuId: при единственном ЛПУ НВ — подставляется автоматически
   527→   527→  const resolvedLpuId = lpuMode === "single" ? (doctorLpus[0]?.id ?? "") : form.lpuId;
   528→   528→
   529→   529→  const lpuCity = useMemo(() => {
   530→   530→    if (!resolvedLpuId) return null;
   531→   531→    const fromDoctor = doctorLpus.find((l) => l.id === resolvedLpuId);
   532→   532→    if (fromDoctor) return fromDoctor.city.name;
   533→   533→    if (request?.lpu.id === resolvedLpuId) return request.lpu.city.name;
   534→   534→    return lpus?.find((l) => l.id === resolvedLpuId)?.city.name ?? null;
   535→   535→  }, [resolvedLpuId, doctorLpus, lpus, request]);
   536→   536→
   537→   537→  const save = useMutation({
   538→   538→    mutationFn: () => {
   539→   539→      const body = {
   540→   540→        date: form.date,
   541→   541→        patientName: form.patientName.trim(),
   542→   542→        doctorId: form.doctorId,
   543→   543→        lpuId: resolvedLpuId,
   544→   544→        destination: form.destination.trim() || null,
   545→   545→        assignedId: form.assignedId || null,
   546→   546→        doneId: form.doneId || null,
   547→   547→        comment: form.comment.trim() || null,
   548→   548→      };
   549→   549→      return request
   550→   550→        ? api.put(`/api/app/requests/${request.id}`, body)
   551→   551→        : api.post("/api/app/requests", body);
   552→   552→    },
   553→   553→    onSuccess: () => {
   554→   554→      toast.success(request ? "Заявка обновлена" : "Заявка зарегистрирована");
   555→   555→      qc.invalidateQueries({ queryKey: ["requests"] });
   556→   556→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   557→   557→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   558→   558→      onOpenChange(false);
   559→   559→    },
   560→   560→    onError: (e: Error) => toast.error(e.message),
   561→   561→  });
   562→   562→
   563→   563→  return (
   564→   564→    <Dialog open onOpenChange={onOpenChange}>
   565→   565→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-lg">
   566→   566→        <DialogHeader>
   567→   567→          <DialogTitle className="font-display text-brand">
   568→   568→            {request ? "Карточка заявки" : "Регистрация заявки"}
   569→   569→          </DialogTitle>
   570→   570→          <DialogDescription>
   571→   571→            {request
   572→   572→              ? "Просмотр и редактирование направления пациента на обследование"
   573→   573→              : "Учет направления пациента на обследование"}
   574→   574→          </DialogDescription>
   575→   575→        </DialogHeader>
   576→   576→
   577→   577→        <div className="space-y-4 py-1">
   578→   578→          {/* 1. Дата */}
   579→   579→          <div className="space-y-1.5">
   580→   580→            <Label>
   581→   581→              Дата <span className="text-terra-dark">*</span>
   582→   582→            </Label>
   583→   583→            <Input
   584→   584→              type="date"
   585→   585→              value={form.date}
   586→   586→              onChange={(e) => setForm({ ...form, date: e.target.value })}
   587→   587→            />
   588→   588→          </div>
   589→   589→
   590→   590→          {/* 2. ФИО пациента */}
   591→   591→          <div className="space-y-1.5">
   592→   592→            <Label>
   593→   593→              ФИО пациента <span className="text-terra-dark">*</span>
   594→   594→            </Label>
   595→   595→            <Input
   596→   596→              value={form.patientName}
   597→   597→              onChange={(e) => setForm({ ...form, patientName: e.target.value })}
   598→   598→              placeholder="Иванов Иван Иванович"
   599→   599→            />
   600→   600→          </div>
   601→   601→
   602→   602→          {/* 3. ФИО НВ */}
   603→   603→          <div className="space-y-1.5">
   604→   604→            <Label>
   605→   605→              ФИО НВ <span className="text-terra-dark">*</span>
   606→   606→            </Label>
   607→   607→            <Select
   608→   608→              value={form.doctorId}
   609→   609→              onValueChange={(v) => {
   610→   610→                const nd = doctors?.find((d) => d.id === v);
   611→   611→                const dl = nd?.lpus.map((x) => x.lpu) ?? [];
   612→   612→                setForm((f) => ({
   613→   613→                  ...f,
   614→   614→                  doctorId: v,
   615→   615→                  lpuId: dl.length === 1 ? dl[0].id : "",
   616→   616→                }));
   617→   617→              }}
   618→   618→            >
   619→   619→              <SelectTrigger className="bg-white">
   620→   620→                <SelectValue placeholder="Выберите направляющего врача" />
   621→   621→              </SelectTrigger>
   622→   622→              <SelectContent className="max-h-72">
   623→   623→                {(doctors ?? []).map((d) => (
   624→   624→                  <SelectItem key={d.id} value={d.id}>
   625→   625→                    {d.fullName}
   626→   626→                  </SelectItem>
   627→   627→                ))}
   628→   628→              </SelectContent>
   629→   629→            </Select>
   630→   630→          </div>
   631→   631→
   632→   632→          {/* 4. ЛПУ и город — автозаполнение из карточки НВ */}
   633→   633→          <div className="space-y-1.5">
   634→   634→            <Label>
   635→   635→              ЛПУ <span className="text-terra-dark">*</span>
   636→   636→            </Label>
   637→   637→            {lpuMode === "none" && (
   638→   638→              <p className="text-xs text-muted-foreground">
   639→   639→                Сначала выберите НВ — ЛПУ подставится из его карточки
   640→   640→              </p>
   641→   641→            )}
   642→   642→            {lpuMode === "single" && (
   643→   643→              <>
   644→   644→                <Input
   645→   645→                  disabled
   646→   646→                  readOnly
   647→   647→                  value={
   648→   648→                    doctorLpus[0] ? `${doctorLpus[0].name} · ${doctorLpus[0].city.name}` : "—"
   649→   649→                  }
   650→   650→                  className="bg-white text-muted-foreground"
   651→   651→                />
   652→   652→                <p className="text-[11px] text-muted-foreground">
   653→   653→                  Единственное ЛПУ врача — подставлено автоматически
   654→   654→                </p>
   655→   655→              </>
   656→   656→            )}
   657→   657→            {lpuMode === "multi" && (
   658→   658→              <Select
   659→   659→                value={form.lpuId}
   660→   660→                onValueChange={(v) => setForm({ ...form, lpuId: v })}
   661→   661→              >
   662→   662→                <SelectTrigger className="bg-white">
   663→   663→                  <SelectValue placeholder="Выберите ЛПУ из привязок НВ" />
   664→   664→                </SelectTrigger>
   665→   665→                <SelectContent className="max-h-72">
   666→   666→                  {multiOptions.map((l) => (
   667→   667→                    <SelectItem key={l.id} value={l.id}>
   668→   668→                      {l.name} · {l.city.name}
   669→   669→                    </SelectItem>
   670→   670→                  ))}
   671→   671→                </SelectContent>
   672→   672→              </Select>
   673→   673→            )}
   674→   674→            {lpuMode === "free" && (
   675→   675→              <Select
   676→   676→                value={form.lpuId}
   677→   677→                onValueChange={(v) => setForm({ ...form, lpuId: v })}
   678→   678→              >
   679→   679→                <SelectTrigger className="bg-white">
   680→   680→                  <SelectValue placeholder="У НВ нет привязок — выберите ЛПУ" />
   681→   681→                </SelectTrigger>
   682→   682→                <SelectContent className="max-h-72">
   683→   683→                  {(lpus ?? []).map((l) => (
   684→   684→                    <SelectItem key={l.id} value={l.id}>
   685→   685→                      {l.name} · {l.city.name}
   686→   686→                    </SelectItem>
   687→   687→                  ))}
   688→   688→                </SelectContent>
   689→   689→              </Select>
   690→   690→            )}
   691→   691→            {lpuCity && (
   692→   692→              <p className="text-[11px] text-muted-foreground">Город: {lpuCity}</p>
   693→   693→            )}
   694→   694→          </div>
   695→   695→
   696→   696→          {/* 5. Куда направление */}
   697→   697→          <div className="space-y-1.5">
   698→   698→            <Label>Куда направление</Label>
   699→   699→            <Input
   700→   700→              value={form.destination}
   701→   701→              onChange={(e) => setForm({ ...form, destination: e.target.value })}
   702→   702→              placeholder="Опционально — отделение, кабинет"
   703→   703→            />
   704→   704→          </div>
   705→   705→
   706→   706→          {/* 6. Что назначено */}
   707→   707→          <div className="space-y-1.5">
   708→   708→            <Label>
   709→   709→              Что назначено <span className="text-terra-dark">*</span>
   710→   710→            </Label>
   711→   711→            <Select
   712→   712→              value={form.assignedId}
   713→   713→              onValueChange={(v) => setForm({ ...form, assignedId: v })}
   714→   714→            >
   715→   715→              <SelectTrigger className="bg-white">
   716→   716→                <SelectValue
   717→   717→                  placeholder={
   718→   718→                    nomenclature?.length ? "Выберите позицию номенклатуры" : "Справочник номенклатуры пуст"
   719→   719→                  }
   720→   720→                />
   721→   721→              </SelectTrigger>
   722→   722→              <SelectContent className="max-h-72">
   723→   723→                {(nomenclature ?? []).map((n) => (
   724→   724→                  <SelectItem key={n.id} value={n.id}>
   725→   725→                    {n.name}
   726→   726→                  </SelectItem>
   727→   727→                ))}
   728→   728→              </SelectContent>
   729→   729→            </Select>
   730→   730→          </div>
   731→   731→
   732→   732→          {/* 7. Что сделано (опционально — можно заполнить позже) */}
   733→   733→          <div className="space-y-1.5">
   734→   734→            <Label>Что сделано</Label>
   735→   735→            <Select
   736→   736→              value={form.doneId || NONE}
   737→   737→              onValueChange={(v) => setForm({ ...form, doneId: v === NONE ? "" : v })}
   738→   738→            >
   739→   739→              <SelectTrigger className="bg-white">
   740→   740→                <SelectValue placeholder="Пока не выполнено" />
   741→   741→              </SelectTrigger>
   742→   742→              <SelectContent className="max-h-72">
   743→   743→                <SelectItem value={NONE}>Пока не выполнено</SelectItem>
   744→   744→                {(nomenclature ?? []).map((n) => (
   745→   745→                  <SelectItem key={n.id} value={n.id}>
   746→   746→                    {n.name}
   747→   747→                  </SelectItem>
   748→   748→                ))}
   749→   749→              </SelectContent>
   750→   750→            </Select>
   751→   751→            <p className="text-[11px] text-muted-foreground">
   752→   752→              Опционально — можно заполнить позже при редактировании заявки
   753→   753→            </p>
   754→   754→          </div>
   755→   755→
   756→   756→          {/* 8. Комментарий (опционально) */}
   757→   757→          <div className="space-y-1.5">
   758→   758→            <Label>Комментарий</Label>
   759→   759→            <Textarea
   760→   760→              rows={2}
   761→   761→              value={form.comment}
   762→   762→              onChange={(e) => setForm({ ...form, comment: e.target.value })}
   763→   763→              placeholder="Опционально — заметки по заявке, договоренности…"
   764→   764→            />
   765→   765→          </div>
   766→   766→        </div>
   767→   767→
   768→   768→        <DialogFooter>
   769→   769→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   770→   770→            Отмена
   771→   771→          </Button>
   772→   772→          <Button
   773→   773→            onClick={() => save.mutate()}
   774→   774→            disabled={
   775→   775→              !form.date ||
   776→   776→              !form.patientName.trim() ||
   777→   777→              !form.doctorId ||
   778→   778→              !resolvedLpuId ||
   779→   779→              !form.assignedId ||
   780→   780→              save.isPending
   781→   781→            }
   782→   782→            className="bg-azure hover:bg-azure-dark"
   783→   783→          >
   784→   784→            {save.isPending && <ClipboardList className="h-4 w-4 animate-spin" strokeWidth={2} />}

   787→   787→        </DialogFooter>
   788→   788→      </DialogContent>
   789→   789→    </Dialog>
   790→   790→  );
   791→   791→}
