     2→     2→

     4→     4→import { useQuery } from "@tanstack/react-query";
     5→     5→import {
     6→     6→  addDays,
     7→     7→  addMonths,
     8→     8→  addWeeks,
     9→     9→  eachDayOfInterval,
    10→    10→  endOfMonth,

    12→    12→  format,

    14→    14→  getMinutes,
    15→    15→  isSameMonth,
    16→    16→  isToday,
    17→    17→  startOfMonth,
    18→    18→  startOfWeek,
    19→    19→} from "date-fns";
    20→    20→import { ru } from "date-fns/locale";
    21→    21→import { api, qs } from "@/lib/api-client";
    22→    22→import type { DoctorItem, UserBasic, VisitItem } from "@/lib/types";
    23→    23→import { VISIT_STATUS_LABELS } from "@/lib/types";
    24→    24→import { EmptyState, RowsSkeleton, StatusChip } from "@/components/shared/ui-bits";
    25→    25→import { VisitDialog } from "@/components/views/visit-dialog";
    26→    26→import { Button } from "@/components/ui/button";
    27→    27→import { Label } from "@/components/ui/label";
    28→    28→import {
    29→    29→  Select,
    30→    30→  SelectContent,
    31→    31→  SelectItem,
    32→    32→  SelectTrigger,
    33→    33→  SelectValue,
    34→    34→} from "@/components/ui/select";
    35→    35→import { ToggleGroup, ToggleGroupItem } from "@/components/ui/toggle-group";
    36→    36→import {
    37→    37→  CalendarClock,
    38→    38→  CalendarDays,
    39→    39→  CalendarPlus,
    40→    40→  CalendarRange,
    41→    41→  ChevronLeft,
    42→    42→  ChevronRight,
    43→    43→  Plus,
    44→    44→} from "lucide-react";
    45→    45→import { cn } from "@/lib/utils";
    46→    46→
    47→    47→type ViewMode = "month" | "week" | "day";
    48→    48→type VisitStatus = VisitItem["status"];
    49→    49→
    50→    50→const WEEKDAYS = ["Пн", "Вт", "Ср", "Чт", "Пт", "Сб", "Вс"];
    51→    51→
    52→    52→/** Границы сетки часов вида «День» */
    53→    53→const DAY_START_HOUR = 7;
    54→    54→const DAY_END_HOUR = 21;
    55→    55→const SLOT_HEIGHT = 56; // px — высота строки часа (min-h слота)
    56→    56→
    57→    57→/** Цвет чипа визита в сетке — по статусу (гайдлайн: лазурь / синий / терракот) */
    58→    58→const CHIP_BG: Record<VisitStatus, string> = {
    59→    59→  PLANNED: "bg-azure hover:bg-azure-dark",
    60→    60→  CONDUCTED: "bg-brand hover:bg-brand-dark",
    61→    61→  CANCELLED: "bg-terra hover:bg-terra-dark",
    62→    62→};
    63→    63→
    64→    64→const cap = (s: string) => s.charAt(0).toUpperCase() + s.slice(1);
    65→    65→const iso = (d: Date) => format(d, "yyyy-MM-dd");
    66→    66→const fmtTime = (d: string | Date) => format(new Date(d), "HH:mm");
    67→    67→const pad2 = (n: number) => String(n).padStart(2, "0");
    68→    68→
    69→    69→export function CalendarView() {
    70→    70→  // —— Вид и навигация ——
    71→    71→  const [mode, setMode] = useState<ViewMode>("month");
    72→    72→  const [cursor, setCursor] = useState<Date>(new Date());
    73→    73→
    74→    74→  // —— Фильтры ("ALL" = без фильтра) ——
    75→    75→  const [mpFilter, setMpFilter] = useState("ALL");
    76→    76→  const [statusFilter, setStatusFilter] = useState("ALL");
    77→    77→  const [doctorFilter, setDoctorFilter] = useState("ALL");
    78→    78→
    79→    79→  // —— Диалог (общий VisitDialog, монтируется условно) ——
    80→    80→  const [dialogOpen, setDialogOpen] = useState(false);
    81→    81→  const [editing, setEditing] = useState<VisitItem | null>(null);
    82→    82→  const [dialogDate, setDialogDate] = useState<Date>(new Date());
    83→    83→  const [dialogTime, setDialogTime] = useState<string | undefined>(undefined);
    84→    84→
    85→    85→  // —— Справочники для фильтров (диалог тянет НВ из общего кеша) ——
    86→    86→  const { data: users } = useQuery<UserBasic[]>({
    87→    87→    queryKey: ["users-basic"],
    88→    88→    queryFn: () => api.get("/api/app/users?basic=1"),
    89→    89→  });
    90→    90→  const mps = useMemo(() => (users ?? []).filter((u) => u.role === "MP"), [users]);
    91→    91→
    92→    92→  const { data: doctors } = useQuery<DoctorItem[]>({
    93→    93→    queryKey: ["doctors"],
    94→    94→    queryFn: () => api.get("/api/app/doctors"),
    95→    95→  });
    96→    96→  const doctorOptions = useMemo(
    97→    97→    () => (doctors ?? []).filter((d) => mpFilter === "ALL" || d.mpId === mpFilter),
    98→    98→    [doctors, mpFilter]
    99→    99→  );
   100→   100→
   101→   101→  // —— Период выборки (включая дни соседних месяцев в сетке; в дне — сам день) ——
   102→   102→  const range = useMemo(() => {
   103→   103→    if (mode === "month") {
   104→   104→      return {
   105→   105→        from: startOfWeek(startOfMonth(cursor), { weekStartsOn: 1 }),
   106→   106→        to: endOfWeek(endOfMonth(cursor), { weekStartsOn: 1 }),
   107→   107→      };
   108→   108→    }
   109→   109→    if (mode === "week") {
   110→   110→      return {
   111→   111→        from: startOfWeek(cursor, { weekStartsOn: 1 }),
   112→   112→        to: endOfWeek(cursor, { weekStartsOn: 1 }),
   113→   113→      };
   114→   114→    }
   115→   115→    return { from: cursor, to: cursor };
   116→   116→  }, [cursor, mode]);
   117→   117→
   118→   118→  // —— Визиты ——
   119→   119→  const visitsQuery = useQuery<VisitItem[]>({
   120→   120→    queryKey: [
   121→   121→      "visits",
   122→   122→      mode,
   123→   123→      iso(range.from),
   124→   124→      iso(range.to),
   125→   125→      mpFilter,
   126→   126→      statusFilter,
   127→   127→      doctorFilter,
   128→   128→    ],
   129→   129→    queryFn: () =>
   130→   130→      api.get<VisitItem[]>(
   131→   131→        `/api/app/visits${qs({
   132→   132→          from: iso(range.from),
   133→   133→          to: `${iso(range.to)}T23:59:59`,
   134→   134→          mpId: mpFilter === "ALL" ? undefined : mpFilter,
   135→   135→          status: statusFilter === "ALL" ? undefined : statusFilter,
   136→   136→          doctorId: doctorFilter === "ALL" ? undefined : doctorFilter,
   137→   137→        })}`
   138→   138→      ),
   139→   139→  });
   140→   140→
   141→   141→  // —— Группировка визитов по дням ——
   142→   142→  const grouped = useMemo(() => {
   143→   143→    const map = new Map<string, VisitItem[]>();
   144→   144→    (visitsQuery.data ?? []).forEach((v) => {
   145→   145→      const key = iso(new Date(v.date));
   146→   146→      const arr = map.get(key) ?? [];
   147→   147→      arr.push(v);
   148→   148→      map.set(key, arr);
   149→   149→    });
   150→   150→    return map;
   151→   151→  }, [visitsQuery.data]);
   152→   152→
   153→   153→  // —— Клетки месячной/недельной сетки (Пн–Вс) ——
   154→   154→  const days = useMemo(
   155→   155→    () => eachDayOfInterval({ start: range.from, end: range.to }),
   156→   156→    [range]
   157→   157→  );
   158→   158→
   159→   159→  // —— Вид «День»: визиты выбранного дня, разбитые по часам ——
   160→   160→  const dayVisits = useMemo(
   161→   161→    () => (mode === "day" ? (grouped.get(iso(cursor)) ?? []) : []),
   162→   162→    [mode, grouped, cursor]
   163→   163→  );
   164→   164→
   165→   165→  // Часы сетки: 07:00–21:00 + часы, в которых есть визиты вне диапазона
   166→   166→  const dayHours = useMemo(() => {
   167→   167→    const base: number[] = [];
   168→   168→    for (let h = DAY_START_HOUR; h <= DAY_END_HOUR; h++) base.push(h);
   169→   169→    const extra = dayVisits
   170→   170→      .map((v) => getHours(new Date(v.date)))
   171→   171→      .filter((h) => h < DAY_START_HOUR || h > DAY_END_HOUR);
   172→   172→    return [...new Set([...base, ...extra])].sort((a, b) => a - b);
   173→   173→  }, [dayVisits]);
   174→   174→
   175→   175→  const visitsByHour = useMemo(() => {
   176→   176→    const map = new Map<number, VisitItem[]>();
   177→   177→    [...dayVisits]
   178→   178→      .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   179→   179→      .forEach((v) => {
   180→   180→        const h = getHours(new Date(v.date));
   181→   181→        const arr = map.get(h) ?? [];
   182→   182→        arr.push(v);
   183→   183→        map.set(h, arr);
   184→   184→      });
   185→   185→    return map;
   186→   186→  }, [dayVisits]);
   187→   187→
   188→   188→  // Автопрокрутка сетки дня к текущему часу (без setState — прямой DOM-скролл)
   189→   189→  const dayScrollRef = useRef<HTMLDivElement | null>(null);
   190→   190→  useEffect(() => {
   191→   191→    const el = dayScrollRef.current;
   192→   192→    if (!el || mode !== "day" || !isToday(cursor)) return;
   193→   193→    const y = (getHours(new Date()) - DAY_START_HOUR) * SLOT_HEIGHT;
   194→   194→    el.scrollTop = Math.max(0, y - SLOT_HEIGHT * 2);
   195→   195→  }, [mode, cursor]);
   196→   196→
   197→   197→  const step = (dir: 1 | -1) =>
   198→   198→    setCursor(
   199→   199→      mode === "month"
   200→   200→        ? addMonths(cursor, dir)
   201→   201→        : mode === "week"
   202→   202→          ? addWeeks(cursor, dir)
   203→   203→          : addDays(cursor, dir)
   204→   204→    );
   205→   205→
   206→   206→  // —— Диалог: открытие ——
   207→   207→  function openCreateForDay(day: Date, time?: string) {
   208→   208→    setEditing(null);
   209→   209→    setDialogDate(day);
   210→   210→    setDialogTime(time);
   211→   211→    setDialogOpen(true);
   212→   212→  }
   213→   213→  function openEdit(v: VisitItem) {
   214→   214→    setEditing(v);
   215→   215→    setDialogOpen(true);
   216→   216→  }
   217→   217→
   218→   218→  const periodTitle =
   219→   219→    mode === "month"
   220→   220→      ? cap(format(cursor, "LLLL yyyy", { locale: ru }))
   221→   221→      : mode === "week"
   222→   222→        ? `${cap(format(range.from, "d MMMM", { locale: ru }))} — ${format(
   223→   223→            range.to,
   224→   224→            "d MMMM yyyy",
   225→   225→            { locale: ru }
   226→   226→          )}`
   227→   227→        : cap(format(cursor, "EEEE, d MMMM yyyy", { locale: ru }));
   228→   228→
   229→   229→  const noVisits =
   230→   230→    !visitsQuery.isLoading && !visitsQuery.error && (visitsQuery.data ?? []).length === 0;
   231→   231→
   232→   232→  return (
   233→   233→    <div className="space-y-6">
   234→   234→      {/* —— Фильтры —— */}
   235→   235→      <SectionFilters
   236→   236→        mps={mps}
   237→   237→        mpFilter={mpFilter}
   238→   238→        setMpFilter={setMpFilter}
   239→   239→        statusFilter={statusFilter}
   240→   240→        setStatusFilter={setStatusFilter}
   241→   241→        doctorOptions={doctorOptions}
   242→   242→        doctorFilter={doctorFilter}
   243→   243→        setDoctorFilter={setDoctorFilter}
   244→   244→        onCreate={() => openCreateForDay(new Date())}
   245→   245→      />
   246→   246→
   247→   247→      {/* —— Календарь —— */}
   248→   248→      <section className="rounded-xl border border-border bg-white p-6 shadow-[0_2px_12px_rgba(25,70,136,0.05)]">
   249→   249→        {/* Навигация периода */}
   250→   250→        <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
   251→   251→          <div className="flex items-center gap-2">
   252→   252→            <Button
   253→   253→              variant="ghost"
   254→   254→              size="icon"
   255→   255→              className="h-8 w-8 text-brand hover:bg-sky-light"
   256→   256→              onClick={() => step(-1)}
   257→   257→              aria-label="Предыдущий период"
   258→   258→            >
   259→   259→              <ChevronLeft className="h-4 w-4" strokeWidth={2} />
   260→   260→            </Button>
   261→   261→            <h2 className="font-display text-lg font-bold text-brand min-w-[190px] text-center">
   262→   262→              {periodTitle}
   263→   263→            </h2>
   264→   264→            <Button
   265→   265→              variant="ghost"
   266→   266→              size="icon"
   267→   267→              className="h-8 w-8 text-brand hover:bg-sky-light"
   268→   268→              onClick={() => step(1)}
   269→   269→              aria-label="Следующий период"
   270→   270→            >
   271→   271→              <ChevronRight className="h-4 w-4" strokeWidth={2} />
   272→   272→            </Button>
   273→   273→            <Button
   274→   274→              variant="outline"
   275→   275→              size="sm"
   276→   276→              className="ml-2 h-8 border-sky/40 text-brand hover:bg-sky-light"
   277→   277→              onClick={() => setCursor(new Date())}
   278→   278→            >
   279→   279→              Сегодня
   280→   280→            </Button>
   281→   281→          </div>
   282→   282→          <ToggleGroup
   283→   283→            type="single"
   284→   284→            size="sm"
   285→   285→            value={mode}
   286→   286→            onValueChange={(v) => {
   287→   287→              if (v === "month" || v === "week" || v === "day") setMode(v);
   288→   288→            }}
   289→   289→            className="rounded-lg border border-border bg-soft p-1"
   290→   290→          >
   291→   291→            <ToggleGroupItem
   292→   292→              value="month"
   293→   293→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   294→   294→            >
   295→   295→              <CalendarDays className="h-3.5 w-3.5" strokeWidth={2} /> Месяц
   296→   296→            </ToggleGroupItem>
   297→   297→            <ToggleGroupItem
   298→   298→              value="week"
   299→   299→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   300→   300→            >
   301→   301→              <CalendarRange className="h-3.5 w-3.5" strokeWidth={2} /> Неделя
   302→   302→            </ToggleGroupItem>
   303→   303→            <ToggleGroupItem
   304→   304→              value="day"
   305→   305→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   306→   306→            >
   307→   307→              <CalendarClock className="h-3.5 w-3.5" strokeWidth={2} /> День
   308→   308→            </ToggleGroupItem>
   309→   309→          </ToggleGroup>
   310→   310→        </div>
   311→   311→
   312→   312→        {visitsQuery.error ? (
   313→   313→          <div className="rounded-xl border border-terra/50 bg-terra-light p-6 text-sm text-brand-dark">
   314→   314→            Не удалось загрузить визиты: {(visitsQuery.error as Error).message}
   315→   315→          </div>
   316→   316→        ) : visitsQuery.isLoading ? (
   317→   317→          <RowsSkeleton rows={6} />
   318→   318→        ) : (
   319→   319→          <>
   320→   320→            {noVisits && (
   321→   321→              <EmptyState
   322→   322→                title="Визитов в выбранном периоде нет"
   323→   323→                hint="Кликните по дню или свободному часу в календаре, либо нажмите «Запланировать визит»"
   324→   324→                className="py-6"
   325→   325→              />
   326→   326→            )}
   327→   327→
   328→   328→            {mode === "month" ? (
   329→   329→              <>
   330→   330→                {/* Шапка дней недели */}
   331→   331→                <div className="mb-2 grid grid-cols-7 gap-2">
   332→   332→                  {WEEKDAYS.map((w) => (
   333→   333→                    <div
   334→   334→                      key={w}
   335→   335→                      className="text-center text-[11px] font-semibold uppercase tracking-wide text-sky"
   336→   336→                    >
   337→   337→                      {w}
   338→   338→                    </div>
   339→   339→                  ))}
   340→   340→                </div>
   341→   341→                {/* Сетка месяца */}
   342→   342→                <div className="grid grid-cols-7 gap-2">
   343→   343→                  {days.map((day) => {
   344→   344→                    const key = iso(day);
   345→   345→                    const list = grouped.get(key) ?? [];
   346→   346→                    const dim = !isSameMonth(day, cursor);
   347→   347→                    return (
   348→   348→                      <div
   349→   349→                        key={key}
   350→   350→                        onClick={() => openCreateForDay(day)}
   351→   351→                        className={cn(
   352→   352→                          "group/cell flex min-h-[112px] cursor-pointer flex-col gap-1 rounded-lg border p-1.5 transition-colors",
   353→   353→                          dim ? "border-border/60 bg-soft" : "border-border bg-white hover:border-sky/50",
   354→   354→                          isToday(day) && "border-azure/60 bg-sky-light/50 ring-1 ring-azure"
   355→   355→                        )}
   356→   356→                      >
   357→   357→                        <div className="flex items-center justify-between">
   358→   358→                          <span
   359→   359→                            className={cn(
   360→   360→                              "inline-flex h-5 w-5 items-center justify-center rounded-full font-display text-[11px] font-bold",
   361→   361→                              isToday(day)
   362→   362→                                ? "bg-azure text-white"
   363→   363→                                : dim
   364→   364→                                  ? "text-muted-foreground/50"
   365→   365→                                  : "text-brand"
   366→   366→                            )}
   367→   367→                          >
   368→   368→                            {day.getDate()}
   369→   369→                          </span>
   370→   370→                          <button
   371→   371→                            type="button"
   372→   372→                            onClick={(e) => {
   373→   373→                              e.stopPropagation();
   374→   374→                              openCreateForDay(day);
   375→   375→                            }}
   376→   376→                            className="text-sky opacity-0 transition-opacity hover:text-azure group-hover/cell:opacity-100"
   377→   377→                            aria-label={`Запланировать визит на ${key}`}
   378→   378→                          >
   379→   379→                            <Plus className="h-3.5 w-3.5" strokeWidth={2} />
   380→   380→                          </button>
   381→   381→                        </div>
   382→   382→                        <div className="flex flex-col gap-1">
   383→   383→                          {list.slice(0, 3).map((v) => (
   384→   384→                            <button
   385→   385→                              key={v.id}
   386→   386→                              type="button"
   387→   387→                              onClick={(e) => {
   388→   388→                                e.stopPropagation();
   389→   389→                                openEdit(v);
   390→   390→                              }}
   391→   391→                              title={`${fmtTime(v.date)} · ${v.doctor?.fullName ?? "—"} · ${v.purpose}`}
   392→   392→                              className={cn(
   393→   393→                                "w-full truncate rounded-md px-1.5 py-0.5 text-left text-[10px] font-medium text-white transition-opacity",
   394→   394→                                CHIP_BG[v.status],
   395→   395→                                dim && "opacity-60"
   396→   396→                              )}
   397→   397→                            >
   398→   398→                              {fmtTime(v.date)} {v.doctor?.fullName ?? "Врач"}
   399→   399→                            </button>
   400→   400→                          ))}
   401→   401→                          {list.length > 3 && (
   402→   402→                            <span className="px-1 font-display text-[10px] font-semibold text-sky">
   403→   403→                              +{list.length - 3} ещё
   404→   404→                            </span>
   405→   405→                          )}
   406→   406→                        </div>
   407→   407→                      </div>
   408→   408→                    );
   409→   409→                  })}
   410→   410→                </div>
   411→   411→              </>
   412→   412→            ) : mode === "day" ? (
   413→   413→              /* —— Сетка дня по часам (07:00–21:00) —— */
   414→   414→              <div ref={dayScrollRef} className="max-h-[600px] overflow-y-auto rounded-lg border border-border">
   415→   415→                {dayHours.map((h) => {
   416→   416→                  const list = visitsByHour.get(h) ?? [];
   417→   417→                  const isCurrent = isToday(cursor) && getHours(new Date()) === h;
   418→   418→                  return (
   419→   419→                    <div key={h} className="flex border-b border-border/60 last:border-b-0">
   420→   420→                      {/* Метка часа */}
   421→   421→                      <div
   422→   422→                        className={cn(
   423→   423→                          "w-16 shrink-0 border-r border-border/60 px-2 py-1.5 text-right font-display text-[11px] font-semibold",
   424→   424→                          isCurrent ? "bg-sky-light text-azure" : "bg-soft/60 text-sky"
   425→   425→                        )}
   426→   426→                      >
   427→   427→                        {pad2(h)}:00
   428→   428→                      </div>
   429→   429→                      {/* Слот часа */}
   430→   430→                      <div
   431→   431→                        onClick={() => openCreateForDay(cursor, `${pad2(h)}:00`)}
   432→   432→                        className={cn(
   433→   433→                          "group/slot relative min-h-[56px] flex-1 cursor-pointer p-1.5 transition-colors hover:bg-sky-light/40",
   434→   434→                          isCurrent && "bg-sky-light/50"
   435→   435→                        )}
