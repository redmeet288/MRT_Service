    92→/**
    93→ * Время сброса в виде «День»: по финальной Y-позиции чипа (начальный top + дельта)
    94→ * находим слот часа и минуты с точностью до четверти часа.
    95→ * Фолбэк — час слота, на который произошел сброс (HH:00).
    96→ */
    97→function resolveSlotTime(
    98→  slotEls: Map<number, HTMLElement>,
    99→  startTop: number | null,
   100→  deltaY: number,
   101→  fallbackHour: number
   102→): { hour: number; minutes: number } {
   103→  if (startTop != null) {
   104→    const finalTop = startTop + deltaY;
   105→    for (const [h, el] of slotEls) {
   106→      const r = el.getBoundingClientRect();
   107→      if (finalTop >= r.top && finalTop < r.bottom) {
   108→        const raw = ((finalTop - r.top) / r.height) * 60;
   109→        const minutes = Math.min(59, Math.max(0, Math.round(raw / 15) * 15));
   110→        return { hour: h, minutes };
   111→      }
   112→    }
   113→  }
   114→  return { hour: fallbackHour, minutes: 0 };
   115→}

   320→  function handleDragEnd(e: DragEndEvent) {
   321→    const visit = (e.active.data.current?.visit as VisitItem | undefined) ?? null;
   322→    const overId = e.over?.id;
   323→    const startTop = dragStartTopRef.current;
   324→    setDragging(null);
   325→    dragStartTopRef.current = null;
   326→    if (!visit || !overId) return;
   327→    const target = String(overId);
   328→    let newDate: Date | null = null;
   329→    if (target.startsWith("day:")) {

   331→      const old = new Date(visit.date);
   332→      const day = new Date(`${target.slice(4)}T00:00:00`);
   333→      day.setHours(old.getHours(), old.getMinutes(), 0, 0);
   334→      newDate = day;
   335→    } else if (target.startsWith("slot:")) {
   336→      // День: тот же день, новое время (точность до 15 минут по курсору сброса)
   337→      const activator = e.activatorEvent as PointerEvent | MouseEvent | undefined;
   338→      const pointerY =
   339→        activator && typeof activator.clientY === "number"
   340→          ? activator.clientY + e.delta.y
   341→          : startTop != null
   342→            ? startTop + e.delta.y
   343→            : null;
   344→      const { hour, minutes } = resolveSlotTime(
   345→        slotElsRef.current,
   346→        pointerY,
   347→        0,
   348→        Number(target.slice(5))
   349→      );
   350→      const slot = new Date(cursor);
   351→      slot.setHours(hour, minutes, 0, 0);
   352→      newDate = slot;

   510→                    <DayHourRow

   514→                      isCurrent={isToday(cursor) && getHours(new Date()) === h}

   553→            {/* Легенда статусов */}

   558→              </span>
   559→              <span className="inline-flex items-center gap-1.5">
   560→                <span className="h-2.5 w-2.5 rounded-full bg-brand" /> Проведен
   561→              </span>
   562→              <span className="inline-flex items-center gap-1.5">

   564→              </span>

   566→          </>
   567→        )}
   568→      </section>
   569→
   570→      {/* —— Общий диалог создания/редактирования визита —— */}
   571→      {dialogOpen && (
   572→        <VisitDialog
   573→          open={dialogOpen}
   574→          onOpenChange={setDialogOpen}

   576→          defaultDate={dialogDate}
   577→          defaultTime={dialogTime}
   578→        />
   579→      )}
   580→    </div>
   581→  );
   582→}

   837→/** Строка часа в виде «День»: пустой слот → создание, сброс чипа → перенос по времени */
   838→function DayHourRow({
   839→  hour,
   840→  visits,
   841→  isCurrent,
   842→  slotEls,
   843→  onCreateSlot,
   844→  onEdit,
   845→}: {
   846→  hour: number;
   847→  visits: VisitItem[];
   848→  isCurrent: boolean;
   849→  slotEls: { current: Map<number, HTMLElement> };
   850→  onCreateSlot: () => void;
   851→  onEdit: EditVisit;
   852→}) {
   853→  const { setNodeRef: setSlotRef, isOver } = useDroppable({ id: `slot:${hour}` });
   854→  return (
   855→    <div className="flex border-b border-border/60 last:border-b-0">
   856→      {/* Метка часа */}
   857→      <div
   858→        className={cn(
   859→          "w-16 shrink-0 border-r border-border/60 px-2 py-1.5 text-right font-display text-[11px] font-semibold",
   860→          isCurrent ? "bg-sky-light text-azure" : "bg-soft/60 text-sky"
   861→        )}
   862→      >
   863→        {pad2(hour)}:00
   864→      </div>
   865→      {/* Слот часа */}
   866→      <div
   867→        ref={(el) => {
   868→          if (el) slotEls.current.set(hour, el);
   869→          else slotEls.current.delete(hour);
   870→          setSlotRef(el);
   871→        }}
   872→        onClick={onCreateSlot}
   873→        className={cn(
   874→          "group/slot relative min-h-[56px] flex-1 cursor-pointer p-1.5 transition-colors hover:bg-sky-light/40",
   875→          isCurrent && "bg-sky-light/50",
   876→          isOver && "bg-sky-light ring-1 ring-inset ring-azure"
   877→        )}
   878→      >
   879→        {/* Индикатор текущего времени */}
   880→        {isCurrent && (
   881→          <div
   882→            className="pointer-events-none absolute left-0 right-0 z-10 flex items-center"
   883→            style={{ top: `${(getMinutes(new Date()) / 60) * 100}%` }}
   884→          >
   885→            <span className="-ml-1 h-2 w-2 rounded-full bg-azure" />
   886→            <span className="h-px flex-1 bg-azure/60" />
   887→          </div>
   888→        )}
   889→        <Plus
   890→          className="pointer-events-none absolute right-2 top-2 h-3.5 w-3.5 text-sky opacity-0 transition-opacity group-hover/slot:opacity-100"
   891→          strokeWidth={2}
   892→          aria-hidden
   893→        />

   895→          <span className="select-none text-[11px] text-transparent transition-colors group-hover/slot:text-muted-foreground/60">
   896→            Запланировать на {pad2(hour)}:00
   897→          </span>
   898→        ) : (
   899→          <div className="flex flex-col gap-1.5">
   900→            {visits.map((v) => (
   901→              <DayVisitCard key={v.id} visit={v} onEdit={onEdit} />
   902→            ))}
   903→          </div>
   904→        )}
   905→      </div>
   906→    </div>
   907→  );
   908→}
