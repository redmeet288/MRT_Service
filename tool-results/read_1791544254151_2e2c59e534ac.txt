    56→    56→/** Сегодняшняя дата в формате YYYY-MM-DD (локальная зона) */
    57→    57→function todayISO(): string {
    58→    58→  const d = new Date();
    59→    59→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);
    60→    60→}
    61→    61→
    62→    62→/** ISO-дата (из drill-down дашборда) → yyyy-MM-dd для input[type=date] */
    63→    63→function isoToYMD(iso: string | undefined): string {
    64→    64→  if (!iso) return "";
    65→    65→  const d = new Date(iso);
    66→    66→  return Number.isNaN(d.getTime()) ? "" : format(d, "yyyy-MM-dd");
    67→    67→}

   465→   465→  patientName: string;

   467→   467→  lpuId: string;
   468→   468→  destination: string;
   469→   469→  /** Мультивыбор из номенклатуры: «Что назначено» — обязателен, «Что сделано» — нет */
   470→   470→  assignedIds: string[];
   471→   471→  doneIds: string[];
   472→   472→  comment: string;
   473→   473→}
   474→   474→
   475→   475→/** Переключение позиции в списке множественного выбора */
   476→   476→function toggleInList(list: string[], id: string, checked: boolean): string[] {
   477→   477→  return checked
   478→   478→    ? list.includes(id)
   479→   479→      ? list
   480→   480→      : [...list, id]
   481→   481→    : list.filter((x) => x !== id);
   482→   482→}

   484→   484→/** Начальное состояние формы: из заявки при редактировании / пустая (дата — сегодня) при создании */
   485→   485→function initRequestForm(request: RequestItem | null): RequestFormState {
   486→   486→  if (!request) {
   487→   487→    return {
   488→   488→      date: todayISO(),
   489→   489→      patientName: "",
   490→   490→      doctorId: "",
   491→   491→      lpuId: "",
   492→   492→      destination: "",
   493→   493→      assignedIds: [],
   494→   494→      doneIds: [],
   495→   495→      comment: "",
   496→   496→    };
   497→   497→  }
   498→   498→  return {
   499→   499→    date: request.date.slice(0, 10),
   500→   500→    patientName: request.patientName,
   501→   501→    doctorId: request.doctorId,
   502→   502→    lpuId: request.lpuId,
   503→   503→    destination: request.destination ?? "",
   504→   504→    assignedIds: request.assignedItems.map((i) => i.id),
   505→   505→    doneIds: request.doneItems.map((i) => i.id),
   506→   506→    comment: request.comment ?? "",
   507→   507→  };
   508→   508→}

   510→   510→function RequestFormDialog({

   516→   516→}) {

   518→   518→  const [form, setForm] = useState<RequestFormState>(() => initRequestForm(request));
   519→   519→
   520→   520→  const { data: doctors } = useQuery<DoctorItem[]>({
   521→   521→    queryKey: ["doctors", "all"],
   522→   522→    queryFn: () => api.get("/api/app/doctors"),
   523→   523→  });
   524→   524→  const { data: nomenclature } = useQuery<NomenclatureItem[]>({
   525→   525→    queryKey: ["nomenclature", "active"],

   537→   537→

   540→   540→
   541→   541→  // Режим выбора ЛПУ: 1 привязка — авто; несколько — Select из привязок; 0 — свободный Select
   542→   542→  const lpuMode: "none" | "single" | "multi" | "free" = !selectedDoctor
   543→   543→    ? "none"
   544→   544→    : doctorLpus.length === 1
   545→   545→      ? "single"
   546→   546→      : doctorLpus.length > 1
   547→   547→        ? "multi"
   548→   548→        : "free";
   549→   549→
   550→   550→  // Варианты для режима «несколько ЛПУ» (+ защита для редактируемой заявки)
   551→   551→  const multiOptions = useMemo(() => {
   552→   552→    const opts = [...doctorLpus];
   553→   553→    if (request && request.lpuId && !opts.some((l) => l.id === request.lpuId)) {
   554→   554→      opts.push(request.lpu);
   555→   555→    }
   556→   556→    return opts;
   557→   557→  }, [doctorLpus, request]);
   558→   558→
   559→   559→  // Итоговый lpuId: при единственном ЛПУ НВ — подставляется автоматически
   560→   560→  const resolvedLpuId = lpuMode === "single" ? (doctorLpus[0]?.id ?? "") : form.lpuId;
   561→   561→
   562→   562→  const lpuCity = useMemo(() => {
   563→   563→    if (!resolvedLpuId) return null;
   564→   564→    const fromDoctor = doctorLpus.find((l) => l.id === resolvedLpuId);
   565→   565→    if (fromDoctor) return fromDoctor.city.name;
   566→   566→    if (request?.lpu.id === resolvedLpuId) return request.lpu.city.name;
   567→   567→    return lpus?.find((l) => l.id === resolvedLpuId)?.city.name ?? null;
   568→   568→  }, [resolvedLpuId, doctorLpus, lpus, request]);
   569→   569→
   570→   570→  const save = useMutation({
   571→   571→    mutationFn: () => {
   572→   572→      const body = {
   573→   573→        date: form.date,
   574→   574→        patientName: form.patientName.trim(),
   575→   575→        doctorId: form.doctorId,
   576→   576→        lpuId: resolvedLpuId,
   577→   577→        destination: form.destination.trim() || "МРТ ТОЧНО",
   578→   578→        assignedIds: form.assignedIds,
   579→   579→        doneIds: form.doneIds,
   580→   580→        comment: form.comment.trim() || null,
   581→   581→      };
   582→   582→      return request
   583→   583→        ? api.put(`/api/app/requests/${request.id}`, body)
   584→   584→        : api.post("/api/app/requests", body);
   585→   585→    },
   586→   586→    onSuccess: () => {
   587→   587→      toast.success(request ? "Заявка обновлена" : "Заявка зарегистрирована");
   588→   588→      qc.invalidateQueries({ queryKey: ["requests"] });
   589→   589→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   590→   590→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   591→   591→      onOpenChange(false);
   592→   592→    },
   593→   593→    onError: (e: Error) => toast.error(e.message),
   594→   594→  });
   595→   595→
   596→   596→  return (
   597→   597→    <Dialog open onOpenChange={onOpenChange}>
   598→   598→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-lg">
   599→   599→        <DialogHeader>
   600→   600→          <DialogTitle className="font-display text-brand">
   601→   601→            {request ? "Карточка заявки" : "Регистрация заявки"}
   602→   602→          </DialogTitle>
   603→   603→          <DialogDescription>
   604→   604→            {request
   605→   605→              ? "Просмотр и редактирование направления пациента на обследование"
   606→   606→              : "Учет направления пациента на обследование"}
   607→   607→          </DialogDescription>
   608→   608→        </DialogHeader>
   609→   609→
   610→   610→        <div className="space-y-4 py-1">
   611→   611→          {/* 1. Дата */}
   612→   612→          <div className="space-y-1.5">
   613→   613→            <Label>
   614→   614→              Дата <span className="text-azure-dark">*</span>
   615→   615→            </Label>
   616→   616→            <Input
   617→   617→              type="date"
   618→   618→              value={form.date}
   619→   619→              onChange={(e) => setForm({ ...form, date: e.target.value })}
   620→   620→            />
   621→   621→          </div>
   622→   622→
   623→   623→          {/* 2. ФИО пациента */}
   624→   624→          <div className="space-y-1.5">
   625→   625→            <Label>
   626→   626→              ФИО пациента <span className="text-azure-dark">*</span>
   627→   627→            </Label>
   628→   628→            <Input
   629→   629→              value={form.patientName}
   630→   630→              onChange={(e) => setForm({ ...form, patientName: e.target.value })}
   631→   631→              placeholder="Иванов Иван Иванович"
   632→   632→            />
   633→   633→          </div>
   634→   634→
   635→   635→          {/* 3. ФИО НВ */}
   636→   636→          <div className="space-y-1.5">
   637→   637→            <Label>
   638→   638→              ФИО НВ <span className="text-azure-dark">*</span>
   639→   639→            </Label>
   640→   640→            <Select
   641→   641→              value={form.doctorId}
   642→   642→              onValueChange={(v) => {
   643→   643→                const nd = doctors?.find((d) => d.id === v);
   644→   644→                const dl = nd?.lpus.map((x) => x.lpu) ?? [];

   735→   735→              placeholder="МРТ ТОЧНО"

   752→   752→            <div className="max-h-40 space-y-1 overflow-y-auto rounded-lg border border-border bg-white p-3">

   853→   853→  );
   854→   854→}
