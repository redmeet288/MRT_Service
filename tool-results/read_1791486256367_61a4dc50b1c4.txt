   125→   125→/* ============================================================
   126→   126→   Транслитерация для PDF (jsPDF стандартно не поддерживает
   127→   127→   кириллицу) — простая карта а-я → a-ya
   128→   128→   ============================================================ */
   129→   129→const TRANSLIT: Record<string, string> = {
   130→   130→  а: "a", б: "b", в: "v", г: "g", д: "d", е: "e", ё: "e", ж: "zh", з: "z",
   131→   131→  и: "i", й: "y", к: "k", л: "l", м: "m", н: "n", о: "o", п: "p", р: "r",
   132→   132→  с: "s", т: "t", у: "u", ф: "f", х: "kh", ц: "ts", ч: "ch", ш: "sh",
   133→   133→  щ: "sch", ъ: "", ы: "y", ь: "", э: "e", ю: "yu", я: "ya",
   134→   134→};
   135→   135→
   136→   136→function translit(value: string): string {
   137→   137→  return (value ?? "")
   138→   138→    .split("")
   139→   139→    .map((ch) => {
   140→   140→      const lower = ch.toLowerCase();
   141→   141→      const t = TRANSLIT[lower];
   142→   142→      if (t === undefined) return ch; // латиница, цифры, пунктуация — как есть
   143→   143→      if (t === "") return "";
   144→   144→      if (ch !== lower) return t.charAt(0).toUpperCase() + t.slice(1);
   145→   145→      return t;
   146→   146→    })
   147→   147→    .join("");
   148→   148→}

   178→   178→function collectRows(node: ReportGroupNode, ctx: GroupCtx, out: ExportRow[]): void {
   179→   179→  const next: GroupCtx = { ...ctx };
   180→   180→  if (node.level === "mp") next.mp = node.name;
   181→   181→  else if (node.level === "lpu") {
   182→   182→    next.lpu = node.name;
   183→   183→    next.city = node.subtitle || ctx.city || "";
   184→   184→  } else if (node.level === "doctor") next.doctor = node.name;
   185→   185→  else next.nom = node.name;
   186→   186→
   187→   187→  if (node.children.length) {
   188→   188→    node.children.forEach((child) => collectRows(child, next, out));
   189→   189→  } else {
   190→   190→    node.requests.forEach((r) =>
   191→   191→      out.push({
   192→   192→        mp: next.mp ?? r.mpName ?? "",
   193→   193→        lpu: next.lpu ?? r.lpuName ?? "",
   194→   194→        city: next.city ?? r.cityName ?? "",
   195→   195→        doctor: next.doctor ?? r.doctorName ?? "",
   196→   196→        nom: next.nom ?? r.assigned ?? "",
   197→   197→        date: fmtDate(r.date),
   198→   198→        patient: r.patientName,
   199→   199→        destination: r.destination ?? "—",
   200→   200→        assigned: r.assigned ?? "—",
   201→   201→        done: r.done ?? "Не выполнено",
   202→   202→      })
   203→   203→    );
   204→   204→  }

   447→   447→  // —— Управление деревом (контролируемое, ключ уникален по пути) ——
   448→   448→  function toggleNode(key: string, depth: number) {
   449→   449→    setTreeState((prev) => {
   450→   450→      const open = new Set(prev.open);
   451→   451→      const closed = new Set(prev.closed);
   452→   452→      const isOpen = !closed.has(key) && (depth === 0 || open.has(key));
   453→   453→      if (isOpen) {
   454→   454→        open.delete(key);
   455→   455→        closed.add(key);
   456→   456→      } else {
   457→   457→        open.add(key);
   458→   458→        closed.delete(key);
   459→   459→      }
   460→   460→      return { open, closed };
   461→   461→    });
   462→   462→  }

  1006→  1006→function TreeRows({
  1007→  1007→  node,
  1008→  1008→  depth,
  1009→  1009→  parentKey,
  1010→  1010→  detailSpan,
  1011→  1011→  show,
  1012→  1012→  expansion,
  1013→  1013→  onToggle,
  1014→  1014→}: {
  1015→  1015→  node: ReportGroupNode;
  1016→  1016→  depth: number;

  1018→  1018→  detailSpan: number;
  1019→  1019→  show: { nv: boolean; lpu: boolean; mp: boolean };
  1020→  1020→  expansion: TreeExpansion;
  1021→  1021→  onToggle: (key: string, depth: number) => void;
  1022→  1022→}) {
  1023→  1023→  // Уникальный ключ по пути в дереве: level:id на каждом уровне
  1024→  1024→  const key = `${parentKey}/${depth}:${node.level}:${node.id}`;

  1026→  1026→  const hasChildren = node.children.length > 0;
  1027→  1027→  const hasContent = hasChildren || node.requests.length > 0;
  1028→  1028→  const Icon = LEVEL_ICONS[node.level] ?? Building2;
  1029→  1029→  const p = pct(node.done, node.total);

  1031→  1031→  return (

  1035→  1035→          <div className="flex items-center gap-1.5" style={{ paddingLeft: depth * 20 }}>
  1036→  1036→            {hasContent ? (

  1053→  1053→            )}
  1054→  1054→          </div>

  1067→  1067→        </td>

  1069→  1069→
  1070→  1070→      {/* Дочерние группы */}
  1071→  1071→      {open &&
  1072→  1072→        hasChildren &&
  1073→  1073→        node.children.map((child) => (
  1074→  1074→          <TreeRows
  1075→  1075→            key={child.id}

  1085→  1085→
  1086→  1086→      {/* Листья: детальные строки заявок */}

  1089→  1089→        (node.requests.length ? (

  1094→  1094→              Нет записей в группе
  1095→  1095→            </td>
  1096→  1096→          </tr>
