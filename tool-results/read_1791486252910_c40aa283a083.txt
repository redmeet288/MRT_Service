   398→/** Чипы ЛПУ: показываются первые 2, остальные — «+N» (полный список в подсказке) */
   399→function LpuChips({ lpus }: { lpus: DoctorItem["lpus"] }) {
   400→  if (!lpus.length) {
   401→    return <span className="text-xs italic text-muted-foreground">—</span>;
   402→  }
   403→  const visible = lpus.slice(0, 2);
   404→  const hidden = lpus.slice(2);
   405→  return (
   406→    <div className="flex max-w-[230px] flex-wrap gap-1">
   407→      {visible.map(({ lpu }) => (
   408→        <span
   409→          key={lpu.id}
   410→          className="inline-flex items-center gap-1 rounded-full bg-sky-light px-2 py-0.5 text-[11px] text-brand"
   411→          title={`${lpu.city.name}${lpu.address ? `, ${lpu.address}` : ""}`}
   412→        >
   413→          <Building2 className="h-3 w-3 text-sky" strokeWidth={2} />
   414→          {lpu.name}
   415→        </span>
   416→      ))}
   417→      {hidden.length > 0 && (
   418→        <span
   419→          className="inline-flex items-center rounded-full border border-border bg-white px-1.5 py-0.5 text-[11px] font-medium text-brand"
   420→          title={hidden.map(({ lpu }) => lpu.name).join(", ")}
   421→        >
   422→          +{hidden.length}
   423→        </span>
   424→      )}
   425→    </div>
   426→  );
   427→}

   432→interface DoctorFormState {
   433→  fullName: string;
   434→  phone: string;
   435→  birthDate: string;
   436→  statusId: string;
   437→  mpId: string;
   438→  cityId: string;
   439→  lpuIds: string[];
   440→  comment: string;
   441→}
   442→
   443→const emptyDoctorForm: DoctorFormState = {
   444→  fullName: "",
   445→  phone: "",
   446→  birthDate: "",
   447→  statusId: NONE,
   448→  mpId: "",
   449→  cityId: NONE,
   450→  lpuIds: [],
   451→  comment: "",
   452→};
   453→
   454→/**
   455→ * Начальное состояние формы: из карточки при редактировании / пустая при создании.
   456→ * Для МП при создании врач по умолчанию закрепляется за текущим пользователем.
   457→ */
   458→function initDoctorForm(doctor: EditableDoctor | null, isMP: boolean, sessionUid?: string): DoctorFormState {
   459→  if (!doctor) {
   460→    return { ...emptyDoctorForm, mpId: isMP && sessionUid ? sessionUid : "" };
   461→  }
   462→  return {
   463→    fullName: doctor.fullName,
   464→    phone: doctor.phone ?? "",
   465→    birthDate: doctor.birthDate ? doctor.birthDate.slice(0, 10) : "",
   466→    statusId: doctor.statusId ?? NONE,
   467→    mpId: doctor.mpId ?? NONE, // null → «— Без МП —» (НВ без куратора)
   468→    cityId: doctor.cityId ?? NONE,
   469→    lpuIds: doctor.lpus?.map((x) => x.lpu.id) ?? [],
   470→    comment: doctor.comment ?? "",
   471→  };
   472→}

   474→function DoctorFormDialog({
   475→  onOpenChange,
   476→  doctor,
   477→}: {
   478→  onOpenChange: (o: boolean) => void;
   479→  doctor: EditableDoctor | null;
   480→}) {
   481→  const qc = useQueryClient();
   482→  const { data: session } = useSession();
   483→  const isMP = (session?.user as SessionUserWithRole)?.role === "MP";
   484→  const [form, setForm] = useState<DoctorFormState>(() =>
   485→    initDoctorForm(doctor, isMP, (session?.user as SessionUserWithRole)?.uid)
   486→  );
   487→
   488→  const { data: users } = useQuery<UserBasic[]>({
   489→    queryKey: ["users-basic"],
   490→    queryFn: () => api.get("/api/app/users?basic=1"),
   491→  });
   492→  const { data: statuses } = useQuery<CooperationStatus[]>({
   493→    queryKey: ["statuses"],
   494→    queryFn: () => api.get("/api/app/statuses"),
   495→  });
   496→  const { data: cities } = useQuery<City[]>({
   497→    queryKey: ["cities"],
   498→    queryFn: () => api.get("/api/app/cities"),
   499→  });
   500→  const { data: lpus } = useQuery<CityData[]>({
   501→    queryKey: ["lpus"],
   502→    queryFn: () => api.get("/api/app/lpus"),
   503→  });
   504→
   505→  const mps = useMemo(() => (users ?? []).filter((u) => u.role === "MP"), [users]);
   506→
   507→  /** Переключение ЛПУ; автоподстановка города первого ЛПУ, если город не указан */
   508→  function toggleLpu(id: string, checked: boolean) {
   509→    setForm((f) => {
   510→      const lpuIds = checked ? [...f.lpuIds, id] : f.lpuIds.filter((x) => x !== id);
   511→      let cityId = f.cityId;
   512→      if (checked && cityId === NONE && lpuIds.length > 0) {
   513→        const first = lpus?.find((l) => l.id === lpuIds[0]);
   514→        if (first) cityId = first.city.id;
   515→      }
   516→      return { ...f, lpuIds, cityId };
   517→    });
   518→  }
   519→
   520→  const save = useMutation({
   521→    mutationFn: () => {
   522→      const body: Record<string, unknown> = {
   523→        fullName: form.fullName.trim(),
   524→        phone: form.phone.trim() || null,
   525→        birthDate: form.birthDate || null,
   526→        statusId: form.statusId === NONE ? null : form.statusId,
   527→        cityId: form.cityId === NONE ? null : form.cityId,
   528→        lpuIds: form.lpuIds,
   529→        comment: form.comment.trim() || null,
   530→      };
   531→      // mpId передается только ролью МП: оператор создает НВ без куратора,
   532→      // МП закрепится позднее. Пустое значение при редактировании = открепить МП.
   533→      if (isMP) body.mpId = form.mpId === NONE ? "" : form.mpId;
   534→      return doctor ? api.put(`/api/app/doctors/${doctor.id}`, body) : api.post("/api/app/doctors", body);
   535→    },
   536→    onSuccess: () => {
   537→      toast.success(doctor ? "Карточка врача обновлена" : "Врач добавлен");
   538→      qc.invalidateQueries({ queryKey: ["doctors"] });
   539→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   540→      qc.invalidateQueries({ queryKey: ["dashboard"] });
