    74→    74→/* ============================================================
    75→    75→   Транслитерация для PDF (jsPDF стандартно не поддерживает
    76→    76→   кириллицу) — простая карта а-я → a-ya
    77→    77→   ============================================================ */
    78→    78→const TRANSLIT: Record<string, string> = {
    79→    79→  а: "a", б: "b", в: "v", г: "g", д: "d", е: "e", ё: "e", ж: "zh", з: "z",
    80→    80→  и: "i", й: "y", к: "k", л: "l", м: "m", н: "n", о: "o", п: "p", р: "r",
    81→    81→  с: "s", т: "t", у: "u", ф: "f", х: "kh", ц: "ts", ч: "ch", ш: "sh",
    82→    82→  щ: "sch", ъ: "", ы: "y", ь: "", э: "e", ю: "yu", я: "ya",
    83→    83→};
    84→    84→
    85→    85→function translit(value: string): string {
    86→    86→  return (value ?? "")
    87→    87→    .split("")
    88→    88→    .map((ch) => {
    89→    89→      const lower = ch.toLowerCase();
    90→    90→      const t = TRANSLIT[lower];
    91→    91→      if (t === undefined) return ch; // латиница, цифры, пунктуация — как есть
    92→    92→      if (t === "") return "";
    93→    93→      if (ch !== lower) return t.charAt(0).toUpperCase() + t.slice(1);
    94→    94→      return t;
    95→    95→    })
    96→    96→    .join("");
    97→    97→}

   110→   110→function toggleSet(
   111→   111→  setter: React.Dispatch<React.SetStateAction<Set<string>>>,
   112→   112→  id: string
   113→   113→) {
   114→   114→  setter((prev) => {
   115→   115→    const next = new Set(prev);
   116→   116→    if (next.has(id)) next.delete(id);
   117→   117→    else next.add(id);
   118→   118→    return next;
   119→   119→  });
   120→   120→}
