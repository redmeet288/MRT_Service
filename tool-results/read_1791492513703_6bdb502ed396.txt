   175→type CityDialogState = { mode: "create" } | { mode: "edit"; city: City } | null;
   176→
   177→function CitiesTab() {
   178→  const qc = useQueryClient();
   179→  const { data: session } = useSession();
   180→  const isMP = (session?.user as SessionUserWithRole)?.role === "MP";
   181→  const [selectedId, setSelectedId] = useState<string | null>(null);
   182→  const [search, setSearch] = useState("");
   183→  const [sort, setSort] = useState<SortState | null>(null);
   184→  const [dialog, setDialog] = useState<CityDialogState>(null);
   185→
   186→  const { data: cities, isLoading } = useQuery<City[]>({
   187→    queryKey: ["cities"],
   188→    queryFn: () => api.get("/api/app/cities"),
   189→  });
   190→
   191→  const filtered = useMemo(() => {
   192→    const q = search.trim().toLowerCase();
   193→    const list = cities ?? [];
   194→    return q ? list.filter((c) => c.name.toLowerCase().includes(q)) : list;
   195→  }, [cities, search]);
   196→
   197→  const sorted = useMemo(
   198→    () =>
   199→      applySort(filtered, sort, (c, f) => {
   200→        switch (f) {
   201→          case "name":
   202→            return c.name;
   203→          case "lpus":
   204→            return c._count?.lpus ?? 0;
   205→          case "doctors":
   206→            return c._count?.doctors ?? 0;
   207→          default:
   208→            return null;
   209→        }
   210→      }),
   211→    [filtered, sort]
   212→  );
   213→
   214→  const selected = (cities ?? []).find((c) => c.id === selectedId) ?? null;
   215→
   216→  const remove = useMutation({
   217→    mutationFn: (id: string) => api.del(`/api/app/cities?id=${id}`),
   218→    onSuccess: () => {
   219→      toast.success("Город удален");
   220→      setSelectedId(null);
   221→      qc.invalidateQueries({ queryKey: ["cities"] });
   222→    },
   223→    onError: (e: Error) => toast.error(e.message),
   224→  });
   225→
   226→  return (
   227→    <>
   228→      <DictWindow
   229→        icon={<MapPin className="h-4 w-4 text-sky" strokeWidth={2} />}
   230→        title="Города"
   231→        count={filtered.length}
   232→        toolbar={
   233→          <>
   234→            <DictCrudButtons
   235→              onCreate={() => setDialog({ mode: "create" })}
   236→              onEdit={() => selected && setDialog({ mode: "edit", city: selected })}
   237→              canEdit={!!selected}
   238→              deleteSlot={
   239→                isMP ? (
   240→                  <DeleteButton
   241→                    disabled={!selected}
   242→                    title={`Удалить город «${selected?.name ?? ""}»?`}
   243→                    onConfirm={() => {
   244→                      if (selected) remove.mutateAsync(selected.id).catch(() => undefined);
   245→                    }}
   246→                  />
   247→                ) : undefined
   248→              }
   249→            />
   250→            <DictImportExport kind="cities" />
   251→            <OnecSearch value={search} onChange={setSearch} placeholder="Поиск по названию города…" />
   252→          </>
   253→        }
   254→      >
   255→        {isLoading ? (
   256→          <div className="p-4"><RowsSkeleton /></div>
   257→        ) : filtered.length === 0 ? (
   258→          <EmptyState
   259→            className="py-8"
   260→            title={search ? "Ничего не найдено" : "Городов пока нет"}
   261→            hint={search ? "Измените условие поиска" : "Нажмите «Создать» или импортируйте список из файла"}
   262→          />
   263→        ) : (
   264→          <div className="max-h-96 overflow-y-auto">
   265→            <table className="onec-table">
   266→              <thead>
   267→                <tr>
   268→                  <SortableTh label="Название" field="name" sort={sort} onSort={(f) => setSort(cycleSort(sort, f))} />
   269→                  <SortableTh label="ЛПУ" field="lpus" sort={sort} onSort={(f) => setSort(cycleSort(sort, f))} />
   270→                  <SortableTh label="НВ" field="doctors" sort={sort} onSort={(f) => setSort(cycleSort(sort, f))} />
   271→                </tr>
   272→              </thead>
   273→              <tbody>
   274→                {sorted.map((c) => (
   275→                  <tr
   276→                    key={c.id}
   277→                    onClick={() => setSelectedId(c.id)}
   278→                    className={cn("cursor-pointer", selectedId === c.id && "onec-row-selected")}
   279→                  >
   280→                    <td>
   281→                      <button
   282→                        type="button"
   283→                        className="font-medium text-brand hover:underline text-left"
   284→                        title="Изменить"
   285→                        onClick={(e) => {
   286→                          e.stopPropagation();
   287→                          setDialog({ mode: "edit", city: c });
   288→                        }}
   289→                      >
   290→                        {c.name}
   291→                      </button>
   292→                    </td>
   293→                    <td>{c._count?.lpus ?? 0}</td>
   294→                    <td>{c._count?.doctors ?? 0}</td>
   295→                  </tr>
   296→                ))}
   297→              </tbody>
   298→            </table>
   299→          </div>
   300→        )}
   301→      </DictWindow>
   302→
   303→      {dialog && (
   304→        <CityDialog
   305→          editing={dialog.mode === "edit" ? dialog.city : null}
   306→          onClose={() => setDialog(null)}
   307→        />
   308→      )}
   309→    </>
   310→  );
   311→}

   313→function CityDialog({ editing, onClose }: { editing: City | null; onClose: () => void }) {
   314→  const qc = useQueryClient();
   315→  const [name, setName] = useState(() => editing?.name ?? "");
   316→
   317→  const save = useMutation({
   318→    mutationFn: () =>
   319→      editing
   320→        ? api.put("/api/app/cities", { id: editing.id, name })
   321→        : api.post("/api/app/cities", { name }),
   322→    onSuccess: () => {
   323→      toast.success(editing ? "Город обновлен" : "Город добавлен");
   324→      qc.invalidateQueries({ queryKey: ["cities"] });
   325→      onClose();
   326→    },
   327→    onError: (e: Error) => toast.error(e.message),
   328→  });
   329→
   330→  return (
   331→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   332→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-md">
   333→        <DialogHeader>
   334→          <DialogTitle className="font-display text-brand">
   335→            {editing ? "Редактирование города" : "Новый город"}
   336→          </DialogTitle>
   337→          <DialogDescription>
   338→            {editing ? "Измените название и нажмите «Сохранить»." : "Введите название города — он станет доступен для ЛПУ и врачей."}
   339→          </DialogDescription>
   340→        </DialogHeader>
   341→        <div className="space-y-4 py-2">
   342→          <div className="space-y-1.5">
   343→            <Label>Название</Label>
   344→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Например: Казань" autoFocus />
   345→          </div>
   346→        </div>
   347→        <DialogFooter>
   348→          <Button variant="outline" onClick={onClose}>Отмена</Button>
   349→          <Button onClick={() => save.mutate()} disabled={!name.trim() || save.isPending}>
   350→            {editing ? "Сохранить" : "Добавить"}
   351→          </Button>
   352→        </DialogFooter>
   353→      </DialogContent>
   354→    </Dialog>
   355→  );
   356→}

  1119→      {dialog && (

  1125→    </>
  1126→  );
