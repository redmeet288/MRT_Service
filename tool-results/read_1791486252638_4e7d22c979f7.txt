    51→/** Пользователь сессии: JWT-колбэк кладет uid и role (типизация next-auth по умолчанию их не знает) */

    54→/** Сегодняшняя дата в формате YYYY-MM-DD (локальная зона) */
    55→function todayISO(): string {
    56→  const d = new Date();
    57→  return new Date(d.getTime() - d.getTimezoneOffset() * 60000).toISOString().slice(0, 10);
    58→}
    59→
    60→/** ISO-дата (из drill-down дашборда) → yyyy-MM-dd для input[type=date] */
    61→function isoToYMD(iso: string | undefined): string {
    62→  if (!iso) return "";
    63→  const d = new Date(iso);
    64→  return Number.isNaN(d.getTime()) ? "" : format(d, "yyyy-MM-dd");
    65→}

   437→/* ============================================================
   438→   Диалог регистрации / редактирования заявки (строго по ТЗ)

   440→interface RequestFormState {
   441→  date: string;
   442→  patientName: string;
   443→  doctorId: string;
   444→  lpuId: string;
   445→  destination: string;
   446→  assignedId: string;
   447→  doneId: string;
   448→  comment: string;
   449→}
   450→
   451→/** Начальное состояние формы: из заявки при редактировании / пустая (дата — сегодня) при создании */
   452→function initRequestForm(request: RequestItem | null): RequestFormState {
   453→  if (!request) {
   454→    return {
   455→      date: todayISO(),
   456→      patientName: "",
   457→      doctorId: "",
   458→      lpuId: "",
   459→      destination: "",
   460→      assignedId: "",
   461→      doneId: "",
   462→      comment: "",
   463→    };
   464→  }
   465→  return {
   466→    date: request.date.slice(0, 10),
   467→    patientName: request.patientName,
   468→    doctorId: request.doctorId,
   469→    lpuId: request.lpuId,
   470→    destination: request.destination ?? "",
   471→    assignedId: request.assignedId ?? "",
   472→    doneId: request.doneId ?? "",
   473→    comment: request.comment ?? "",
   474→  };
   475→}
   476→
   477→function RequestFormDialog({
   478→  request,
   479→  onOpenChange,
   480→}: {
   481→  request: RequestItem | null;
   482→  onOpenChange: (o: boolean) => void;
   483→}) {
   484→  const qc = useQueryClient();
   485→  const [form, setForm] = useState<RequestFormState>(() => initRequestForm(request));
   486→
   487→  const { data: doctors } = useQuery<DoctorItem[]>({
   488→    queryKey: ["doctors", "all"],
   489→    queryFn: () => api.get("/api/app/doctors"),
   490→  });
   491→  const { data: nomenclature } = useQuery<NomenclatureItem[]>({
   492→    queryKey: ["nomenclature", "active"],
   493→    queryFn: () => api.get("/api/app/nomenclature?active=1"),
   494→  });
   495→  const { data: lpus } = useQuery<CityData[]>({
   496→    queryKey: ["lpus"],
   497→    queryFn: () => api.get("/api/app/lpus"),
   498→  });
   499→
   500→  const selectedDoctor = useMemo(
   501→    () => doctors?.find((d) => d.id === form.doctorId) ?? null,
   502→    [doctors, form.doctorId]
   503→  );
   504→
   505→  // ЛПУ, привязанные к выбранному НВ (данные lpus уже приходят в ответе doctors)
   506→  const doctorLpus = useMemo(() => selectedDoctor?.lpus.map((x) => x.lpu) ?? [], [selectedDoctor]);
   507→
   508→  // Режим выбора ЛПУ: 1 привязка — авто; несколько — Select из привязок; 0 — свободный Select
   509→  const lpuMode: "none" | "single" | "multi" | "free" = !selectedDoctor
   510→    ? "none"
   511→    : doctorLpus.length === 1
   512→      ? "single"
   513→      : doctorLpus.length > 1
   514→        ? "multi"
   515→        : "free";
   516→
   517→  // Варианты для режима «несколько ЛПУ» (+ защита для редактируемой заявки)
   518→  const multiOptions = useMemo(() => {
   519→    const opts = [...doctorLpus];
   520→    if (request && request.lpuId && !opts.some((l) => l.id === request.lpuId)) {
   521→      opts.push(request.lpu);
   522→    }
   523→    return opts;
   524→  }, [doctorLpus, request]);
   525→
   526→  // Итоговый lpuId: при единственном ЛПУ НВ — подставляется автоматически
   527→  const resolvedLpuId = lpuMode === "single" ? (doctorLpus[0]?.id ?? "") : form.lpuId;
   528→
   529→  const lpuCity = useMemo(() => {
   530→    if (!resolvedLpuId) return null;
   531→    const fromDoctor = doctorLpus.find((l) => l.id === resolvedLpuId);
   532→    if (fromDoctor) return fromDoctor.city.name;
   533→    if (request?.lpu.id === resolvedLpuId) return request.lpu.city.name;
   534→    return lpus?.find((l) => l.id === resolvedLpuId)?.city.name ?? null;
   535→  }, [resolvedLpuId, doctorLpus, lpus, request]);
   536→
   537→  const save = useMutation({
   538→    mutationFn: () => {
   539→      const body = {
   540→        date: form.date,
   541→        patientName: form.patientName.trim(),
   542→        doctorId: form.doctorId,
   543→        lpuId: resolvedLpuId,
   544→        destination: form.destination.trim() || null,
   545→        assignedId: form.assignedId || null,
   546→        doneId: form.doneId || null,
   547→        comment: form.comment.trim() || null,
   548→      };
   549→      return request
   550→        ? api.put(`/api/app/requests/${request.id}`, body)
   551→        : api.post("/api/app/requests", body);
   552→    },
   553→    onSuccess: () => {
   554→      toast.success(request ? "Заявка обновлена" : "Заявка зарегистрирована");
   555→      qc.invalidateQueries({ queryKey: ["requests"] });
   556→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   557→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   558→      onOpenChange(false);
   559→    },
   560→    onError: (e: Error) => toast.error(e.message),
   561→  });
   562→
   563→  return (
   564→    <Dialog open onOpenChange={onOpenChange}>
   565→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-lg">
   566→        <DialogHeader>
   567→          <DialogTitle className="font-display text-brand">
   568→            {request ? "Карточка заявки" : "Регистрация заявки"}
   569→          </DialogTitle>
   570→          <DialogDescription>
   571→            {request
   572→              ? "Просмотр и редактирование направления пациента на обследование"
   573→              : "Учет направления пациента на обследование"}
   574→          </DialogDescription>
   575→        </DialogHeader>
   576→
   577→        <div className="space-y-4 py-1">
   578→          {/* 1. Дата */}
   579→          <div className="space-y-1.5">
   580→            <Label>
   581→              Дата <span className="text-terra-dark">*</span>
   582→            </Label>
   583→            <Input
   584→              type="date"
   585→              value={form.date}
   586→              onChange={(e) => setForm({ ...form, date: e.target.value })}
   587→            />
   588→          </div>
   589→
   590→          {/* 2. ФИО пациента */}
   591→          <div className="space-y-1.5">
   592→            <Label>
   593→              ФИО пациента <span className="text-terra-dark">*</span>
   594→            </Label>
   595→            <Input
   596→              value={form.patientName}
   597→              onChange={(e) => setForm({ ...form, patientName: e.target.value })}
   598→              placeholder="Иванов Иван Иванович"
   599→            />
   600→          </div>
   601→
   602→          {/* 3. ФИО НВ */}
   603→          <div className="space-y-1.5">
   604→            <Label>
   605→              ФИО НВ <span className="text-terra-dark">*</span>
   606→            </Label>
   607→            <Select
   608→              value={form.doctorId}
   609→              onValueChange={(v) => {
   610→                const nd = doctors?.find((d) => d.id === v);
   611→                const dl = nd?.lpus.map((x) => x.lpu) ?? [];
   612→                setForm((f) => ({
   613→                  ...f,
   614→                  doctorId: v,
   615→                  lpuId: dl.length === 1 ? dl[0].id : "",
   616→                }));
   617→              }}
   618→            >
   619→              <SelectTrigger className="bg-white">
   620→                <SelectValue placeholder="Выберите направляющего врача" />
   621→              </SelectTrigger>
   622→              <SelectContent className="max-h-72">
   623→                {(doctors ?? []).map((d) => (
   624→                  <SelectItem key={d.id} value={d.id}>
   625→                    {d.fullName}
   626→                  </SelectItem>
   627→                ))}
   628→              </SelectContent>
   629→            </Select>
   630→          </div>
   631→
   632→          {/* 4. ЛПУ и город — автозаполнение из карточки НВ */}
   633→          <div className="space-y-1.5">
   634→            <Label>
   635→              ЛПУ <span className="text-terra-dark">*</span>
   636→            </Label>
   637→            {lpuMode === "none" && (
   638→              <p className="text-xs text-muted-foreground">
   639→                Сначала выберите НВ — ЛПУ подставится из его карточки
   640→              </p>
   641→            )}
   642→            {lpuMode === "single" && (
   643→              <>
   644→                <Input
   645→                  disabled
   646→                  readOnly
   647→                  value={
   648→                    doctorLpus[0] ? `${doctorLpus[0].name} · ${doctorLpus[0].city.name}` : "—"
   649→                  }
   650→                  className="bg-white text-muted-foreground"
   651→                />
   652→                <p className="text-[11px] text-muted-foreground">
   653→                  Единственное ЛПУ врача — подставлено автоматически
   654→                </p>
   655→              </>
   656→            )}
   657→            {lpuMode === "multi" && (
   658→              <Select
   659→                value={form.lpuId}
   660→                onValueChange={(v) => setForm({ ...form, lpuId: v })}
   661→              >
   662→                <SelectTrigger className="bg-white">
   663→                  <SelectValue placeholder="Выберите ЛПУ из привязок НВ" />
   664→                </SelectTrigger>
   665→                <SelectContent className="max-h-72">
   666→                  {multiOptions.map((l) => (
   667→                    <SelectItem key={l.id} value={l.id}>
   668→                      {l.name} · {l.city.name}
   669→                    </SelectItem>
   670→                  ))}
   671→                </SelectContent>
   672→              </Select>
   673→            )}
   674→            {lpuMode === "free" && (
   675→              <Select
   676→                value={form.lpuId}
   677→                onValueChange={(v) => setForm({ ...form, lpuId: v })}
   678→              >
   679→                <SelectTrigger className="bg-white">
   680→                  <SelectValue placeholder="У НВ нет привязок — выберите ЛПУ" />
   681→                </SelectTrigger>
   682→                <SelectContent className="max-h-72">
   683→                  {(lpus ?? []).map((l) => (
   684→                    <SelectItem key={l.id} value={l.id}>
   685→                      {l.name} · {l.city.name}
   686→                    </SelectItem>
   687→                  ))}
   688→                </SelectContent>
   689→              </Select>
   690→            )}
   691→            {lpuCity && (
   692→              <p className="text-[11px] text-muted-foreground">Город: {lpuCity}</p>
   693→            )}
   694→          </div>
   695→
   696→          {/* 5. Куда направление */}
   697→          <div className="space-y-1.5">
   698→            <Label>Куда направление</Label>
   699→            <Input
   700→              value={form.destination}
   701→              onChange={(e) => setForm({ ...form, destination: e.target.value })}
   702→              placeholder="Опционально — отделение, кабинет"
   703→            />
   704→          </div>
   705→
   706→          {/* 6. Что назначено */}
   707→          <div className="space-y-1.5">
   708→            <Label>
   709→              Что назначено <span className="text-terra-dark">*</span>
   710→            </Label>
   711→            <Select
   712→              value={form.assignedId}
   713→              onValueChange={(v) => setForm({ ...form, assignedId: v })}
   714→            >
   715→              <SelectTrigger className="bg-white">
   716→                <SelectValue
   717→                  placeholder={
   718→                    nomenclature?.length ? "Выберите позицию номенклатуры" : "Справочник номенклатуры пуст"
   719→                  }
   720→                />
   721→              </SelectTrigger>
   722→              <SelectContent className="max-h-72">
   723→                {(nomenclature ?? []).map((n) => (
   724→                  <SelectItem key={n.id} value={n.id}>
   725→                    {n.name}
   726→                  </SelectItem>
   727→                ))}
   728→              </SelectContent>
   729→            </Select>
   730→          </div>
   731→
   732→          {/* 7. Что сделано (опционально — можно заполнить позже) */}
   733→          <div className="space-y-1.5">
   734→            <Label>Что сделано</Label>
   735→            <Select
   736→              value={form.doneId || NONE}
   737→              onValueChange={(v) => setForm({ ...form, doneId: v === NONE ? "" : v })}
   738→            >
   739→              <SelectTrigger className="bg-white">
   740→                <SelectValue placeholder="Пока не выполнено" />
   741→              </SelectTrigger>
   742→              <SelectContent className="max-h-72">
   743→                <SelectItem value={NONE}>Пока не выполнено</SelectItem>
   744→                {(nomenclature ?? []).map((n) => (
   745→                  <SelectItem key={n.id} value={n.id}>
   746→                    {n.name}
   747→                  </SelectItem>
   748→                ))}
   749→              </SelectContent>
   750→            </Select>
   751→            <p className="text-[11px] text-muted-foreground">
   752→              Опционально — можно заполнить позже при редактировании заявки
   753→            </p>
   754→          </div>
   755→
   756→          {/* 8. Комментарий (опционально) */}
   757→          <div className="space-y-1.5">
   758→            <Label>Комментарий</Label>
   759→            <Textarea
   760→              rows={2}
   761→              value={form.comment}
   762→              onChange={(e) => setForm({ ...form, comment: e.target.value })}
   763→              placeholder="Опционально — заметки по заявке, договоренности…"
   764→            />
   765→          </div>
   766→        </div>
   767→
   768→        <DialogFooter>
   769→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   770→            Отмена
   771→          </Button>
   772→          <Button
   773→            onClick={() => save.mutate()}
   774→            disabled={
   775→              !form.date ||
   776→              !form.patientName.trim() ||
   777→              !form.doctorId ||
   778→              !resolvedLpuId ||
   779→              !form.assignedId ||
   780→              save.isPending
   781→            }
   782→            className="bg-azure hover:bg-azure-dark"
   783→          >
   784→            {save.isPending && <ClipboardList className="h-4 w-4 animate-spin" strokeWidth={2} />}
   785→            {request ? "Сохранить" : "Зарегистрировать"}
   786→          </Button>
   787→        </DialogFooter>
   788→      </DialogContent>
   789→    </Dialog>
   790→  );
   791→}
