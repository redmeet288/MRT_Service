    74→/* ============================================================
    75→   Транслитерация для PDF (jsPDF стандартно не поддерживает
    76→   кириллицу) — простая карта а-я → a-ya
    77→   ============================================================ */
    78→const TRANSLIT: Record<string, string> = {
    79→  а: "a", б: "b", в: "v", г: "g", д: "d", е: "e", ё: "e", ж: "zh", з: "z",
    80→  и: "i", й: "y", к: "k", л: "l", м: "m", н: "n", о: "o", п: "p", р: "r",
    81→  с: "s", т: "t", у: "u", ф: "f", х: "kh", ц: "ts", ч: "ch", ш: "sh",
    82→  щ: "sch", ъ: "", ы: "y", ь: "", э: "e", ю: "yu", я: "ya",
    83→};
    84→
    85→function translit(value: string): string {
    86→  return (value ?? "")
    87→    .split("")
    88→    .map((ch) => {
    89→      const lower = ch.toLowerCase();
    90→      const t = TRANSLIT[lower];
    91→      if (t === undefined) return ch; // латиница, цифры, пунктуация — как есть
    92→      if (t === "") return "";
    93→      if (ch !== lower) return t.charAt(0).toUpperCase() + t.slice(1);
    94→      return t;
    95→    })
    96→    .join("");
    97→}

   110→function toggleSet(
   111→  setter: React.Dispatch<React.SetStateAction<Set<string>>>,
   112→  id: string
   113→) {
   114→  setter((prev) => {
   115→    const next = new Set(prev);
   116→    if (next.has(id)) next.delete(id);
   117→    else next.add(id);
   118→    return next;
   119→  });
   120→}

   543→            {/* Задел этапа 2: колонка «Начисленный бонус» (недоступна на старте) */}
   544→            <div
   545→              className="flex items-center gap-2.5 rounded-lg border border-border bg-soft px-3 py-2 opacity-75"
   546→              title="Функционал этапа 2 — бонусная система"
   547→            >
   548→              <Coins className="h-4 w-4 text-sky/70" strokeWidth={2} />
   549→              <div>
   550→                <p className="text-[11px] font-medium leading-tight text-muted-foreground">
   551→                  Колонка «Начисленный бонус»
   552→                </p>
   553→                <p className="text-[10px] leading-tight text-muted-foreground/70">
   554→                  Этап 2 · бонусная система
   555→                </p>
   556→              </div>
   557→              <Switch
   558→                checked={false}
   559→                disabled
   560→                aria-readonly
   561→                aria-label="Колонка «Начисленный бонус» — этап 2"
   562→              />
   563→            </div>
   564→            <Button
   565→              variant="outline"
   566→              onClick={() => doExport("Excel")}
   567→              disabled={!flatRows.length}
   568→              className="gap-1.5 border-sky/40 text-brand hover:bg-sky-light"
   569→            >
   570→              <FileSpreadsheet className="h-4 w-4 text-azure" strokeWidth={2} /> Excel
   571→            </Button>
   572→            <Button
   573→              onClick={() => doExport("PDF")}
   574→              disabled={!flatRows.length}
   575→              className="gap-1.5"
   576→            >
   577→              <FileDown className="h-4 w-4" strokeWidth={2} /> PDF
   578→            </Button>
   579→          </div>
   580→        </div>
   581→
   582→        {/* Верхний итог */}
   583→        <div className="mb-5 flex flex-wrap items-center gap-x-8 gap-y-3 rounded-lg bg-soft px-4 py-3">
   584→          <div>
   585→            <p className="text-[11px] uppercase tracking-wide text-muted-foreground">
   586→              Всего заявок
   587→            </p>
   588→            <p className="font-display text-2xl font-bold text-brand">{total}</p>
   589→          </div>
   590→          <div>
   591→            <p className="text-[11px] uppercase tracking-wide text-muted-foreground">
   592→              Выполнено
   593→            </p>
   594→            <p className="font-display text-2xl font-bold text-azure-dark">{doneTotal}</p>
   595→          </div>
   596→          <Badge className={cn("font-display text-sm", pctBadgeClass(pct(doneTotal, total)))}>
   597→            {pct(doneTotal, total)}% выполнения
   598→          </Badge>
   599→          {reportQuery.isFetching && (
   600→            <span className="text-xs text-muted-foreground">Обновление…</span>
   601→          )}
   602→        </div>
   603→
   604→        {reportQuery.error ? (
   605→          <div className="rounded-xl border border-terra/50 bg-terra-light p-6 text-sm text-brand-dark">
   606→            Не удалось построить отчет: {(reportQuery.error as Error).message}
   607→          </div>
   608→        ) : reportQuery.isLoading || !data ? (
   609→          <RowsSkeleton rows={6} />
   610→        ) : data.tree.length === 0 ? (
   611→          <EmptyState
   612→            title="Нет заявок за выбранный период"
   613→            hint="Измените фильтры или расширьте период — данные появятся здесь"
   614→          />
   615→        ) : (
   616→          <div className="max-h-[620px] space-y-3 overflow-y-auto pr-1">
   617→            {data.tree.map((mp) => (
   618→              <div key={mp.id} className="overflow-hidden rounded-xl border border-border">
   619→                {/* Уровень 1: МП */}
   620→                <GroupRow
   621→                  icon={UserRound}
   622→                  label={mp.name}
   623→                  total={mp.total}
   624→                  done={mp.done}
   625→                  depth={0}
   626→                  open={openMp.has(mp.id)}
   627→                  onToggle={() => toggleSet(setOpenMp, mp.id)}
   628→                />
   629→                {openMp.has(mp.id) &&
   630→                  (mp.lpus.length === 0 ? (
   631→                    <div className="border-t border-border/60 bg-soft/40 px-4">
   632→                      <EmptyState title="У МП нет заявок по фильтрам" className="py-5" />
   633→                    </div>
   634→                  ) : (
   635→                    <div className="space-y-2 border-t border-border/60 bg-soft/40 p-2">
   636→                      {mp.lpus.map((lpu) => (
   637→                        <div
   638→                          key={lpu.id}
   639→                          className="overflow-hidden rounded-lg border border-border/70 bg-white"
   640→                        >
   641→                          {/* Уровень 2: ЛПУ */}
   642→                          <GroupRow
   643→                            icon={Building2}
   644→                            label={lpu.name}
   645→                            hint={lpu.city}
   646→                            total={lpu.total}
   647→                            done={lpu.done}
   648→                            depth={1}
   649→                            open={openLpu.has(lpu.id)}
   650→                            onToggle={() => toggleSet(setOpenLpu, lpu.id)}
   651→                          />
   652→                          {openLpu.has(lpu.id) &&
   653→                            (lpu.doctors.length === 0 ? (
   654→                              <div className="border-t border-border/60 px-4">
   655→                                <EmptyState title="Нет заявок по этому ЛПУ" className="py-4" />
   656→                              </div>
   657→                            ) : (
   658→                              <div className="border-t border-border/60">
   659→                                {lpu.doctors.map((doc) => (
   660→                                  <div key={doc.id}>
   661→                                    {/* Уровень 3: НВ */}
   662→                                    <GroupRow
   663→                                      icon={Stethoscope}
   664→                                      label={doc.name}
   665→                                      total={doc.total}
   666→                                      done={doc.done}
   667→                                      depth={2}
   668→                                      open={openDoc.has(doc.id)}
   669→                                      onToggle={() => toggleSet(setOpenDoc, doc.id)}
   670→                                    />
   671→                                    {/* Уровень 4: детализация заявок */}
   672→                                    {openDoc.has(doc.id) && <RequestsTable rows={doc.requests} />}
   673→                                  </div>
   674→                                ))}
   675→                              </div>
   676→                            ))}
   677→                        </div>
   678→                      ))}
   679→                    </div>
   680→                  ))}
   681→              </div>
   682→            ))}
   683→          </div>
   684→        )}
   685→      </section>
   686→
   687→      {/* —— Диалог сохранения пресета —— */}
   688→      <Dialog open={saveOpen} onOpenChange={setSaveOpen}>
   689→        <DialogContent className="sm:max-w-sm">
   690→          <DialogHeader>
   691→            <DialogTitle className="font-display text-brand">
   692→              Сохранить пресет фильтров
   693→            </DialogTitle>
   694→            <DialogDescription>
   695→              Пресет доступен только вам в списке «Пресеты».
   696→            </DialogDescription>
   697→          </DialogHeader>
   698→          <div className="space-y-1.5 py-2">
   699→            <Label>Название пресета</Label>
   700→            <Input
   701→              value={presetName}
   702→              onChange={(e) => setPresetName(e.target.value)}
   703→              placeholder="Например: Квартал · МРТ головы"
   704→              onKeyDown={(e) => {
   705→                if (e.key === "Enter" && presetName.trim()) savePreset.mutate();
   706→              }}
   707→            />
   708→            <p className="text-[11px] text-muted-foreground">
   709→              Сохраняются: период, МП, ЛПУ, НВ, назначение, выполненность.
   710→            </p>
   711→          </div>
   712→          <DialogFooter>
   713→            <Button variant="outline" onClick={() => setSaveOpen(false)}>
   714→              Отмена
   715→            </Button>
   716→            <Button
   717→              onClick={() => savePreset.mutate()}
   718→              disabled={!presetName.trim() || savePreset.isPending}
   719→            >
   720→              Сохранить
   721→            </Button>
   722→          </DialogFooter>
   723→        </DialogContent>
   724→      </Dialog>
   725→    </div>
   726→  );
   727→}

   729→/* ============ Строка-заголовок группы дерева ============ */
   730→function GroupRow({
   731→  icon: Icon,
   732→  label,
   733→  hint,
   734→  total,
   735→  done,
   736→  depth,
   737→  open,
   738→  onToggle,
   739→}: {
   740→  icon: React.ElementType;
   741→  label: string;
   742→  hint?: string;
   743→  total: number;
   744→  done: number;
   745→  depth: 0 | 1 | 2;
   746→  open: boolean;
   747→  onToggle: () => void;
   748→}) {
   749→  return (
   750→    <button
   751→      type="button"
   752→      onClick={onToggle}
   753→      aria-expanded={open}
   754→      className={cn(
   755→        "flex w-full items-center gap-2.5 py-2.5 pr-4 text-left transition-colors hover:bg-sky-light/70",
   756→        depth === 0 && "bg-sky-light/50 px-4",
   757→        depth === 1 && "bg-white pl-6",
   758→        depth === 2 && "bg-white pl-10"
   759→      )}
   760→    >
   761→      <ChevronRight
   762→        className={cn(
   763→          "h-4 w-4 shrink-0 text-sky transition-transform duration-200",
   764→          open && "rotate-90"
   765→        )}
   766→        strokeWidth={2}
   767→      />
   768→      <span
   769→        className={cn(
   770→          "inline-flex h-6 w-6 shrink-0 items-center justify-center rounded-md",
   771→          depth === 0 ? "bg-white text-brand" : "bg-sky-light text-brand"
   772→        )}
   773→      >
   774→        <Icon className="h-3.5 w-3.5" strokeWidth={2} />
   775→      </span>
   776→      <span
   777→        className={cn(
   778→          "truncate text-brand",
   779→          depth === 0 ? "font-display text-sm font-bold" : "text-sm font-medium"
   780→        )}
   781→      >
   782→        {label}
   783→      </span>
   784→      {hint && <span className="truncate text-xs text-muted-foreground">· {hint}</span>}
   785→      <span className="ml-auto flex shrink-0 items-center gap-3 text-xs">
   786→        <span className="text-muted-foreground">
   787→          Заявок: <span className="font-display font-bold text-brand">{total}</span>
   788→        </span>
   789→        <span className="hidden text-muted-foreground sm:inline">
   790→          Выполнено: <span className="font-display font-bold text-azure-dark">{done}</span>
   791→        </span>
   792→        <span
   793→          className={cn(
   794→            "rounded-full px-2 py-0.5 font-display text-[10px] font-semibold",
   795→            pctBadgeClass(pct(done, total))
   796→          )}
   797→        >
   798→          {pct(done, total)}%
   799→        </span>
   800→      </span>
   801→    </button>
   802→  );
   803→}
