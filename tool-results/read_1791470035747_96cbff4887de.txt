   340→   340→const emptyDoctorForm: DoctorFormState = {
   341→   341→  fullName: "",
   342→   342→  phone: "",
   343→   343→  birthDate: "",
   344→   344→  statusId: NONE,
   345→   345→  mpId: "",

   351→   351→/** Начальное состояние формы: из карточки при редактировании / пустая при создании */
   352→   352→function initDoctorForm(doctor: EditableDoctor | null): DoctorFormState {
   353→   353→  if (!doctor) return emptyDoctorForm;
   354→   354→  return {
   355→   355→    fullName: doctor.fullName,
   356→   356→    phone: doctor.phone ?? "",
   357→   357→    birthDate: doctor.birthDate ? doctor.birthDate.slice(0, 10) : "",
   358→   358→    statusId: doctor.statusId ?? NONE,
   359→   359→    mpId: doctor.mpId,
   360→   360→    cityId: doctor.cityId ?? NONE,
   361→   361→    lpuIds: doctor.lpus?.map((x) => x.lpu.id) ?? [],
   362→   362→    comment: doctor.comment ?? "",
   363→   363→  };
   364→   364→}

   366→   366→function DoctorFormDialog({
   367→   367→  onOpenChange,
   368→   368→  doctor,
   369→   369→}: {
   370→   370→  onOpenChange: (o: boolean) => void;
   371→   371→  doctor: EditableDoctor | null;
   372→   372→}) {
   373→   373→  const qc = useQueryClient();
   374→   374→  const [form, setForm] = useState<DoctorFormState>(() => initDoctorForm(doctor));
   375→   375→
   376→   376→  const { data: users } = useQuery<UserBasic[]>({
   377→   377→    queryKey: ["users-basic"],
   378→   378→    queryFn: () => api.get("/api/app/users?basic=1"),
   379→   379→  });
   380→   380→  const { data: statuses } = useQuery<CooperationStatus[]>({
   381→   381→    queryKey: ["statuses"],
   382→   382→    queryFn: () => api.get("/api/app/statuses"),
   383→   383→  });
   384→   384→  const { data: cities } = useQuery<City[]>({
   385→   385→    queryKey: ["cities"],
   386→   386→    queryFn: () => api.get("/api/app/cities"),
   387→   387→  });
   388→   388→  const { data: lpus } = useQuery<CityData[]>({
   389→   389→    queryKey: ["lpus"],
   390→   390→    queryFn: () => api.get("/api/app/lpus"),
   391→   391→  });
   392→   392→
   393→   393→  const mps = useMemo(() => (users ?? []).filter((u) => u.role === "MP"), [users]);
   394→   394→
   395→   395→  /** Переключение ЛПУ; автоподстановка города первого ЛПУ, если город не указан */
   396→   396→  function toggleLpu(id: string, checked: boolean) {
   397→   397→    setForm((f) => {
   398→   398→      const lpuIds = checked ? [...f.lpuIds, id] : f.lpuIds.filter((x) => x !== id);
   399→   399→      let cityId = f.cityId;
   400→   400→      if (checked && cityId === NONE && lpuIds.length > 0) {
   401→   401→        const first = lpus?.find((l) => l.id === lpuIds[0]);
   402→   402→        if (first) cityId = first.city.id;
   403→   403→      }
   404→   404→      return { ...f, lpuIds, cityId };
   405→   405→    });
   406→   406→  }
   407→   407→
   408→   408→  const save = useMutation({
   409→   409→    mutationFn: () => {
   410→   410→      const body = {
   411→   411→        fullName: form.fullName.trim(),
   412→   412→        phone: form.phone.trim() || null,
   413→   413→        birthDate: form.birthDate || null,
   414→   414→        statusId: form.statusId === NONE ? null : form.statusId,
   415→   415→        mpId: form.mpId,
   416→   416→        cityId: form.cityId === NONE ? null : form.cityId,
   417→   417→        lpuIds: form.lpuIds,
   418→   418→        comment: form.comment.trim() || null,
   419→   419→      };
   420→   420→      return doctor ? api.put(`/api/app/doctors/${doctor.id}`, body) : api.post("/api/app/doctors", body);
   421→   421→    },
   422→   422→    onSuccess: () => {
   423→   423→      toast.success(doctor ? "Карточка врача обновлена" : "Врач добавлен");
   424→   424→      qc.invalidateQueries({ queryKey: ["doctors"] });
   425→   425→      qc.invalidateQueries({ queryKey: ["doctor-card"] });
   426→   426→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   427→   427→      onOpenChange(false);
   428→   428→    },
   429→   429→    onError: (e: Error) => toast.error(e.message),
   430→   430→  });
   431→   431→
   432→   432→  return (
   433→   433→    <Dialog open onOpenChange={onOpenChange}>
   434→   434→      <DialogContent className="sm:max-w-2xl">
   435→   435→        <DialogHeader>
   436→   436→          <DialogTitle className="font-display text-brand">
   437→   437→            {doctor ? "Редактирование врача" : "Новый направляющий врач"}
   438→   438→          </DialogTitle>
   439→   439→          <DialogDescription>
   440→   440→            Карточка НВ — ключевая сущность CRM: статус, курирующий МП и привязки к ЛПУ
   441→   441→          </DialogDescription>
   442→   442→        </DialogHeader>
   443→   443→
   444→   444→        <div className="grid gap-4 py-1 md:grid-cols-2">
   445→   445→          <div className="space-y-1.5">
   446→   446→            <Label>
   447→   447→              ФИО <span className="text-terra-dark">*</span>
   448→   448→            </Label>
   449→   449→            <Input
   450→   450→              value={form.fullName}
