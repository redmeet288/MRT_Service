     1→"use client";
     2→
     3→import { useEffect, useMemo, useRef, useState } from "react";
     4→import { useQuery } from "@tanstack/react-query";
     5→import {
     6→  addDays,
     7→  addMonths,
     8→  addWeeks,
     9→  eachDayOfInterval,
    10→  endOfMonth,
    11→  endOfWeek,
    12→  format,
    13→  getHours,
    14→  getMinutes,
    15→  isSameMonth,
    16→  isToday,
    17→  startOfMonth,
    18→  startOfWeek,
    19→} from "date-fns";
    20→import { ru } from "date-fns/locale";
    21→import { api, qs } from "@/lib/api-client";
    22→import type { DoctorItem, UserBasic, VisitItem } from "@/lib/types";
    23→import { VISIT_STATUS_LABELS } from "@/lib/types";
    24→import { EmptyState, RowsSkeleton, StatusChip } from "@/components/shared/ui-bits";
    25→import { VisitDialog } from "@/components/views/visit-dialog";
    26→import { Button } from "@/components/ui/button";
    27→import { Label } from "@/components/ui/label";
    28→import {
    29→  Select,
    30→  SelectContent,
    31→  SelectItem,
    32→  SelectTrigger,
    33→  SelectValue,
    34→} from "@/components/ui/select";
    35→import { ToggleGroup, ToggleGroupItem } from "@/components/ui/toggle-group";
    36→import {
    37→  CalendarClock,
    38→  CalendarDays,
    39→  CalendarPlus,
    40→  CalendarRange,
    41→  ChevronLeft,
    42→  ChevronRight,
    43→  Plus,
    44→} from "lucide-react";
    45→import { cn } from "@/lib/utils";
    46→
    47→type ViewMode = "month" | "week" | "day";
    48→type VisitStatus = VisitItem["status"];
    49→
    50→const WEEKDAYS = ["Пн", "Вт", "Ср", "Чт", "Пт", "Сб", "Вс"];
    51→
    52→/** Границы сетки часов вида «День» */
    53→const DAY_START_HOUR = 7;
    54→const DAY_END_HOUR = 21;
    55→const SLOT_HEIGHT = 56; // px — высота строки часа (min-h слота)
    56→
    57→/** Цвет чипа визита в сетке — по статусу (гайдлайн: лазурь / синий / терракот) */
    58→const CHIP_BG: Record<VisitStatus, string> = {
    59→  PLANNED: "bg-azure hover:bg-azure-dark",
    60→  CONDUCTED: "bg-brand hover:bg-brand-dark",
    61→  CANCELLED: "bg-terra hover:bg-terra-dark",
    62→};
    63→
    64→const cap = (s: string) => s.charAt(0).toUpperCase() + s.slice(1);
    65→const iso = (d: Date) => format(d, "yyyy-MM-dd");
    66→const fmtTime = (d: string | Date) => format(new Date(d), "HH:mm");
    67→const pad2 = (n: number) => String(n).padStart(2, "0");
    68→
    69→export function CalendarView() {
    70→  // —— Вид и навигация ——
    71→  const [mode, setMode] = useState<ViewMode>("month");
    72→  const [cursor, setCursor] = useState<Date>(new Date());
    73→
    74→  // —— Фильтры ("ALL" = без фильтра) ——
    75→  const [mpFilter, setMpFilter] = useState("ALL");
    76→  const [statusFilter, setStatusFilter] = useState("ALL");
    77→  const [doctorFilter, setDoctorFilter] = useState("ALL");
    78→
    79→  // —— Диалог (общий VisitDialog, монтируется условно) ——
    80→  const [dialogOpen, setDialogOpen] = useState(false);
    81→  const [editing, setEditing] = useState<VisitItem | null>(null);
    82→  const [dialogDate, setDialogDate] = useState<Date>(new Date());
    83→  const [dialogTime, setDialogTime] = useState<string | undefined>(undefined);
    84→
    85→  // —— Справочники для фильтров (диалог тянет НВ из общего кеша) ——
    86→  const { data: users } = useQuery<UserBasic[]>({
    87→    queryKey: ["users-basic"],
    88→    queryFn: () => api.get("/api/app/users?basic=1"),
    89→  });
    90→  const mps = useMemo(() => (users ?? []).filter((u) => u.role === "MP"), [users]);
    91→
    92→  const { data: doctors } = useQuery<DoctorItem[]>({
    93→    queryKey: ["doctors"],
    94→    queryFn: () => api.get("/api/app/doctors"),
    95→  });
    96→  const doctorOptions = useMemo(
    97→    () => (doctors ?? []).filter((d) => mpFilter === "ALL" || d.mpId === mpFilter),
    98→    [doctors, mpFilter]
    99→  );
   100→
   101→  // —— Период выборки (включая дни соседних месяцев в сетке; в дне — сам день) ——
   102→  const range = useMemo(() => {
   103→    if (mode === "month") {
   104→      return {
   105→        from: startOfWeek(startOfMonth(cursor), { weekStartsOn: 1 }),
   106→        to: endOfWeek(endOfMonth(cursor), { weekStartsOn: 1 }),
   107→      };
   108→    }
   109→    if (mode === "week") {
   110→      return {
   111→        from: startOfWeek(cursor, { weekStartsOn: 1 }),
   112→        to: endOfWeek(cursor, { weekStartsOn: 1 }),
   113→      };
   114→    }
   115→    return { from: cursor, to: cursor };
   116→  }, [cursor, mode]);
   117→
   118→  // —— Визиты ——
   119→  const visitsQuery = useQuery<VisitItem[]>({
   120→    queryKey: [
   121→      "visits",
   122→      mode,
   123→      iso(range.from),
   124→      iso(range.to),
   125→      mpFilter,
   126→      statusFilter,
   127→      doctorFilter,
   128→    ],
   129→    queryFn: () =>
   130→      api.get<VisitItem[]>(
   131→        `/api/app/visits${qs({
   132→          from: iso(range.from),
   133→          to: `${iso(range.to)}T23:59:59`,
   134→          mpId: mpFilter === "ALL" ? undefined : mpFilter,
   135→          status: statusFilter === "ALL" ? undefined : statusFilter,
   136→          doctorId: doctorFilter === "ALL" ? undefined : doctorFilter,
   137→        })}`
   138→      ),
   139→  });
   140→
   141→  // —— Группировка визитов по дням ——
   142→  const grouped = useMemo(() => {
   143→    const map = new Map<string, VisitItem[]>();
   144→    (visitsQuery.data ?? []).forEach((v) => {
   145→      const key = iso(new Date(v.date));
   146→      const arr = map.get(key) ?? [];
   147→      arr.push(v);
   148→      map.set(key, arr);
   149→    });
   150→    return map;
   151→  }, [visitsQuery.data]);
   152→
   153→  // —— Клетки месячной/недельной сетки (Пн–Вс) ——
   154→  const days = useMemo(
   155→    () => eachDayOfInterval({ start: range.from, end: range.to }),
   156→    [range]
   157→  );
   158→
   159→  // —— Вид «День»: визиты выбранного дня, разбитые по часам ——
   160→  const dayVisits = useMemo(
   161→    () => (mode === "day" ? (grouped.get(iso(cursor)) ?? []) : []),
   162→    [mode, grouped, cursor]
   163→  );
   164→
   165→  // Часы сетки: 07:00–21:00 + часы, в которых есть визиты вне диапазона
   166→  const dayHours = useMemo(() => {
   167→    const base: number[] = [];
   168→    for (let h = DAY_START_HOUR; h <= DAY_END_HOUR; h++) base.push(h);
   169→    const extra = dayVisits
   170→      .map((v) => getHours(new Date(v.date)))
   171→      .filter((h) => h < DAY_START_HOUR || h > DAY_END_HOUR);
   172→    return [...new Set([...base, ...extra])].sort((a, b) => a - b);
   173→  }, [dayVisits]);
   174→
   175→  const visitsByHour = useMemo(() => {
   176→    const map = new Map<number, VisitItem[]>();
   177→    [...dayVisits]
   178→      .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   179→      .forEach((v) => {
   180→        const h = getHours(new Date(v.date));
   181→        const arr = map.get(h) ?? [];
   182→        arr.push(v);
   183→        map.set(h, arr);
   184→      });
   185→    return map;
   186→  }, [dayVisits]);
   187→
   188→  // Автопрокрутка сетки дня к текущему часу (без setState — прямой DOM-скролл)
   189→  const dayScrollRef = useRef<HTMLDivElement | null>(null);
   190→  useEffect(() => {
   191→    const el = dayScrollRef.current;
   192→    if (!el || mode !== "day" || !isToday(cursor)) return;
   193→    const y = (getHours(new Date()) - DAY_START_HOUR) * SLOT_HEIGHT;
   194→    el.scrollTop = Math.max(0, y - SLOT_HEIGHT * 2);
   195→  }, [mode, cursor]);
   196→
   197→  const step = (dir: 1 | -1) =>
   198→    setCursor(
   199→      mode === "month"
   200→        ? addMonths(cursor, dir)
   201→        : mode === "week"
   202→          ? addWeeks(cursor, dir)
   203→          : addDays(cursor, dir)
   204→    );
   205→
   206→  // —— Диалог: открытие ——
   207→  function openCreateForDay(day: Date, time?: string) {
   208→    setEditing(null);
   209→    setDialogDate(day);
   210→    setDialogTime(time);
   211→    setDialogOpen(true);
   212→  }
   213→  function openEdit(v: VisitItem) {
   214→    setEditing(v);
   215→    setDialogOpen(true);
   216→  }
   217→
   218→  const periodTitle =
   219→    mode === "month"
   220→      ? cap(format(cursor, "LLLL yyyy", { locale: ru }))
   221→      : mode === "week"
   222→        ? `${cap(format(range.from, "d MMMM", { locale: ru }))} — ${format(
   223→            range.to,
   224→            "d MMMM yyyy",
   225→            { locale: ru }
   226→          )}`
   227→        : cap(format(cursor, "EEEE, d MMMM yyyy", { locale: ru }));
   228→
   229→  const noVisits =
   230→    !visitsQuery.isLoading && !visitsQuery.error && (visitsQuery.data ?? []).length === 0;
   231→
   232→  return (
   233→    <div className="space-y-6">
   234→      {/* —— Фильтры —— */}
   235→      <SectionFilters
   236→        mps={mps}
   237→        mpFilter={mpFilter}
   238→        setMpFilter={setMpFilter}
   239→        statusFilter={statusFilter}
   240→        setStatusFilter={setStatusFilter}
   241→        doctorOptions={doctorOptions}
   242→        doctorFilter={doctorFilter}
   243→        setDoctorFilter={setDoctorFilter}
   244→        onCreate={() => openCreateForDay(new Date())}
   245→      />
   246→
   247→      {/* —— Календарь —— */}
   248→      <section className="rounded-xl border border-border bg-white p-6 shadow-[0_2px_12px_rgba(25,70,136,0.05)]">
   249→        {/* Навигация периода */}
   250→        <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
   251→          <div className="flex items-center gap-2">
   252→            <Button
   253→              variant="ghost"
   254→              size="icon"
   255→              className="h-8 w-8 text-brand hover:bg-sky-light"
   256→              onClick={() => step(-1)}
   257→              aria-label="Предыдущий период"
   258→            >
   259→              <ChevronLeft className="h-4 w-4" strokeWidth={2} />
   260→            </Button>
   261→            <h2 className="font-display text-lg font-bold text-brand min-w-[190px] text-center">
   262→              {periodTitle}
   263→            </h2>
   264→            <Button
   265→              variant="ghost"
   266→              size="icon"
   267→              className="h-8 w-8 text-brand hover:bg-sky-light"
   268→              onClick={() => step(1)}
   269→              aria-label="Следующий период"
   270→            >
   271→              <ChevronRight className="h-4 w-4" strokeWidth={2} />
   272→            </Button>
   273→            <Button
   274→              variant="outline"
   275→              size="sm"
   276→              className="ml-2 h-8 border-sky/40 text-brand hover:bg-sky-light"
   277→              onClick={() => setCursor(new Date())}
   278→            >
   279→              Сегодня
   280→            </Button>
   281→          </div>
   282→          <ToggleGroup
   283→            type="single"
   284→            size="sm"
   285→            value={mode}
   286→            onValueChange={(v) => {
   287→              if (v === "month" || v === "week" || v === "day") setMode(v);
   288→            }}
   289→            className="rounded-lg border border-border bg-soft p-1"
   290→          >
   291→            <ToggleGroupItem
   292→              value="month"
   293→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   294→            >
   295→              <CalendarDays className="h-3.5 w-3.5" strokeWidth={2} /> Месяц
   296→            </ToggleGroupItem>
   297→            <ToggleGroupItem
   298→              value="week"
   299→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   300→            >
   301→              <CalendarRange className="h-3.5 w-3.5" strokeWidth={2} /> Неделя
   302→            </ToggleGroupItem>
   303→            <ToggleGroupItem
   304→              value="day"
   305→              className="gap-1.5 rounded-md px-3 text-xs data-[state=on]:bg-sky-light data-[state=on]:text-brand"
   306→            >
   307→              <CalendarClock className="h-3.5 w-3.5" strokeWidth={2} /> День
   308→            </ToggleGroupItem>
   309→          </ToggleGroup>
   310→        </div>
   311→
   312→        {visitsQuery.error ? (
   313→          <div className="rounded-xl border border-terra/50 bg-terra-light p-6 text-sm text-brand-dark">
   314→            Не удалось загрузить визиты: {(visitsQuery.error as Error).message}
   315→          </div>
   316→        ) : visitsQuery.isLoading ? (
   317→          <RowsSkeleton rows={6} />
   318→        ) : (
   319→          <>
   320→            {noVisits && (
   321→              <EmptyState
   322→                title="Визитов в выбранном периоде нет"
   323→                hint="Кликните по дню или свободному часу в календаре, либо нажмите «Запланировать визит»"
   324→                className="py-6"
   325→              />
   326→            )}
   327→
   328→            {mode === "month" ? (
   329→              <>
   330→                {/* Шапка дней недели */}
   331→                <div className="mb-2 grid grid-cols-7 gap-2">
   332→                  {WEEKDAYS.map((w) => (
   333→                    <div
   334→                      key={w}
   335→                      className="text-center text-[11px] font-semibold uppercase tracking-wide text-sky"
   336→                    >
   337→                      {w}
   338→                    </div>
   339→                  ))}
   340→                </div>
   341→                {/* Сетка месяца */}
   342→                <div className="grid grid-cols-7 gap-2">
   343→                  {days.map((day) => {
   344→                    const key = iso(day);
   345→                    const list = grouped.get(key) ?? [];
   346→                    const dim = !isSameMonth(day, cursor);
   347→                    return (
   348→                      <div
   349→                        key={key}
   350→                        onClick={() => openCreateForDay(day)}
   351→                        className={cn(
   352→                          "group/cell flex min-h-[112px] cursor-pointer flex-col gap-1 rounded-lg border p-1.5 transition-colors",
   353→                          dim ? "border-border/60 bg-soft" : "border-border bg-white hover:border-sky/50",
   354→                          isToday(day) && "border-azure/60 bg-sky-light/50 ring-1 ring-azure"
   355→                        )}
   356→                      >
   357→                        <div className="flex items-center justify-between">
   358→                          <span
   359→                            className={cn(
   360→                              "inline-flex h-5 w-5 items-center justify-center rounded-full font-display text-[11px] font-bold",
   361→                              isToday(day)
   362→                                ? "bg-azure text-white"
   363→                                : dim
   364→                                  ? "text-muted-foreground/50"
   365→                                  : "text-brand"
   366→                            )}
   367→                          >
   368→                            {day.getDate()}
   369→                          </span>
   370→                          <button
   371→                            type="button"
   372→                            onClick={(e) => {
   373→                              e.stopPropagation();
   374→                              openCreateForDay(day);
   375→                            }}
   376→                            className="text-sky opacity-0 transition-opacity hover:text-azure group-hover/cell:opacity-100"
   377→                            aria-label={`Запланировать визит на ${key}`}
   378→                          >
   379→                            <Plus className="h-3.5 w-3.5" strokeWidth={2} />
   380→                          </button>
   381→                        </div>
   382→                        <div className="flex flex-col gap-1">
   383→                          {list.slice(0, 3).map((v) => (
   384→                            <button
   385→                              key={v.id}
   386→                              type="button"
   387→                              onClick={(e) => {
   388→                                e.stopPropagation();
   389→                                openEdit(v);
   390→                              }}
   391→                              title={`${fmtTime(v.date)} · ${v.doctor?.fullName ?? "—"} · ${v.purpose}`}
   392→                              className={cn(
   393→                                "w-full truncate rounded-md px-1.5 py-0.5 text-left text-[10px] font-medium text-white transition-opacity",
   394→                                CHIP_BG[v.status],
   395→                                dim && "opacity-60"
   396→                              )}
   397→                            >
   398→                              {fmtTime(v.date)} {v.doctor?.fullName ?? "Врач"}
   399→                            </button>
   400→                          ))}
   401→                          {list.length > 3 && (
   402→                            <span className="px-1 font-display text-[10px] font-semibold text-sky">
   403→                              +{list.length - 3} ещё
   404→                            </span>
   405→                          )}
   406→                        </div>
   407→                      </div>
   408→                    );
   409→                  })}
   410→                </div>
   411→              </>
   412→            ) : mode === "day" ? (
   413→              /* —— Сетка дня по часам (07:00–21:00) —— */
   414→              <div ref={dayScrollRef} className="max-h-[600px] overflow-y-auto rounded-lg border border-border">
   415→                {dayHours.map((h) => {
   416→                  const list = visitsByHour.get(h) ?? [];
   417→                  const isCurrent = isToday(cursor) && getHours(new Date()) === h;
   418→                  return (
   419→                    <div key={h} className="flex border-b border-border/60 last:border-b-0">
   420→                      {/* Метка часа */}
   421→                      <div
   422→                        className={cn(
   423→                          "w-16 shrink-0 border-r border-border/60 px-2 py-1.5 text-right font-display text-[11px] font-semibold",
   424→                          isCurrent ? "bg-sky-light text-azure" : "bg-soft/60 text-sky"
   425→                        )}
   426→                      >
   427→                        {pad2(h)}:00
   428→                      </div>
   429→                      {/* Слот часа */}
   430→                      <div
   431→                        onClick={() => openCreateForDay(cursor, `${pad2(h)}:00`)}
   432→                        className={cn(
   433→                          "group/slot relative min-h-[56px] flex-1 cursor-pointer p-1.5 transition-colors hover:bg-sky-light/40",
   434→                          isCurrent && "bg-sky-light/50"
   435→                        )}
   436→                      >
   437→                        {/* Индикатор текущего времени */}
   438→                        {isCurrent && (
   439→                          <div
   440→                            className="pointer-events-none absolute left-0 right-0 z-10 flex items-center"
   441→                            style={{ top: `${(getMinutes(new Date()) / 60) * 100}%` }}
   442→                          >
   443→                            <span className="-ml-1 h-2 w-2 rounded-full bg-azure" />
   444→                            <span className="h-px flex-1 bg-azure/60" />
   445→                          </div>
   446→                        )}
   447→                        <Plus
   448→                          className="pointer-events-none absolute right-2 top-2 h-3.5 w-3.5 text-sky opacity-0 transition-opacity group-hover/slot:opacity-100"
   449→                          strokeWidth={2}
   450→                          aria-hidden
   451→                        />
   452→                        {list.length === 0 ? (
   453→                          <span className="select-none text-[11px] text-transparent transition-colors group-hover/slot:text-muted-foreground/60">
   454→                            Запланировать на {pad2(h)}:00
   455→                          </span>
   456→                        ) : (
   457→                          <div className="flex flex-col gap-1.5">
   458→                            {list.map((v) => (
   459→                              <button
   460→                                key={v.id}
   461→                                type="button"
   462→                                onClick={(e) => {
   463→                                  e.stopPropagation();
   464→                                  openEdit(v);
   465→                                }}
   466→                                className="flex w-full items-center justify-between gap-3 rounded-lg border border-border bg-white p-2.5 text-left shadow-[0_1px_6px_rgba(25,70,136,0.06)] transition-colors hover:border-azure/60"
   467→                              >
   468→                                <div className="min-w-0">
   469→                                  <p className="truncate text-xs font-semibold text-brand">
   470→                                    <span className="font-display text-[11px] font-bold text-azure">
   471→                                      {fmtTime(v.date)}
   472→                                    </span>{" "}
   473→                                    {v.doctor?.fullName ?? "—"}
   474→                                  </p>
   475→                                  <p className="mt-0.5 truncate text-[11px] text-muted-foreground">
   476→                                    {v.purpose}
   477→                                  </p>
   478→                                  {v.lpu && (
   479→                                    <p className="mt-0.5 truncate text-[10px] text-sky">
   480→                                      {v.lpu.name}
   481→                                    </p>
   482→                                  )}
   483→                                </div>
   484→                                <div className="shrink-0">
   485→                                  <StatusChip status={v.status} />
   486→                                </div>
   487→                              </button>
   488→                            ))}
   489→                          </div>
   490→                        )}
   491→                      </div>
   492→                    </div>
   493→                  );
   494→                })}
   495→              </div>
   496→            ) : (
   497→              /* —— Сетка недели —— */
   498→              <div className="grid grid-cols-7 gap-3">
   499→                {days.map((day) => {
   500→                  const key = iso(day);
   501→                  const list = grouped.get(key) ?? [];
   502→                  return (
   503→                    <div
   504→                      key={key}
   505→                      className={cn(
   506→                        "flex min-h-[400px] flex-col rounded-lg border",
   507→                        isToday(day)
   508→                          ? "border-azure/60 bg-sky-light/40"
   509→                          : "border-border bg-white"
   510→                      )}
   511→                    >
   512→                      <div className="border-b border-border/70 px-2 py-2 text-center">
   513→                        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   514→                          {cap(format(day, "EEEEEE", { locale: ru }))}
   515→                        </div>
   516→                        <div
   517→                          className={cn(
   518→                            "font-display text-sm font-bold",
