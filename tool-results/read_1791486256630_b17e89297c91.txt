   155→   155→function CitiesTab() {
   156→   156→  const qc = useQueryClient();
   157→   157→  const { data: session } = useSession();
   158→   158→  const isMP = session?.user?.role === "MP";
   159→   159→  const [selectedId, setSelectedId] = useState<string | null>(null);
   160→   160→  const [search, setSearch] = useState("");
   161→   161→  const [dialog, setDialog] = useState<CityDialogState>(null);
   162→   162→
   163→   163→  const { data: cities, isLoading } = useQuery<City[]>({
   164→   164→    queryKey: ["cities"],
   165→   165→    queryFn: () => api.get("/api/app/cities"),
   166→   166→  });
   167→   167→
   168→   168→  const filtered = useMemo(() => {
   169→   169→    const q = search.trim().toLowerCase();
   170→   170→    const list = cities ?? [];
   171→   171→    return q ? list.filter((c) => c.name.toLowerCase().includes(q)) : list;
   172→   172→  }, [cities, search]);
   173→   173→
   174→   174→  const selected = (cities ?? []).find((c) => c.id === selectedId) ?? null;
   175→   175→
   176→   176→  const remove = useMutation({
   177→   177→    mutationFn: (id: string) => api.del(`/api/app/cities?id=${id}`),
   178→   178→    onSuccess: () => {
   179→   179→      toast.success("Город удален");
   180→   180→      setSelectedId(null);
   181→   181→      qc.invalidateQueries({ queryKey: ["cities"] });
   182→   182→    },
   183→   183→    onError: (e: Error) => toast.error(e.message),
   184→   184→  });
   185→   185→
   186→   186→  return (
   187→   187→    <>
   188→   188→      <DictWindow
   189→   189→        icon={<MapPin className="h-4 w-4 text-sky" strokeWidth={2} />}
   190→   190→        title="Города"
   191→   191→        count={filtered.length}
   192→   192→        toolbar={
   193→   193→          <>
   194→   194→            <DictCrudButtons
   195→   195→              onCreate={() => setDialog({ mode: "create" })}
   196→   196→              onEdit={() => selected && setDialog({ mode: "edit", city: selected })}
   197→   197→              canEdit={!!selected}
   198→   198→              deleteSlot={
   199→   199→                isMP ? (
   200→   200→                  <DeleteButton
   201→   201→                    disabled={!selected}
   202→   202→                    title={`Удалить город «${selected?.name ?? ""}»?`}
   203→   203→                    onConfirm={() => {
   204→   204→                      if (selected) remove.mutateAsync(selected.id).catch(() => undefined);
   205→   205→                    }}
   206→   206→                  />
   207→   207→                ) : undefined
   208→   208→              }
   209→   209→            />
   210→   210→            <DictImportExport kind="cities" />
   211→   211→            <OnecSearch value={search} onChange={setSearch} placeholder="Поиск по названию города…" />
   212→   212→          </>
   213→   213→        }
   214→   214→      >
   215→   215→        {isLoading ? (
   216→   216→          <div className="p-4"><RowsSkeleton /></div>
   217→   217→        ) : filtered.length === 0 ? (
   218→   218→          <EmptyState
   219→   219→            className="py-8"
   220→   220→            title={search ? "Ничего не найдено" : "Городов пока нет"}
   221→   221→            hint={search ? "Измените условие поиска" : "Нажмите «Создать» или импортируйте список из файла"}
   222→   222→          />
   223→   223→        ) : (
   224→   224→          <div className="max-h-96 overflow-y-auto">
   225→   225→            <table className="onec-table">
   226→   226→              <thead>
   227→   227→                <tr>
   228→   228→                  <th>Название</th>
   229→   229→                  <th>ЛПУ</th>
   230→   230→                  <th>НВ</th>
   231→   231→                </tr>
   232→   232→              </thead>
   233→   233→              <tbody>
   234→   234→                {filtered.map((c) => (
   235→   235→                  <tr
   236→   236→                    key={c.id}
   237→   237→                    onClick={() => setSelectedId(c.id)}
   238→   238→                    className={cn("cursor-pointer", selectedId === c.id && "onec-row-selected")}
   239→   239→                  >
   240→   240→                    <td>
   241→   241→                      <button
   242→   242→                        type="button"
   243→   243→                        className="font-medium text-brand hover:underline text-left"
   244→   244→                        title="Изменить"
   245→   245→                        onClick={(e) => {
   246→   246→                          e.stopPropagation();
   247→   247→                          setDialog({ mode: "edit", city: c });
   248→   248→                        }}
   249→   249→                      >
   250→   250→                        {c.name}
   251→   251→                      </button>
   252→   252→                    </td>
   253→   253→                    <td>{c._count?.lpus ?? 0}</td>
   254→   254→                    <td>{c._count?.doctors ?? 0}</td>
   255→   255→                  </tr>
   256→   256→                ))}
   257→   257→              </tbody>
   258→   258→            </table>
   259→   259→          </div>
   260→   260→        )}
   261→   261→      </DictWindow>
   262→   262→
   263→   263→      {dialog && (
   264→   264→        <CityDialog
   265→   265→          editing={dialog.mode === "edit" ? dialog.city : null}
   266→   266→          onClose={() => setDialog(null)}
   267→   267→        />
   268→   268→      )}
   269→   269→    </>

   273→   273→function CityDialog({ editing, onClose }: { editing: City | null; onClose: () => void }) {
   274→   274→  const qc = useQueryClient();
   275→   275→  const [name, setName] = useState(() => editing?.name ?? "");
   276→   276→
   277→   277→  const save = useMutation({
   278→   278→    mutationFn: () =>
   279→   279→      editing
   280→   280→        ? api.put("/api/app/cities", { id: editing.id, name })
   281→   281→        : api.post("/api/app/cities", { name }),
   282→   282→    onSuccess: () => {
   283→   283→      toast.success(editing ? "Город обновлен" : "Город добавлен");
   284→   284→      qc.invalidateQueries({ queryKey: ["cities"] });
   285→   285→      onClose();
   286→   286→    },
   287→   287→    onError: (e: Error) => toast.error(e.message),
   288→   288→  });
   289→   289→
   290→   290→  return (
   291→   291→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   292→   292→      <DialogContent className="max-h-[calc(100dvh-4rem)] overflow-y-auto sm:max-w-md">
   293→   293→        <DialogHeader>
   294→   294→          <DialogTitle className="font-display text-brand">
   295→   295→            {editing ? "Редактирование города" : "Новый город"}
   296→   296→          </DialogTitle>
   297→   297→          <DialogDescription>
   298→   298→            {editing ? "Измените название и нажмите «Сохранить»." : "Введите название города — он станет доступен для ЛПУ и врачей."}
   299→   299→          </DialogDescription>
   300→   300→        </DialogHeader>
   301→   301→        <div className="space-y-4 py-2">
   302→   302→          <div className="space-y-1.5">
   303→   303→            <Label>Название</Label>
   304→   304→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Например: Казань" autoFocus />
   305→   305→          </div>
   306→   306→        </div>
   307→   307→        <DialogFooter>
   308→   308→          <Button variant="outline" onClick={onClose}>Отмена</Button>
   309→   309→          <Button onClick={() => save.mutate()} disabled={!name.trim() || save.isPending}>
   310→   310→            {editing ? "Сохранить" : "Добавить"}
   311→   311→          </Button>
   312→   312→        </DialogFooter>
   313→   313→      </DialogContent>
   314→   314→    </Dialog>
   315→   315→  );
   316→   316→}
